     1→'use client'
     2→
     3→import { useState, useEffect, useMemo } from 'react'
     4→import { toast } from 'sonner'
     5→import { Loader2, Plus, Trash2, Check, ChevronsUpDown } from 'lucide-react'
     6→
     7→import { Button } from '@/components/ui/button'
     8→import { Input } from '@/components/ui/input'
     9→import { Textarea } from '@/components/ui/textarea'
    10→import { Label } from '@/components/ui/label'
    11→import { Badge } from '@/components/ui/badge'
    12→import { Separator } from '@/components/ui/separator'
    13→import {
    14→  Select,
    15→  SelectContent,
    16→  SelectItem,
    17→  SelectTrigger,
    18→  SelectValue,
    19→} from '@/components/ui/select'
    20→import {
    21→  Popover,
    22→  PopoverContent,
    23→  PopoverTrigger,
    24→} from '@/components/ui/popover'
    25→import {
    26→  Command,
    27→  CommandEmpty,
    28→  CommandGroup,
    29→  CommandInput,
    30→  CommandItem,
    31→  CommandList,
    32→} from '@/components/ui/command'
    33→import { cn } from '@/lib/utils'
    34→
    35→interface Proveedor {
    36→  id: number
    37→  nombre: string
    38→  apellido: string
    39→  razon_social: string | null
    40→}
    41→
    42→interface EstadoGeneral {
    43→  id: number
    44→  nombre_estado: string
    45→  entidad_aplicable: string | null
    46→}
    47→
    48→interface ProductoItem {
    49→  id: number
    50→  nombre: string
    51→  id_unidad_base: number
    52→  unidadBase: { id: number; codigo: string; nombre: string }
    53→  precio_compra_referencia: number
    54→  tipo: 'mp' | 'insumo'
    55→}
    56→
    57→interface DetalleRow {
    58→  key: string
    59→  tipo: 'mp' | 'insumo'
    60→  id_producto: string
    61→  cantidad: string
    62→  unidad: string
    63→  unidadCodigo: string
    64→  precioEstimado: string
    65→}
    66→
    67→interface PedidoProveedorFormProps {
    68→  pedido?: any | null
    69→  onSuccess: () => void
    70→  onCancel: () => void
    71→}
    72→
    73→export default function PedidoProveedorForm({ pedido, onSuccess, onCancel }: PedidoProveedorFormProps) {
    74→  const isEditing = !!pedido
    75→
    76→  // Form fields
    77→  const [idProveedor, setIdProveedor] = useState('')
    78→  const [fechaPedido, setFechaPedido] = useState('')
    79→  const [fechaEntregaEstimada, setFechaEntregaEstimada] = useState('')
    80→  const [idEstado, setIdEstado] = useState('')
    81→  const [observaciones, setObservaciones] = useState('')
    82→
    83→  // Detail rows
    84→  const [detalles, setDetalles] = useState<DetalleRow[]>([])
    85→
    86→  // Data
    87→  const [proveedores, setProveedores] = useState<Proveedor[]>([])
    88→  const [estados, setEstados] = useState<EstadoGeneral[]>([])
    89→  const [materiasPrimas, setMateriasPrimas] = useState<ProductoItem[]>([])
    90→  const [insumos, setInsumos] = useState<ProductoItem[]>([])
    91→  const [submitting, setSubmitting] = useState(false)
    92→  const [loadingData, setLoadingData] = useState(true)
    93→
    94→  // Combobox open states
    95→  const [proveedorOpen, setProveedorOpen] = useState(false)
    96→
    97→  // Load data
    98→  useEffect(() => {
    99→    async function loadData() {
   100→      try {
   101→        const [provRes, estRes, mpRes, insRes] = await Promise.all([
   102→          fetch('/api/personas?tipo=proveedor&limite=100'),
   103→          fetch('/api/estados-generales?entidad_aplicable=pedido_proveedor'),
   104→          fetch('/api/materias-primas?limite=200&estado=true'),
   105→          fetch('/api/insumos?limite=200&estado=true'),
   106→        ])
   107→
   108→        const provData = await provRes.json()
   109→        const estData = await estRes.json()
   110→        const mpData = await mpRes.json()
   111→        const insData = await insRes.json()
   112→
   113→        setProveedores(provData.personas || [])
   114→        setEstados(Array.isArray(estData) ? estData : [])
   115→        setMateriasPrimas(
   116→          (mpData.data || []).map((mp: any) => ({
   117→            id: mp.id,
   118→            nombre: mp.nombre,
   119→            id_unidad_base: mp.id_unidad_base,
   120→            unidadBase: mp.unidadBase,
   121→            precio_compra_referencia: mp.precio_compra_referencia,
   122→            tipo: 'mp' as const,
   123→          }))
   124→        )
   125→        setInsumos(
   126→          (insData.data || []).map((ins: any) => ({
   127→            id: ins.id,
   128→            nombre: ins.nombre,
   129→            id_unidad_base: ins.id_unidad_base,
   130→            unidadBase: ins.unidadBase,
   131→            precio_compra_referencia: ins.precio_compra_referencia,
   132→            tipo: 'insumo' as const,
   133→          }))
   134→        )
   135→      } catch {
   136→        toast.error('Error al cargar datos del formulario')
   137→      } finally {
   138→        setLoadingData(false)
   139→      }
   140→    }
   141→    loadData()
   142→  }, [])
   143→
   144→  // Populate form when editing
   145→  useEffect(() => {
   146→    if (pedido) {
   147→      setIdProveedor(pedido.id_proveedor?.toString() || '')
   148→      setFechaPedido(pedido.fecha_pedido ? pedido.fecha_pedido.split('T')[0] : '')
   149→      setFechaEntregaEstimada(
   150→        pedido.fecha_entrega_estimada ? pedido.fecha_entrega_estimada.split('T')[0] : ''
   151→      )
   152→      setIdEstado(pedido.id_estado?.toString() || '')
   153→      setObservaciones(pedido.observaciones || '')
   154→
   155→      if (pedido.detalle && Array.isArray(pedido.detalle)) {
   156→        setDetalles(
   157→          pedido.detalle.map((d: any) => ({
   158→            key: `row-${d.id}`,
   159→            tipo: d.id_materia_prima ? 'mp' : 'insumo',
   160→            id_producto: (d.id_materia_prima || d.id_insumo)?.toString() || '',
   161→            cantidad: d.cantidad_pedida?.toString() || '',
   162→            unidad: d.unidad?.id?.toString() || '',
   163→            unidadCodigo: d.unidad?.codigo || '',
   164→            precioEstimado: d.precio_estimado?.toString() || '',
   165→          }))
   166→        )
   167→      }
   168→    } else {
   169→      // Defaults for new
   170→      setFechaPedido(new Date().toISOString().split('T')[0])
   171→      addDetailRow()
   172→    }
   173→  }, [pedido])
   174→
   175→  // Products filtered by selected tipo for each row
   176→  const getProductsForTipo = (tipo: 'mp' | 'insumo') => {
   177→    return tipo === 'mp' ? materiasPrimas : insumos
   178→  }
   179→
   180→  // Find product by id
   181→  const findProduct = (tipo: 'mp' | 'insumo', id: string) => {
   182→    const products = getProductsForTipo(tipo)
   183→    return products.find((p) => p.id.toString() === id)
   184→  }
   185→
   186→  // Add detail row
   187→  const addDetailRow = () => {
   188→    setDetalles((prev) => [
   189→      ...prev,
   190→      {
   191→        key: `row-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   192→        tipo: 'mp',
   193→        id_producto: '',
   194→        cantidad: '',
   195→        unidad: '',
   196→        unidadCodigo: '',
   197→        precioEstimado: '',
   198→      },
   199→    ])
   200→  }
   201→
   202→  // Remove detail row
   203→  const removeDetailRow = (key: string) => {
   204→    setDetalles((prev) => prev.filter((d) => d.key !== key))
   205→  }
   206→
   207→  // Update detail row
   208→  const updateDetalle = (key: string, field: keyof DetalleRow, value: string) => {
   209→    setDetalles((prev) =>
   210→      prev.map((d) => {
   211→        if (d.key !== key) return d
   212→
   213→        const updated = { ...d, [field]: value }
   214→
   215→        // When tipo changes, reset producto and unidad
   216→        if (field === 'tipo') {
   217→          updated.id_producto = ''
   218→          updated.unidad = ''
   219→          updated.unidadCodigo = ''
   220→          updated.precioEstimado = ''
   221→        }
   222→
   223→        // When product changes, auto-fill unidad and precio_referencia
   224→        if (field === 'id_producto' && value) {
   225→          const product = findProduct(updated.tipo, value)
   226→          if (product) {
   227→            updated.unidad = product.unidadBase?.id?.toString() || ''
   228→            updated.unidadCodigo = product.unidadBase?.codigo || ''
   229→            updated.precioEstimado = product.precio_compra_referencia?.toString() || ''
   230→          }
   231→        }
   232→
   233→        return updated
   234→      })
   235→    )
   236→  }
   237→
   238→  // Calculate total estimado
   239→  const totalEstimado = useMemo(() => {
   240→    return detalles.reduce((sum, d) => {
   241→      const cant = parseFloat(d.cantidad)
   242→      const precio = parseFloat(d.precioEstimado)
   243→      if (isNaN(cant) || isNaN(precio)) return sum
   244→      return sum + cant * precio
   245→    }, 0)
   246→  }, [detalles])
   247→
   248→  const formatCurrency = (value: number) => {
   249→    return new Intl.NumberFormat('es-AR', { style: 'currency', currency: 'ARS' }).format(value)
   250→  }
   251→
   252→  // Submit
   253→  const handleSubmit = async () => {
   254→    // Validation
   255→    if (!isEditing) {
   256→      if (!idProveedor) {
   257→        toast.error('Seleccioná un proveedor')
   258→        return
   259→      }
   260→      if (!fechaPedido) {
   261→        toast.error('Ingresá la fecha del pedido')
   262→        return
   263→      }
   264→      if (detalles.length === 0) {
   265→        toast.error('Agregá al menos un detalle al pedido')
   266→        return
   267→      }
   268→      for (let i = 0; i < detalles.length; i++) {
   269→        const d = detalles[i]
   270→        if (!d.id_producto) {
   271→          toast.error(`Fila ${i + 1}: Seleccioná un producto`)
   272→          return
   273→        }
   274→        if (!d.cantidad || parseFloat(d.cantidad) <= 0) {
   275→          toast.error(`Fila ${i + 1}: Ingresá una cantidad válida`)
   276→          return
   277→        }
   278→      }
   279→    }
   280→
   281→    setSubmitting(true)
   282→    try {
   283→      const payload = isEditing
   284→        ? {
   285→            id: pedido.id,
   286→            fecha_entrega_estimada: fechaEntregaEstimada || null,
   287→            id_estado: idEstado ? parseInt(idEstado) : undefined,
   288→            observaciones: observaciones.trim() || null,
   289→          }
   290→        : {
   291→            id_proveedor: parseInt(idProveedor),
   292→            fecha_pedido: fechaPedido,
   293→            fecha_entrega_estimada: fechaEntregaEstimada || null,
   294→            observaciones: observaciones.trim() || null,
   295→            total_estimado: totalEstimado,
   296→            detalle: detalles.map((d) => ({
   297→              tipo: d.tipo,
   298→              id_producto: parseInt(d.id_producto),
   299→              cantidad_pedida: parseFloat(d.cantidad),
   300→              id_unidad: parseInt(d.unidad),
   301→              precio_estimado: parseFloat(d.precioEstimado) || 0,
   302→            })),
   303→          }
   304→
   305→      const url = isEditing ? `/api/pedidos-proveedores/${pedido.id}` : '/api/pedidos-proveedores'
   306→      const method = isEditing ? 'PUT' : 'POST'
   307→
   308→      const res = await fetch(url, {
   309→        method,
   310→        headers: { 'Content-Type': 'application/json' },
   311→        body: JSON.stringify(payload),
   312→      })
   313→
   314→      if (!res.ok) {
   315→        const errorData = await res.json().catch(() => ({}))
   316→        throw new Error(errorData.error || 'Error al guardar pedido')
   317→      }
   318→
   319→      toast.success(isEditing ? 'Pedido actualizado' : 'Pedido registrado', {
   320→        description: isEditing
   321→          ? 'Los cambios se guardaron correctamente'
   322→          : 'El nuevo pedido se registró exitosamente',
   323→      })
   324→      onSuccess()
   325→    } catch (error: any) {
   326→      toast.error('Error al guardar', {
   327→        description: error.message || 'Intentá de nuevo más tarde',
   328→      })
   329→    } finally {
   330→      setSubmitting(false)
   331→    }
   332→  }
   333→
   334→  if (loadingData) {
   335→    return (
   336→      <div className="flex items-center justify-center py-8">
   337→        <Loader2 className="h-6 w-6 animate-spin text-mostaza" />
   338→        <span className="ml-2 text-muted-foreground">Cargando datos...</span>
   339→      </div>
   340→    )
   341→  }
   342→
   343→  return (
   344→    <div className="space-y-6">
   345→      {/* Header info */}
   346→      {isEditing && (
   347→        <div className="bg-muted/30 rounded-lg p-3 border border-marron/10">
   348→          <p className="text-sm text-muted-foreground">
   349→            Pedido #{pedido.id} — Proveedor: {pedido.proveedor?.razon_social || `${pedido.proveedor?.nombre} ${pedido.proveedor?.apellido}`}
   350→          </p>
   351→        </div>
   352→      )}
   353→
   354→      {/* Main fields */}
   355→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   356→        {/* Proveedor - searchable combobox */}
   357→        <div>
   358→          <Label className="text-sm font-medium text-marron mb-1 block">Proveedor *</Label>
   359→          {isEditing ? (
   360→            <Input
   361→              value={pedido.proveedor?.razon_social || `${pedido.proveedor?.nombre} ${pedido.proveedor?.apellido}`}
   362→              disabled
   363→              className="bg-muted/50"
   364→            />
   365→          ) : (
   366→            <Popover open={proveedorOpen} onOpenChange={setProveedorOpen}>
   367→              <PopoverTrigger asChild>
   368→                <Button
   369→                  variant="outline"
   370→                  role="combobox"
   371→                  aria-expanded={proveedorOpen}
   372→                  className={cn(
   373→                    'w-full justify-between',
   374→                    !idProveedor && 'text-muted-foreground'
   375→                  )}
   376→                >
   377→                  {idProveedor
   378→                    ? (() => {
   379→                        const prov = proveedores.find((p) => p.id.toString() === idProveedor)
   380→                        return prov?.razon_social || `${prov?.nombre} ${prov?.apellido}` || 'Seleccionar...'
   381→                      })()
   382→                    : 'Buscar proveedor...'}
   383→                  <ChevronsUpDown className="ml-2 h-4 w-4 shrink-0 opacity-50" />
   384→                </Button>
   385→              </PopoverTrigger>
   386→              <PopoverContent className="w-full p-0" align="start">
   387→                <Command>
   388→                  <CommandInput placeholder="Buscar proveedor..." />
   389→                  <CommandList>
   390→                    <CommandEmpty>No se encontró proveedor</CommandEmpty>
   391→                    <CommandGroup>
   392→                      {proveedores.map((prov) => (
   393→                        <CommandItem
   394→                          key={prov.id}
   395→                          value={prov.razon_social || `${prov.nombre} ${prov.apellido}`}
   396→                          onSelect={() => {
   397→                            setIdProveedor(prov.id.toString())
   398→                            setProveedorOpen(false)
   399→                          }}
   400→                        >
   401→                          <Check
   402→                            className={cn(
   403→                              'mr-2 h-4 w-4',
   404→                              idProveedor === prov.id.toString() ? 'opacity-100' : 'opacity-0'
   405→                            )}
   406→                          />
   407→                          {prov.razon_social || `${prov.nombre} ${prov.apellido}`}
   408→                        </CommandItem>
   409→                      ))}
   410→                    </CommandGroup>
   411→                  </CommandList>
   412→                </Command>
   413→              </PopoverContent>
   414→            </Popover>
   415→          )}
   416→        </div>
   417→
   418→        {/* Fecha pedido */}
   419→        <div>
   420→          <Label className="text-sm font-medium text-marron mb-1 block">Fecha Pedido *</Label>
   421→          <Input
   422→            type="date"
   423→            value={fechaPedido}
   424→            onChange={(e) => setFechaPedido(e.target.value)}
   425→            disabled={isEditing}
   426→          />
   427→        </div>
   428→
   429→        {/* Fecha entrega estimada */}
   430→        <div>
   431→          <Label className="text-sm font-medium text-marron mb-1 block">Entrega Estimada</Label>
   432→          <Input
   433→            type="date"
   434→            value={fechaEntregaEstimada}
   435→            onChange={(e) => setFechaEntregaEstimada(e.target.value)}
   436→          />
   437→        </div>
   438→
   439→        {/* Estado (only in edit mode) */}
   440→        {isEditing && (
   441→          <div>
   442→            <Label className="text-sm font-medium text-marron mb-1 block">Estado</Label>
   443→            <Select value={idEstado} onValueChange={setIdEstado}>
   444→              <SelectTrigger>
   445→                <SelectValue placeholder="Seleccionar..." />
   446→              </SelectTrigger>
   447→              <SelectContent>
   448→                {estados.map((est) => (
   449→                  <SelectItem key={est.id} value={est.id.toString()}>
   450→                    {est.nombre_estado}
   451→                  </SelectItem>
   452→                ))}
   453→              </SelectContent>
   454→            </Select>
   455→          </div>
   456→        )}
   457→      </div>
   458→
   459→      {/* Observaciones */}
   460→      <div>
   461→        <Label className="text-sm font-medium text-marron mb-1 block">Observaciones</Label>
   462→        <Textarea
   463→          placeholder="Observaciones sobre el pedido..."
   464→          className="resize-none"
   465→          rows={2}
   466→          value={observaciones}
   467→          onChange={(e) => setObservaciones(e.target.value)}
   468→        />
   469→      </div>
   470→
   471→      {/* Detail rows - only in create mode, or show read-only in edit */}
   472→      {!isEditing && (
   473→        <>
   474→          <Separator />
   475→          <div className="space-y-4">
   476→            <Label className="text-sm font-semibold text-marron block">Detalle del Pedido</Label>
   477→
   478→            {/* Detail rows - fills 100% width, no horizontal scroll */}
   479→            <div className="space-y-3">
   480→              {/* Desktop column headers (aligned with rows) */}
   481→              <div className="hidden lg:grid lg:grid-cols-[1.4fr_3fr_1fr_1fr_1.2fr_44px] gap-3 px-3">
   482→                <div className="text-xs font-semibold text-muted-foreground">Tipo</div>
   483→                <div className="text-xs font-semibold text-muted-foreground">Producto</div>
   484→                <div className="text-xs font-semibold text-muted-foreground">Cantidad</div>
   485→                <div className="text-xs font-semibold text-muted-foreground">Unidad</div>
   486→                <div className="text-xs font-semibold text-muted-foreground">Precio Est.</div>
   487→                <div></div>
   488→              </div>
   489→
   490→              {/* Detail rows */}
   491→              {detalles.map((detalle) => (
   492→                <div
   493→                  key={detalle.key}
   494→                  className="p-3 rounded-lg border border-marron/10 bg-muted/20"
   495→                >
   496→                  <div className="grid grid-cols-2 lg:grid-cols-[1.4fr_3fr_1fr_1fr_1.2fr_44px] gap-3 items-center">
   497→                    {/* Tipo - full width on mobile */}
   498→                    <div className="col-span-2 lg:col-span-1">
   499→                      <Label htmlFor={`tipo-${detalle.key}`} className="text-xs text-muted-foreground mb-1 block lg:sr-only">Tipo</Label>
   500→                      <Select
   501→                        value={detalle.tipo}
   502→                        onValueChange={(v) => updateDetalle(detalle.key, 'tipo', v)}
   503→                      >
   504→                        <SelectTrigger className="h-9">
   505→                          <SelectValue />
   506→                        </SelectTrigger>
   507→                        <SelectContent>
   508→                          <SelectItem value="mp">Materia Prima</SelectItem>
   509→                          <SelectItem value="insumo">Insumo</SelectItem>
   510→                        </SelectContent>
   511→                      </Select>
   512→                    </div>
   513→
   514→                    {/* Producto - full width on mobile */}
   515→                    <div className="col-span-2 lg:col-span-1">
   516→                      <Label htmlFor={`prod-${detalle.key}`} className="text-xs text-muted-foreground mb-1 block lg:sr-only">Producto</Label>
   517→                      <Select
   518→                        value={detalle.id_producto}
   519→                        onValueChange={(v) => updateDetalle(detalle.key, 'id_producto', v)}
   520→                      >
   521→                        <SelectTrigger className="h-9">
   522→                          <SelectValue placeholder="Seleccionar..." />
   523→                        </SelectTrigger>
   524→                        <SelectContent>
   525→                          {getProductsForTipo(detalle.tipo).map((prod) => (
   526→                            <SelectItem key={prod.id} value={prod.id.toString()}>
   527→                              {prod.nombre}
   528→                            </SelectItem>
   529→                          ))}
   530→                        </SelectContent>
   531→                      </Select>
   532→                    </div>
   533→
   534→                    {/* Cantidad */}
   535→                    <div>
   536→                      <Label htmlFor={`cant-${detalle.key}`} className="text-xs text-muted-foreground mb-1 block lg:sr-only">Cantidad</Label>
   537→                      <Input
   538→                        id={`cant-${detalle.key}`}
   539→                        type="number"
   540→                        step="0.01"
   541→                        min="0"
   542→                        placeholder="0"
   543→                        className="h-9"
   544→                        value={detalle.cantidad}
   545→                        onChange={(e) => updateDetalle(detalle.key, 'cantidad', e.target.value)}
   546→                      />
   547→                    </div>
   548→
   549→                    {/* Unidad (auto-filled) */}
   550→                    <div>
   551→                      <Label htmlFor={`uni-${detalle.key}`} className="text-xs text-muted-foreground mb-1 block lg:sr-only">Unidad</Label>
   552→                      <Input
   553→                        id={`uni-${detalle.key}`}
   554→                        className="h-9 bg-muted/50"
   555→                        value={detalle.unidadCodigo || '-'}
   556→                        disabled
   557→                      />

   622→
   623→      {/* Edit mode: show read-only detail summary */}
   624→      {isEditing && pedido.detalle && pedido.detalle.length > 0 && (
