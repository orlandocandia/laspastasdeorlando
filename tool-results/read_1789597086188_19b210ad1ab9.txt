   321→function SupplyFormDialog({ open, mode, item, onClose, onSaved }: { open: boolean; mode: FormMode; item: CmSupply | null; onClose: () => void; onSaved: () => void }) {
   322→  const [form, setForm] = React.useState({
   323→    name: '', description: '', category: '' as string,
   324→    purchaseUnit: 'u' as CmSupplyUnit, purchasePrice: '',
   325→    purchaseUnitType: '' as string, unitsPurchased: '', measurePerUnit: '',
   326→    measureUnit: 'u' as string, totalPrice: '',
   327→    usageUnit: 'u' as string, equivalenceValue: '', equivalenceUnit: 'u' as string,
   328→    image: '' as string, isActive: true,
   329→  })
   330→  const [saving, setSaving] = React.useState(false)
   331→  const [error, setError] = React.useState<string | null>(null)
   332→
   333→  React.useEffect(() => {
   334→    if (open) {
   335→      if (mode === 'edit' && item) {
   336→        setForm({
   337→          name: item.name, description: item.description || '', category: item.category || '',
   338→          purchaseUnit: item.purchaseUnit, purchasePrice: String(item.purchasePrice),
   339→          purchaseUnitType: item.purchaseUnitType || '', unitsPurchased: item.unitsPurchased ? String(item.unitsPurchased) : '',
   340→          measurePerUnit: item.measurePerUnit ? String(item.measurePerUnit) : '',
   341→          measureUnit: item.measureUnit || 'u', totalPrice: item.totalPrice ? String(item.totalPrice) : '',
   342→          usageUnit: item.usageUnit || 'u', equivalenceValue: item.equivalenceValue ? String(item.equivalenceValue) : '',
   343→          equivalenceUnit: item.equivalenceUnit || 'u',
   344→          image: item.image || '', isActive: item.isActive,
   345→        })
   346→      } else {
   347→        setForm({ name: '', description: '', category: '', purchaseUnit: 'u', purchasePrice: '', image: '', isActive: true })
   348→      }
   349→      setError(null)
   350→    }
   351→  }, [open, mode, item])
   352→
   353→  const setField = <K extends keyof typeof form>(key: K, value: (typeof form)[K]) => setForm((f) => ({ ...f, [key]: value }))
   354→
   355→  const handleSubmit = async (e: React.FormEvent) => {
   356→    e.preventDefault()
   357→    setError(null)
   358→    if (!form.name.trim()) return setError('El nombre es obligatorio')
   359→    // Validate totalPrice (the field the user actually fills in)
   360→    const totalPrice = parseFloat(form.totalPrice)
   361→    if (form.totalPrice && (isNaN(totalPrice) || totalPrice < 0)) {
   362→      return setError('El precio total debe ser un número válido')
   363→    }
   364→
   365→    setSaving(true)
   366→    try {
   367→      const body: Record<string, unknown> = {
   368→        name: form.name, description: form.description || null,
   369→        category: form.category || null, purchaseUnit: form.purchaseUnit,
   370→        purchaseUnitType: form.purchaseUnitType || null,
   371→        unitsPurchased: form.unitsPurchased ? parseFloat(form.unitsPurchased) : null,
   372→        measurePerUnit: form.measurePerUnit ? parseFloat(form.measurePerUnit) : null,
   373→        measureUnit: form.measureUnit || null,
   374→        totalPrice: form.totalPrice ? parseFloat(form.totalPrice) : null,
   375→        usageUnit: getAutoUsageUnit(form.purchaseUnitType),
   376→        equivalenceValue: form.equivalenceValue ? parseFloat(form.equivalenceValue) : null,
   377→        equivalenceUnit: form.equivalenceUnit || null,
   378→        image: form.image || null, isActive: form.isActive,
   379→      }
   380→      const url = mode === 'create' ? '/api/cocina-movil/supplies' : `/api/cocina-movil/supplies/${item!.id}`
   381→      const res = await fetch(url, { method: mode === 'create' ? 'POST' : 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
   382→      const data = await res.json().catch(() => ({}))
   383→      if (!res.ok) throw new Error(data.error || 'HTTP ' + res.status)
   384→      toast.success(mode === 'create' ? 'Insumo creado' : 'Insumo actualizado')
   385→      onSaved()
   386→    } catch (err) { setError(err instanceof Error ? err.message : 'Error al guardar') }
   387→    finally { setSaving(false) }
   388→  }
   389→
   390→  return (
   391→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   392→      <DialogContent className="sm:max-w-2xl max-h-[90vh] flex flex-col overflow-hidden p-0">
   393→        <DialogHeader className="px-6 pt-6 pb-4 shrink-0">
   394→          <DialogTitle className="text-[#5C3A21]">{mode === 'create' ? 'Nuevo Insumo' : 'Editar Insumo'}</DialogTitle>
   395→          <DialogDescription>{mode === 'create' ? 'Agregar material no comestible para la operación' : `Editando: ${item?.name}`}</DialogDescription>
   396→        </DialogHeader>
   397→        <form onSubmit={handleSubmit} className="flex-1 flex flex-col overflow-hidden">
   398→          <div className="flex-1 overflow-y-auto px-6 space-y-4 pb-6">
   399→          {error && <div className="text-sm text-[#B91C1C] bg-[#B91C1C]/5 border border-[#B91C1C]/20 rounded-md px-3 py-2">{error}</div>}
   400→
   401→          <div className="flex items-center gap-3"><div className="h-7 w-7 rounded-full bg-[#5C3A21] text-[#FFF8E7] flex items-center justify-center text-xs font-bold">1</div><h3 className="text-sm font-semibold text-[#5C3A21]">Datos del Insumo</h3></div>
   402→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 pl-10">
   403→            <div className="space-y-1.5"><Label className="text-[#5C3A21]">Nombre *</Label><Input value={form.name} onChange={(e) => setField('name', e.target.value)} className="border-[#5C3A21]/15 h-10" autoFocus /></div>
   404→            <div className="space-y-1.5"><Label className="text-[#5C3A21]">Categoría</Label><Select value={form.category || 'none'} onValueChange={(v) => setField('category', v === 'none' ? '' : v)}><SelectTrigger className="border-[#5C3A21]/15 h-10"><SelectValue placeholder="Sin categoría" /></SelectTrigger><SelectContent><SelectItem value="none">— Sin categoría —</SelectItem>{CATEGORIES.map((c) => <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>)}</SelectContent></Select></div>
   405→            <div className="space-y-1.5 sm:col-span-2"><Label className="text-[#5C3A21]">Descripción</Label><Textarea value={form.description} onChange={(e) => setField('description', e.target.value)} className="border-[#5C3A21]/15" rows={2} /></div>
   406→          </div>
   407→
   408→          <Separator />
   409→          <div className="flex items-center gap-3"><div className="h-7 w-7 rounded-full bg-[#5C3A21] text-[#FFF8E7] flex items-center justify-center text-xs font-bold">2</div><h3 className="text-sm font-semibold text-[#5C3A21]">Compra</h3></div>
   410→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 pl-10">
   411→            <div className="space-y-1.5"><Label className="text-[#5C3A21]">Tipo de Unidad de Compra</Label><Select value={form.purchaseUnitType || 'none'} onValueChange={(v) => { const nt = v === 'none' ? '' : v; setField('purchaseUnitType', nt); if (isMeasureFieldHidden(nt)) setField('measurePerUnit', '1'); setField('usageUnit', getAutoUsageUnit(nt)) }}><SelectTrigger className="border-[#5C3A21]/15 h-10"><SelectValue placeholder="Sin tipo" /></SelectTrigger><SelectContent><SelectItem value="none">— Sin tipo —</SelectItem>{PURCHASE_TYPES.map((t) => <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>)}</SelectContent></Select></div>
   412→            <div className="space-y-1.5"><Label className="text-[#5C3A21]">{getQuantityLabel(form.purchaseUnitType)}</Label><Input type="number" step="0.01" min="0" value={form.unitsPurchased} onChange={(e) => setField('unitsPurchased', e.target.value)} placeholder="ej: 1 (caja), 6 (rollos)" className="border-[#5C3A21]/15 h-10" /></div>
   413→            {!isMeasureFieldHidden(form.purchaseUnitType) && (
   414→              <>
   415→                <div className="space-y-1.5"><Label className="text-[#5C3A21]">{getMeasureLabel(form.purchaseUnitType)}</Label><Input type="number" step="0.01" min="0" value={form.measurePerUnit} onChange={(e) => setField('measurePerUnit', e.target.value)} placeholder="ej: 100" className="border-[#5C3A21]/15 h-10" /></div>
   416→                <div className="space-y-1.5"><Label className="text-[#5C3A21]">Unidad de Medida</Label><Select value={form.measureUnit} onValueChange={(v) => setField('measureUnit', v)}><SelectTrigger className="border-[#5C3A21]/15 h-10"><SelectValue /></SelectTrigger><SelectContent>{MEASURE_UNITS.map((u) => <SelectItem key={u.value} value={u.value}>{u.label}</SelectItem>)}</SelectContent></Select></div>
   417→              </>
   418→            )}
   419→            <div className="space-y-1.5"><Label className="text-[#5C3A21]">Precio Total Pagado ($)</Label><Input type="number" step="0.01" min="0" value={form.totalPrice} onChange={(e) => setField('totalPrice', e.target.value)} placeholder="ej: 2500" className="border-[#5C3A21]/15 h-10" /></div>
   420→            {form.unitsPurchased && form.totalPrice && parseFloat(form.unitsPurchased) > 0 && parseFloat(form.totalPrice) > 0 && (
   421→              <div className="sm:col-span-2 bg-[#E1AD01]/10 border border-[#E1AD01]/30 rounded-md p-3 space-y-1">
   422→                <p className="text-xs font-semibold text-[#7a5c00]">📊 Cálculo automático:</p>
   423→                {(() => {
   424→                  const qty = parseFloat(form.unitsPurchased)
   425→                  const total = parseFloat(form.totalPrice)
   426→                  const measure = form.measurePerUnit && !isMeasureFieldHidden(form.purchaseUnitType) ? parseFloat(form.measurePerUnit) : 1
   427→                  const pType = form.purchaseUnitType
   428→                  const usageUnit = getAutoUsageUnit(pType)
   429→                  
   430→                  // Price per purchase unit (per metro, per unidad, per kg, etc.)
   431→                  const pricePerPurchase = total / (qty * measure)
   432→                  
   433→                  // Conversion to usage unit
   434→                  let pricePerUsage: number | null = null
   435→                  let purchaseUnitLabel = ''
   436→                  let usageUnitLabel = ''
   437→                  
   438→                  if (pType === 'rollo' || pType === 'metro_suelto') {
   439→                    // metros → centímetros: dividir por 100
   440→                    pricePerUsage = pricePerPurchase / 100
   441→                    purchaseUnitLabel = '/m'
   442→                    usageUnitLabel = '/cm'
   443→                  } else if (pType === 'kg_suelto') {
   444→                    // kg → gramos: dividir por 1000
   445→                    pricePerUsage = pricePerPurchase / 1000
   446→                    purchaseUnitLabel = '/kg'
   447→                    usageUnitLabel = '/g'
   448→                  } else {
   449→                    // unidad, caja, paquete → unidad (sin conversión)
   450→                    pricePerUsage = pricePerPurchase
   451→                    purchaseUnitLabel = '/u'
   452→                    usageUnitLabel = '/u'
   453→                  }
   454→                  
   455→                  return (
   456→                    <>
   457→                      <p className="text-sm text-[#5C3A21]">Precio por unidad de compra: <strong>{fmtCurrency(pricePerPurchase)}{purchaseUnitLabel}</strong></p>
   458→                      <p className="text-sm text-[#5C3A21]">Precio por unidad de uso ({usageUnit === 'u' ? 'Unidades' : usageUnit === 'g' ? 'Gramos' : 'Centímetros'}): <strong>{fmtCurrency(pricePerUsage)}{usageUnitLabel}</strong></p>
   459→                    </>
   460→                  )
   461→                })()}
   462→              </div>
   463→            )}
   464→            {mode === 'edit' && (
   465→              <div className="space-y-1.5 flex items-center gap-3 pt-2"><Label className="text-[#5C3A21]">Activo</Label><Switch checked={form.isActive} onCheckedChange={(v) => setField('isActive', v)} /></div>
   466→            )}
   467→          </div>
   468→
   469→          <Separator />
   470→          <div className="flex items-center gap-3"><div className="h-7 w-7 rounded-full bg-[#5C3A21] text-[#FFF8E7] flex items-center justify-center text-xs font-bold">3</div><h3 className="text-sm font-semibold text-[#5C3A21]">Imagen</h3></div>
   471→          <div className="pl-10">
   472→            <ImageUploader value={form.image || null} onChange={(url) => setField('image', url || '')} uploadUrl="/api/cocina-movil/supplies/upload-image" label="Imagen del insumo" aspectRatio="4/3" disabled={saving} />
   473→          </div>
   474→
   475→          </div>
   476→          <div className="shrink-0 bg-white border-t border-[#5C3A21]/10 px-6 py-3 flex justify-end gap-2">
   477→            <Button type="button" variant="outline" onClick={onClose}>Cancelar</Button>
   478→            <Button type="submit" disabled={saving} className="bg-[#E1AD01] hover:bg-[#E1AD01]/90 text-[#1F1611]">{saving ? <Loader2 className="h-4 w-4 animate-spin" /> : null}{mode === 'create' ? 'Crear' : 'Guardar'}</Button>
   479→          </div>
   480→        </form>
   481→      </DialogContent>
   482→    </Dialog>
   483→  )
   484→}
