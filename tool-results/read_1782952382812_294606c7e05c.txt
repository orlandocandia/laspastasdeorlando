   387→   387→  async function onSubmit(data: PersonaFormValues) {

   397→   397→        razon_social: data.razon_social || null,

   407→   407→        direcciones: data.direccion
   408→   408→          ? [
   409→   409→              {
   410→   410→                id_tipo_direccion: data.id_tipo_direccion
   411→   411→                  ? parseInt(data.id_tipo_direccion)
   412→   412→                  : 1,
   413→   413→                id_municipio: data.id_municipio
   414→   414→                  ? parseInt(data.id_municipio)
   415→   415→                  : null,
   416→   416→                direccion: data.direccion,
   417→   417→                referencia: data.referencia || null,
   418→   418→                es_principal: data.es_principal_direccion,
   419→   419→              },
   420→   420→            ]
   421→   421→          : [],
   422→   422→      }
   423→   423→
   424→   424→      const url = isEditing ? `/api/personas/${persona.id}` : '/api/personas'
   425→   425→      const method = isEditing ? 'PUT' : 'POST'
   426→   426→
   427→   427→      const res = await fetch(url, {
   428→   428→        method,
   429→   429→        headers: { 'Content-Type': 'application/json' },
   430→   430→        body: JSON.stringify(payload),
   431→   431→      })
   432→   432→
   433→   433→      if (!res.ok) {
   434→   434→        const errorData = await res.json()
   435→   435→        throw new Error(errorData.error || 'Error al guardar persona')
   436→   436→      }
   437→   437→
   438→   438→      toast.success(isEditing ? 'Persona actualizada' : 'Persona creada', {
   439→   439→        description: isEditing
   440→   440→          ? 'Los cambios se guardaron correctamente'
   441→   441→          : 'La nueva persona fue registrada exitosamente',
   442→   442→      })

   444→   444→      onSuccess()
   445→   445→    } catch (err) {

   448→   448→    } finally {
   449→   449→      setSubmitting(false)
   450→   450→    }
   451→   451→  }
