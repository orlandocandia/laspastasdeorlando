    87→    87→
    88→    88→export default function VentaForm({ venta, fromPedido, onSuccess, onCancel }: VentaFormProps) {
    89→    89→  const isEditing = !!venta
    90→    90→
    91→    91→  // Form fields
    92→    92→  const [idCliente, setIdCliente] = useState('')
    93→    93→  const [fechaVenta, setFechaVenta] = useState('')
    94→    94→  const [idFormaPago, setIdFormaPago] = useState('')
    95→    95→  const [idVendedor, setIdVendedor] = useState('')
    96→    96→  const [numeroComprobante, setNumeroComprobante] = useState('')
    97→    97→  const [idEstado, setIdEstado] = useState('')
    98→    98→  const [observaciones, setObservaciones] = useState('')
    99→    99→  const [idPedido, setIdPedido] = useState('')
   100→   100→
   101→   101→  // Detail rows
   102→   102→  const [detalles, setDetalles] = useState<DetalleRow[]>([])
   103→   103→
   104→   104→  // Data
   105→   105→  const [clientes, setClientes] = useState<Cliente[]>([])
   106→   106→  const [formasPago, setFormasPago] = useState<FormaPago[]>([])
   107→   107→  const [estados, setEstados] = useState<EstadoGeneral[]>([])
   108→   108→  const [productosTerminados, setProductosTerminados] = useState<ProductoTerminado[]>([])
   109→   109→  const [vendedores, setVendedores] = useState<Vendedor[]>([])
   110→   110→  const [pedidosPendientes, setPedidosPendientes] = useState<PedidoPendiente[]>([])
   111→   111→  const [submitting, setSubmitting] = useState(false)
   112→   112→  const [loadingData, setLoadingData] = useState(true)
   113→   113→
   114→   114→  // Barcode scanner ref
   115→   115→  const codigoBarrasRef = useRef<HTMLInputElement>(null)
   116→   116→
   117→   117→  // Combobox open states
   118→   118→  const [clienteOpen, setClienteOpen] = useState(false)
   119→   119→  const [pedidoOpen, setPedidoOpen] = useState(false)
   120→   120→
   121→   121→  // Load data
   122→   122→  useEffect(() => {
   123→   123→    async function loadData() {
   124→   124→      try {
   125→   125→        const [cliRes, fpRes, estRes, ptRes, venRes, pedRes] = await Promise.all([
   126→   126→          fetch('/api/personas?tipo=cliente&limite=100'),
   127→   127→          fetch('/api/formas-pago'),
   128→   128→          fetch('/api/estados-generales?entidad_aplicable=venta'),
   129→   129→          fetch('/api/productos-terminados?limite=200&estado=true'),
   130→   130→          fetch('/api/usuarios'),
   131→   131→          fromPedido
   132→   132→            ? fetch('/api/pedidos-clientes?limite=100&id_estado=pendiente')
   133→   133→            : Promise.resolve(new Response('[]', { status: 200 })),
   134→   134→        ])
   135→   135→
   136→   136→        const cliData = await cliRes.json()
   137→   137→        const fpData = await fpRes.json()
   138→   138→        const estData = await estRes.json()
   139→   139→        const ptData = await ptRes.json()
   140→   140→        const venData = await venRes.json()
   141→   141→        const pedData = await pedRes.json()
   142→   142→
   143→   143→        setClientes(cliData.personas || [])
   144→   144→        setFormasPago(Array.isArray(fpData) ? fpData : [])
   145→   145→        setEstados(Array.isArray(estData) ? estData : [])
   146→   146→        setProductosTerminados(
   147→   147→          (ptData.data || []).map((pt: any) => ({
   148→   148→            id: pt.id,
   149→   149→            nombre: pt.nombre,
   150→   150→            precio_venta: pt.precio_venta,
   151→   151→          }))
   152→   152→        )
   153→   153→        setVendedores(venData.usuarios || venData || [])
   154→   154→        if (fromPedido) {
   155→   155→          setPedidosPendientes(pedData.data || [])
   156→   156→        }
   157→   157→      } catch {
   158→   158→        toast.error('Error al cargar datos del formulario')
   159→   159→      } finally {
   160→   160→        setLoadingData(false)
   161→   161→      }
   162→   162→    }
   163→   163→    loadData()
   164→   164→  }, [fromPedido])
   165→   165→
   166→   166→  // Populate form when editing
   167→   167→  useEffect(() => {
   168→   168→    if (venta) {
   169→   169→      setIdCliente(venta.id_cliente?.toString() || '')
   170→   170→      setFechaVenta(venta.fecha_venta ? venta.fecha_venta.split('T')[0] : '')
   171→   171→      setIdFormaPago(venta.id_forma_pago?.toString() || '')
   172→   172→      setIdVendedor(venta.id_vendedor?.toString() || '')
   173→   173→      setNumeroComprobante(venta.numero_comprobante || '')
   174→   174→      setIdEstado(venta.id_estado?.toString() || '')
   175→   175→      setObservaciones(venta.observaciones || '')
   176→   176→      setIdPedido(venta.id_pedido?.toString() || '')
   177→   177→
   178→   178→      if (venta.detalle && Array.isArray(venta.detalle)) {
   179→   179→        setDetalles(
   180→   180→          venta.detalle.map((d: any) => ({
   181→   181→            key: `row-${d.id}`,
   182→   182→            idProductoTerminado: d.id_producto_terminado?.toString() || '',
   183→   183→            cantidad: d.cantidad?.toString() || '',
   184→   184→            precioUnitario: d.precio_unitario?.toString() || '',
   185→   185→            subtotal: d.subtotal?.toString() || '',
   186→   186→          }))
   187→   187→        )
   188→   188→      }
   189→   189→    } else {
   190→   190→      // Defaults for new
   191→   191→      setFechaVenta(new Date().toISOString().split('T')[0])
   192→   192→      addDetailRow()
   193→   193→    }
   194→   194→  }, [venta])
   195→   195→
   196→   196→  // Handle pedido selection (fromPedido mode)
   197→   197→  const handlePedidoSelect = (pedidoId: string) => {
   198→   198→    setIdPedido(pedidoId)
   199→   199→    const pedido = pedidosPendientes.find((p) => p.id.toString() === pedidoId)
   200→   200→    if (pedido) {
   201→   201→      setIdCliente(pedido.cliente?.id?.toString() || '')
   202→   202→      setDetalles(
   203→   203→        pedido.detalle.map((d: any) => ({
   204→   204→          key: `row-${d.id || Date.now() + Math.random()}`,
   205→   205→          idProductoTerminado: d.id_producto_terminado?.toString() || '',
   206→   206→          cantidad: d.cantidad?.toString() || '',
   207→   207→          precioUnitario: d.precio_unitario?.toString() || d.productoTerminado?.precio_venta?.toString() || '',
   208→   208→          subtotal: (d.cantidad * (d.precio_unitario || d.productoTerminado?.precio_venta || 0)).toFixed(2),
   209→   209→        }))
   210→   210→      )
   211→   211→    }
   212→   212→  }
   213→   213→
   214→   214→  // Find product by id
   215→   215→  const findProducto = (id: string) => {
   216→   216→    return productosTerminados.find((p) => p.id.toString() === id)
   217→   217→  }
   218→   218→
   219→   219→  // Search product by barcode
   220→   220→  const buscarProductoPorCodigo = async (codigo: string) => {
   221→   221→    if (!codigo.trim()) return
   222→   222→    try {
   223→   223→      const res = await fetch(`/api/productos-terminados/buscar-por-codigo?codigo=${encodeURIComponent(codigo.trim())}`)
   224→   224→      const data = await res.json()
   225→   225→
   226→   226→      if (!res.ok || data.error) {
   227→   227→        toast.error('Producto no encontrado')
   228→   228→        return
   229→   229→      }
   230→   230→
   231→   231→      if (data.multiples) {
   232→   232→        toast.error('Múltiples productos coinciden, use búsqueda manual')
   233→   233→        return
   234→   234→      }
   235→   235→
   236→   236→      // Check if product already in detail rows
   237→   237→      const existingIndex = detalles.findIndex(d => d.idProductoTerminado === data.id.toString())
   238→   238→      if (existingIndex >= 0) {
   239→   239→        // Increment quantity
   240→   240→        const newDetalles = [...detalles]
   241→   241→        const existing = newDetalles[existingIndex]
   242→   242→        const newCantidad = (parseFloat(existing.cantidad) || 0) + 1
   243→   243→        newDetalles[existingIndex] = {
   244→   244→          ...existing,
   245→   245→          cantidad: newCantidad.toString(),
   246→   246→        }
   247→   247→        newDetalles[existingIndex].subtotal = (newCantidad * parseFloat(existing.precioUnitario)).toFixed(2)
   248→   248→        setDetalles(newDetalles)
   249→   249→        toast.success(`${data.nombre} - cantidad actualizada`)
   250→   250→      } else {
   251→   251→        // Add new row
   252→   252→        const newKey = `row-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`
   253→   253→        setDetalles(prev => [...prev, {
   254→   254→          key: newKey,
   255→   255→          idProductoTerminado: data.id.toString(),
   256→   256→          cantidad: '1',
   257→   257→          precioUnitario: data.precio_venta?.toString() || '',
   258→   258→          subtotal: data.precio_venta?.toString() || '',
   259→   259→        }])
   260→   260→        toast.success(`${data.nombre} agregado`)
   261→   261→      }
   262→   262→
   263→   263→      // Clear and refocus
   264→   264→      if (codigoBarrasRef.current) {
   265→   265→        codigoBarrasRef.current.value = ''
   266→   266→        codigoBarrasRef.current.focus()
   267→   267→      }
   268→   268→    } catch {
   269→   269→      toast.error('Error al buscar producto')
   270→   270→    }
   271→   271→  }
   272→   272→
   273→   273→  // Add detail row
   274→   274→  const addDetailRow = () => {
   275→   275→    setDetalles((prev) => [
   276→   276→      ...prev,
   277→   277→      {
   278→   278→        key: `row-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   279→   279→        idProductoTerminado: '',
   280→   280→        cantidad: '',
   281→   281→        precioUnitario: '',
   282→   282→        subtotal: '',
   283→   283→      },
   284→   284→    ])
   285→   285→  }
   286→   286→
   287→   287→  // Remove detail row
   288→   288→  const removeDetailRow = (key: string) => {
   289→   289→    setDetalles((prev) => prev.filter((d) => d.key !== key))
   290→   290→  }
   291→   291→
   292→   292→  // Update detail row
   293→   293→  const updateDetalle = (key: string, field: keyof DetalleRow, value: string) => {
   294→   294→    setDetalles((prev) =>
   295→   295→      prev.map((d) => {
   296→   296→        if (d.key !== key) return d
   297→   297→
   298→   298→        const updated = { ...d, [field]: value }
   299→   299→
   300→   300→        // When product changes, auto-fill precio_venta
   301→   301→        if (field === 'idProductoTerminado' && value) {
   302→   302→          const product = findProducto(value)
   303→   303→          if (product) {
   304→   304→            updated.precioUnitario = product.precio_venta?.toString() || ''
   305→   305→          }
   306→   306→        }
   307→   307→
   308→   308→        // Auto-calculate subtotal when cantidad or precio_unitario changes
   309→   309→        if (field === 'cantidad' || field === 'precioUnitario') {
   310→   310→          const cant = parseFloat(field === 'cantidad' ? value : updated.cantidad)
   311→   311→          const precio = parseFloat(field === 'precioUnitario' ? value : updated.precioUnitario)
   312→   312→          if (!isNaN(cant) && !isNaN(precio)) {
   313→   313→            updated.subtotal = (cant * precio).toFixed(2)
   314→   314→          } else {
   315→   315→            updated.subtotal = ''
   316→   316→          }
   317→   317→        }
   318→   318→
   319→   319→        return updated
   320→   320→      })
   321→   321→    )
   322→   322→  }
   323→   323→
   324→   324→  // Calculate totals
   325→   325→  const { subtotal, iva, total } = useMemo(() => {
   326→   326→    const sub = detalles.reduce((sum, d) => {
   327→   327→      const st = parseFloat(d.subtotal)
   328→   328→      return sum + (isNaN(st) ? 0 : st)
   329→   329→    }, 0)
   330→   330→    const ivaCalc = sub * IVA_RATE
   331→   331→    return {
   332→   332→      subtotal: sub,
   333→   333→      iva: ivaCalc,
   334→   334→      total: sub + ivaCalc,
   335→   335→    }
   336→   336→  }, [detalles])
   337→   337→
   338→   338→  const formatCurrency = (value: number) => {
   339→   339→    return new Intl.NumberFormat('es-AR', { style: 'currency', currency: 'ARS' }).format(value)
   340→   340→  }
   341→   341→
   342→   342→  // Submit
   343→   343→  const handleSubmit = async () => {
   344→   344→    // Validation
   345→   345→    if (!isEditing) {
   346→   346→      if (!idCliente) {
   347→   347→        toast.error('Seleccioná un cliente')
   348→   348→        return
   349→   349→      }
   350→   350→      if (!fechaVenta) {
   351→   351→        toast.error('Ingresá la fecha de venta')
   352→   352→        return
   353→   353→      }
   354→   354→      if (!idFormaPago) {
   355→   355→        toast.error('Seleccioná una forma de pago')
   356→   356→        return
   357→   357→      }
   358→   358→      if (!idVendedor) {
   359→   359→        toast.error('Seleccioná un vendedor')
   360→   360→        return
   361→   361→      }
   362→   362→      if (detalles.length === 0) {
   363→   363→        toast.error('Agregá al menos un detalle a la venta')
   364→   364→        return
   365→   365→      }
   366→   366→      for (let i = 0; i < detalles.length; i++) {
   367→   367→        const d = detalles[i]
   368→   368→        if (!d.idProductoTerminado) {
   369→   369→          toast.error(`Fila ${i + 1}: Seleccioná un producto terminado`)
   370→   370→          return
   371→   371→        }
   372→   372→        if (!d.cantidad || parseFloat(d.cantidad) <= 0) {
   373→   373→          toast.error(`Fila ${i + 1}: Ingresá una cantidad válida`)
   374→   374→          return
   375→   375→        }
   376→   376→        if (!d.precioUnitario || parseFloat(d.precioUnitario) <= 0) {
   377→   377→          toast.error(`Fila ${i + 1}: Ingresá un precio unitario válido`)
   378→   378→          return
   379→   379→        }
   380→   380→      }
   381→   381→    }
   382→   382→
   383→   383→    setSubmitting(true)
   384→   384→    try {
   385→   385→      const payload = isEditing
   386→   386→        ? {
   387→   387→            id: venta.id,
   388→   388→            numero_comprobante: numeroComprobante.trim() || null,
   389→   389→            id_estado: idEstado ? parseInt(idEstado) : undefined,
   390→   390→            observaciones: observaciones.trim() || null,
   391→   391→          }
   392→   392→        : {
   393→   393→            id_cliente: parseInt(idCliente),
   394→   394→            id_vendedor: parseInt(idVendedor),
   395→   395→            id_forma_pago: parseInt(idFormaPago),
   396→   396→            id_pedido: idPedido ? parseInt(idPedido) : null,
   397→   397→            numero_comprobante: numeroComprobante.trim() || null,
   398→   398→            fecha_venta: fechaVenta,
   399→   399→            subtotal,
   400→   400→            iva,
   401→   401→            total,
   402→   402→            observaciones: observaciones.trim() || null,
   403→   403→            detalle: detalles.map((d) => ({
   404→   404→              id_producto_terminado: parseInt(d.idProductoTerminado),
   405→   405→              cantidad: parseFloat(d.cantidad),
   406→   406→              precio_unitario: parseFloat(d.precioUnitario),
   407→   407→              subtotal: parseFloat(d.subtotal),
   408→   408→            })),
   409→   409→          }
   410→   410→
   411→   411→      const url = isEditing ? `/api/ventas/${venta.id}` : '/api/ventas'
   412→   412→      const method = isEditing ? 'PUT' : 'POST'
   413→   413→
   414→   414→      const res = await fetch(url, {
   415→   415→        method,
   416→   416→        headers: { 'Content-Type': 'application/json' },
   417→   417→        body: JSON.stringify(payload),
   418→   418→      })
   419→   419→
   420→   420→      if (!res.ok) {
   421→   421→        const errorData = await res.json().catch(() => ({}))
   422→   422→        throw new Error(errorData.error || 'Error al guardar venta')
   423→   423→      }
   424→   424→
   425→   425→      toast.success(isEditing ? 'Venta actualizada' : 'Venta registrada', {
   426→   426→        description: isEditing
   427→   427→          ? 'Los cambios se guardaron correctamente'
   428→   428→          : 'La nueva venta se registró exitosamente',
   429→   429→      })
   430→   430→      onSuccess()
   431→   431→    } catch (error: any) {
   432→   432→      toast.error('Error al guardar', {
   433→   433→        description: error.message || 'Intentá de nuevo más tarde',
   434→   434→      })
   435→   435→    } finally {
   436→   436→      setSubmitting(false)
   437→   437→    }
   438→   438→  }
   439→   439→
   440→   440→  if (loadingData) {
   441→   441→    return (
   442→   442→      <div className="flex items-center justify-center py-8">
   443→   443→        <Loader2 className="h-6 w-6 animate-spin text-mostaza" />
   444→   444→        <span className="ml-2 text-muted-foreground">Cargando datos...</span>
   445→   445→      </div>
   446→   446→    )
   447→   447→  }
   448→   448→
   449→   449→  return (
   450→   450→    <div className="space-y-6">
   451→   451→      {/* Header info */}
   452→   452→      {isEditing && (
   453→   453→        <div className="bg-muted/30 rounded-lg p-3 border border-marron/10">
   454→   454→          <p className="text-sm text-muted-foreground">
   455→   455→            Venta #{venta.id} — Cliente: {venta.cliente?.razon_social || `${venta.cliente?.nombre} ${venta.cliente?.apellido}`}
   456→   456→          </p>
   457→   457→        </div>
   458→   458→      )}
   459→   459→
   460→   460→      {/* From Pedido selector */}
   461→   461→      {fromPedido && (
   462→   462→        <div className="bg-mostaza/5 rounded-lg p-3 border border-mostaza/20">
   463→   463→          <Label className="text-sm font-medium text-marron mb-1 block">Seleccionar Pedido Pendiente *</Label>
   464→   464→          <Popover open={pedidoOpen} onOpenChange={setPedidoOpen}>
   465→   465→            <PopoverTrigger asChild>
