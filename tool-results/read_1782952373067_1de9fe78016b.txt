   149→export default function PersonaForm({ persona, onSuccess }: PersonaFormProps) {
   150→  const [submitting, setSubmitting] = useState(false)
   151→  const [imageUrl, setImageUrl] = useState(persona?.imagen || '')
   152→  const [uploading, setUploading] = useState(false)
   153→  const [fiscalOpen, setFiscalOpen] = useState(!!persona?.cuit || !!persona?.razon_social)
   154→  const fileInputRef = useRef<HTMLInputElement>(null)
   155→
   156→  // Geographic data
   157→  const [tiposPersona, setTiposPersona] = useState<TipoPersona[]>([])
   158→  const [tiposDireccion, setTiposDireccion] = useState<TipoDireccion[]>([])
   159→  const [tiposContacto, setTiposContacto] = useState<TipoContacto[]>([])
   160→  const [paises, setPaises] = useState<GeoOption[]>([])
   161→  const [provincias, setProvincias] = useState<GeoOption[]>([])
   162→  const [departamentos, setDepartamentos] = useState<GeoOption[]>([])
   163→  const [municipios, setMunicipios] = useState<GeoOption[]>([])
   164→
   165→  // Contactos
   166→  const [contactos, setContactos] = useState<ContactoInput[]>([])
   167→
   168→  // Ubicación en mapa
   169→  const [latitud, setLatitud] = useState<number | null>(persona?.latitud ?? null)
   170→  const [longitud, setLongitud] = useState<number | null>(persona?.longitud ?? null)
   171→  const [ubicacionValida, setUbicacionValida] = useState(persona?.ubicacion_valida ?? false)
   172→
   173→  const isEditing = !!persona
   174→
   175→  // Determine initial geographic values from existing persona
   176→  const initialPais = persona?.municipio?.departamento?.provincia?.id_pais
   177→    ? String(persona.municipio.departamento.provincia.id_pais)

   180→    ? String(persona.municipio.departamento.id_provincia)

   183→    ? String(persona.municipio.departamento.id)

   189→  // Get the primary direccion from persona

   197→      numero_documento: persona?.numero_documento || '',
   198→      fecha_nacimiento: persona?.fecha_nacimiento

   203→      razon_social: persona?.razon_social || '',

   231→      const tiposP = await tiposPRes.json()
   232→      const tiposD = await tiposDRes.json()
   233→      const tiposC = await tiposCRes.json()
   234→      const paisesData = await paisesRes.json()
   235→
   236→      setTiposPersona(Array.isArray(tiposP) ? tiposP : [])
   237→      setTiposDireccion(Array.isArray(tiposD) ? tiposD : [])
   238→      setTiposContacto(Array.isArray(tiposC) ? tiposC : [])
   239→      setPaises(Array.isArray(paisesData) ? paisesData : [])
   240→    } catch {
   241→      toast.error('Error al cargar datos iniciales')
   242→    }
   243→  }, [])
   244→
   245→  // Load initial contactos from persona
   246→  useEffect(() => {
   247→    if (persona?.contactos) {
   248→      setContactos(
   249→        persona.contactos.map((c) => ({
   250→          id_tipo_contacto: c.id_tipo_contacto,
   251→          valor: c.valor,
   252→          es_principal: c.es_principal,
   253→        }))
   254→      )
   255→    }
   256→  }, [persona])
   257→
   258→  // Load initial data (paises, tipos)
   259→  useEffect(() => {
   260→    fetchInitialData()
   261→  }, [fetchInitialData])
   262→
   263→  // ==================== Precargar datos geográficos al editar ====================
   264→  // Cuando se edita una persona existente con municipio, precargar en cascada:
   265→  // provincias del país → departamentos de la provincia → municipios del departamento
   266→  useEffect(() => {
   267→    if (!isEditing || !initialPais) return
   268→
   269→    let cancelled = false
   270→
   271→    async function preloadGeo() {
   272→      try {
   273→        // 1. Cargar provincias del país inicial
   274→        const provRes = await fetch(`/api/geografia?tipo=provincias&id=${initialPais}`)
   275→        const provData = await provRes.json()
   276→        if (cancelled) return
   277→        setProvincias(Array.isArray(provData) ? provData : [])
   278→
   279→        // 2. Cargar departamentos de la provincia inicial
   280→        if (initialProvincia) {
   281→          const depRes = await fetch(`/api/geografia?tipo=departamentos&id=${initialProvincia}`)
   282→          const depData = await depRes.json()
   283→          if (cancelled) return
   284→          setDepartamentos(Array.isArray(depData) ? depData : [])
   285→
   286→          // 3. Cargar municipios del departamento inicial
   287→          if (initialDepartamento) {
   288→            const munRes = await fetch(`/api/geografia?tipo=municipios&id=${initialDepartamento}`)
   289→            const munData = await munRes.json()
   290→            if (cancelled) return
   291→            setMunicipios(Array.isArray(munData) ? munData : [])
   292→          }
   293→        }
   294→      } catch (err) {
   295→        console.error('Error al precargar datos geográficos:', err)
   296→      }
   297→    }
   298→
   299→    preloadGeo()
   300→    return () => { cancelled = true }
   301→  }, [isEditing, initialPais, initialProvincia, initialDepartamento])
   302→
   303→  // ==================== Cascading geographic selects ====================
   304→
   305→  // These handlers are the SOLE source of truth for cascading selects.
   306→  // No useEffect subscriptions needed — the Select onValueChange calls these directly.
   307→
   308→  const handlePaisChange = (value: string) => {

   310→    form.setValue('id_provincia', '')

   314→    setDepartamentos([])
   315→    setMunicipios([])
   316→    if (value) {
   317→      fetch(`/api/geografia?tipo=provincias&id=${value}`)
   318→        .then((res) => res.json())
   319→        .then((data) => setProvincias(Array.isArray(data) ? data : []))
   320→        .catch(() => setProvincias([]))
   321→    }
   322→  }
   323→
   324→  const handleProvinciaChange = (value: string) => {
   325→    form.setValue('id_provincia', value)
   326→    form.setValue('id_departamento', '')
   327→    form.setValue('id_municipio', '')
   328→    setDepartamentos([])
   329→    setMunicipios([])
   330→    if (value) {
   331→      fetch(`/api/geografia?tipo=departamentos&id=${value}`)
   332→        .then((res) => res.json())
   333→        .then((data) => setDepartamentos(Array.isArray(data) ? data : []))
   334→        .catch(() => setDepartamentos([]))
   335→    }
   336→  }
   337→
   338→  const handleDepartamentoChange = (value: string) => {
   339→    form.setValue('id_departamento', value)
   340→    form.setValue('id_municipio', '')
   341→    setMunicipios([])
   342→    if (value) {
   343→      fetch(`/api/geografia?tipo=municipios&id=${value}`)
   344→        .then((res) => res.json())
   345→        .then((data) => setMunicipios(Array.isArray(data) ? data : []))
   346→        .catch(() => setMunicipios([]))
   347→    }
   348→  }
   349→
   350→  // ==================== Image upload ====================
   351→
   352→  const uploadImage = async (file: File) => {
   353→    if (!file.type.startsWith('image/')) {
   354→      toast.error('Solo se permiten imágenes')
   355→      return
   356→    }
   357→    if (file.size > 5 * 1024 * 1024) {
   358→      toast.error('La imagen no puede superar 5MB')
   359→      return
   360→    }
   361→
   362→    setUploading(true)
   363→    try {
   364→      const formData = new FormData()
   365→      formData.append('file', file)
   366→
   367→      const res = await fetch('/api/upload/persona', {
   368→        method: 'POST',
   369→        body: formData,
   370→      })
   371→
   372→      if (!res.ok) throw new Error('Error al subir imagen')
   373→
   374→      const data = await res.json()
   375→      setImageUrl(data.url)
   376→      form.setValue('imagen', data.url)
   377→      toast.success('Foto subida correctamente')
   378→    } catch {
   379→      toast.error('Error al subir la foto')
   380→    } finally {
   381→      setUploading(false)
   382→    }
   383→  }
   384→
   385→  // ==================== Submit ====================
   386→
   387→  async function onSubmit(data: PersonaFormValues) {
   388→    setSubmitting(true)
   389→    try {
   390→      const payload: Record<string, unknown> = {
   391→        nombre: data.nombre,
   392→        apellido: data.apellido,
   393→        numero_documento: data.numero_documento,
   394→        fecha_nacimiento: data.fecha_nacimiento || null,
   395→        tipo_persona: data.tipo_persona,
   396→        observaciones: data.observaciones || null,
   397→        razon_social: data.razon_social || null,
   398→        cuit: data.cuit || null,
   399→        condicion_iva: data.condicion_iva || null,
   400→        imagen: imageUrl || null,

   402→        contactos: contactos.filter((c) => c.valor.trim() !== ''),
   403→        latitud: latitud,
   404→        longitud: longitud,
   405→        direccion_mapa: data.direccion || null,
   406→        ubicacion_valida: ubicacionValida,
   407→        direcciones: data.direccion

   409→              {
   410→                id_tipo_direccion: data.id_tipo_direccion

   412→                  : 1,
   413→                id_municipio: data.id_municipio
   414→                  ? parseInt(data.id_municipio)
   415→                  : null,
   416→                direccion: data.direccion,
   417→                referencia: data.referencia || null,
   418→                es_principal: data.es_principal_direccion,
   419→              },
   420→            ]

   427→      const res = await fetch(url, {

   429→        headers: { 'Content-Type': 'application/json' },
   430→        body: JSON.stringify(payload),

   434→        const errorData = await res.json()

   439→        description: isEditing
   440→          ? 'Los cambios se guardaron correctamente'
   441→          : 'La nueva persona fue registrada exitosamente',
   442→      })

   444→      onSuccess()
   445→    } catch (err) {
   446→      const message = err instanceof Error ? err.message : 'Error al guardar'
   447→      toast.error('Error al guardar', { description: message })
   448→    } finally {
   449→      setSubmitting(false)
   450→    }
   451→  }
   452→
   453→  return (

   456→        {/* ===== Datos Personales ===== */}

   471→              )}

   478→                <FormItem>
   479→                  <FormLabel>Apellido *</FormLabel>
   480→                  <FormControl>

   482→                  </FormControl>
   483→                  <FormMessage />
   484→                </FormItem>
   485→              )}
   486→            />
   487→
   488→            <FormField

   491→              render={({ field }) => (

   495→                    <Input placeholder="12345678" {...field} />
   496→                  </FormControl>
   497→                  <FormMessage />
   498→                </FormItem>
   499→              )}
   500→            />
   501→
   502→            <FormField
   503→              control={form.control}
   504→              name="fecha_nacimiento"
   505→              render={({ field }) => (
   506→                <FormItem>
   507→                  <FormLabel>Fecha de nacimiento</FormLabel>
   508→                  <FormControl>
   509→                    <Input type="date" {...field} />
   510→                  </FormControl>
   511→                  <FormMessage />
   512→                </FormItem>
   513→              )}
   514→            />
   515→
   516→            <FormField
   517→              control={form.control}
   518→              name="tipo_persona"
   519→              render={({ field }) => (
   520→                <FormItem>
   521→                  <FormLabel>Tipo de persona *</FormLabel>
   522→                  <Select onValueChange={field.onChange} value={field.value}>
   523→                    <FormControl>
   524→                      <SelectTrigger>

   531→                          {tipo.nombre}
   532→                        </SelectItem>
   533→                      ))}
   534→                    </SelectContent>
   535→                  </Select>
   536→                  <FormMessage />
   537→                </FormItem>
   538→              )}
   539→            />
   540→          </div>
   541→
   542→          <FormField
   543→            control={form.control}
   544→            name="observaciones"
   545→            render={({ field }) => (
   546→              <FormItem className="mt-4">
   547→                <FormLabel>Observaciones</FormLabel>
   548→                <FormControl>
   549→                  <Textarea
   550→                    placeholder="Notas adicionales..."
   551→                    className="resize-none"
   552→                    rows={2}

   554→                  />
   555→                </FormControl>
   556→                <FormMessage />
   557→              </FormItem>

   559→          />

   563→
   564→        {/* ===== Datos Fiscales (colapsable) ===== */}

   573→              <ChevronUp className="h-4 w-4" />
   574→            ) : (
   575→              <ChevronDown className="h-4 w-4" />
   576→            )}
   577→          </button>
