   129→   129→
   130→   130→interface ProductosProps {
   131→   131→  filtroActivo?: FiltroHarina

   135→   135→export default function Productos({ filtroActivo = 'con_gluten', onFiltroChange }: ProductosProps) {
   136→   136→  // Cache: productos pre-fetcheados por tipo de harina
   137→   137→  const [cache, setCache] = useState<Record<FiltroHarina, ProductoPublico[]>>({
   138→   138→    con_gluten: [],
   139→   139→    integral: [],
   140→   140→    sin_gluten: [],
   141→   141→  })
   142→   142→  const [loading, setLoading] = useState(true)
   143→   143→  const [error, setError] = useState('')
   144→   144→  const [filtro, setFiltro] = useState<FiltroHarina>(filtroActivo)
   145→   145→  // null = no section selected (step 1), 'pastas'|'horneados' = section selected (step 2)
   146→   146→  const [seccionActiva, setSeccionActiva] = useState<Seccion | null>(null)
   147→   147→  const [familiaActiva, setFamiliaActiva] = useState<string | null>(null)
   148→   148→  const [visibleCount, setVisibleCount] = useState(PRODUCTS_PER_PAGE)
   149→   149→  const fetchedRef = useRef(false)
   150→   150→  const productosGridRef = useRef<HTMLDivElement>(null)
   151→   151→
   152→   152→  // Pre-fetch all 3 filter types in parallel on mount — no refetch on filter change
   153→   153→  useEffect(() => {
   154→   154→    if (fetchedRef.current) return
   155→   155→    fetchedRef.current = true
   156→   156→
   157→   157→    async function fetchAll() {
   158→   158→      setLoading(true)
   159→   159→      setError('')
   160→   160→      try {
   161→   161→        const results = await Promise.all(
   162→   162→          TODOS_FILTROS.map(async (tipo) => {
   163→   163→            const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   164→   164→            if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   165→   165→            const data = await res.json()
   166→   166→            return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   167→   167→          })
   168→   168→        )
   169→   169→        setCache((prev) => {
   170→   170→          const next = { ...prev }
   171→   171→          for (const r of results) next[r.tipo] = r.productos
   172→   172→          return next
   173→   173→        })
   174→   174→      } catch {
   175→   175→        setError('No se pudieron cargar los productos. Intentá más tarde.')
   176→   176→      } finally {
   177→   177→        setLoading(false)
   178→   178→      }
   179→   179→    }
   180→   180→    fetchAll()
   181→   181→  }, [])
   182→   182→
   183→   183→  // Reset visible count when family or section changes
   184→   184→  useEffect(() => {
   185→   185→    setVisibleCount(PRODUCTS_PER_PAGE)
   186→   186→  }, [familiaActiva, seccionActiva])
   187→   187→
   188→   188→  // Sync with parent filtro state — instant, no refetch
   189→   189→  useEffect(() => {
   190→   190→    if (filtroActivo !== filtro) {
   191→   191→      setFiltro(filtroActivo ?? 'con_gluten')
   192→   192→      setFamiliaActiva(null)
   193→   193→    }
   194→   194→  }, [filtroActivo, filtro])
   195→   195→
   196→   196→  const handleFiltroChange = useCallback((nuevoFiltro: FiltroHarina) => {
   197→   197→    setFiltro(nuevoFiltro)
   198→   198→    setFamiliaActiva(null)
   199→   199→    onFiltroChange?.(nuevoFiltro)
   200→   200→  }, [onFiltroChange])
   201→   201→
   202→   202→  const handleSeccionChange = useCallback((nuevaSeccion: Seccion) => {
   203→   203→    setSeccionActiva(nuevaSeccion)
   204→   204→    setFamiliaActiva(null)
   205→   205→    setFiltro('con_gluten') // Reset flour filter when changing section
   206→   206→    onFiltroChange?.('con_gluten')
   207→   207→  }, [onFiltroChange])
   208→   208→
   209→   209→  const handleVolverASecciones = useCallback(() => {
   210→   210→    setSeccionActiva(null)
   211→   211→    setFamiliaActiva(null)
   212→   212→    setFiltro('con_gluten')
   213→   213→    onFiltroChange?.('con_gluten')
   214→   214→  }, [onFiltroChange])
   215→   215→
   216→   216→  // Current products from cache — instant swap, no network
   217→   217→  const productos = cache[filtro]
   218→   218→
   219→   219→  // Build families dynamically from product categories, grouped by seccion
   220→   220→  const familiasBySeccion = useMemo(() => {
   221→   221→    const result: Record<string, Familia[]> = { pastas: [], horneados: [], other: [] }
   222→   222→    const seen = new Map<string, Familia>()
   223→   223→
   224→   224→    for (const p of productos) {
   225→   225→      const catNombre = p.categoria.nombre
   226→   226→      if (!seen.has(catNombre)) {
   227→   227→        seen.set(catNombre, {
   228→   228→          nombre: catNombre,
   229→   229→          seccion: p.categoria.seccion,
   230→   230→          imagen: p.categoria.imagen || null,
   231→   231→          imagen_integral: p.categoria.imagen_integral || null,
   232→   232→          imagen_sin_gluten: p.categoria.imagen_sin_gluten || null,
   233→   233→          descripcion: p.categoria.descripcion || DESCRIPCIONES_DEFAULT[catNombre] || `Productos de ${catNombre.toLowerCase()}`,
   234→   234→          descripcionDB: p.categoria.descripcion,
   235→   235→          categoriaId: p.categoria.id,
   236→   236→        })
   237→   237→      }
   238→   238→    }
   239→   239→
   240→   240→    const knownOrder = ['Sorrentinos', 'Ñoquis', 'Tallarines', 'Cintas Anchas', 'Ravioles', 'Tapas', 'Empanadas', 'Tartas', 'Pastas frescas', 'Pastas secas', 'Salsas', 'Lasagnas y canelones', 'Postres', 'Facturas', 'Chipás', 'Pizzas']
   241→   241→
   242→   242→    const sorted = Array.from(seen.values()).sort((a, b) => {
   243→   243→      const seccionOrder = { pastas: 0, horneados: 1 }
   244→   244→      const sa = seccionOrder[a.seccion as keyof typeof seccionOrder] ?? 2
   245→   245→      const sb = seccionOrder[b.seccion as keyof typeof seccionOrder] ?? 2
   246→   246→      if (sa !== sb) return sa - sb
   247→   247→
   248→   248→      const ia = knownOrder.indexOf(a.nombre)
   249→   249→      const ib = knownOrder.indexOf(b.nombre)
   250→   250→      if (ia !== -1 && ib !== -1) return ia - ib
   251→   251→      if (ia !== -1) return -1
   252→   252→      if (ib !== -1) return 1
   253→   253→      return a.nombre.localeCompare(b.nombre)
   254→   254→    })
   255→   255→
   256→   256→    for (const familia of sorted) {
   257→   257→      const sec = familia.seccion === 'pastas' ? 'pastas' : familia.seccion === 'horneados' ? 'horneados' : 'other'
   258→   258→      result[sec].push(familia)
   259→   259→    }
   260→   260→
   261→   261→    return result
   262→   262→  }, [productos])
   263→   263→
   264→   264→  // Current section families
   265→   265→  const familias = useMemo(() => {
   266→   266→    if (!seccionActiva) return []
   267→   267→    const secFamilies = familiasBySeccion[seccionActiva] || []
   268→   268→    const otherFamilies = familiasBySeccion.other || []
   269→   269→    return [...secFamilies, ...otherFamilies]
   270→   270→  }, [familiasBySeccion, seccionActiva])
   271→   271→
   272→   272→  // Check which sections have products for current filter
   273→   273→  const seccionesConProductos = useMemo(() => {
   274→   274→    const result: Seccion[] = []
   275→   275→    for (const sec of SECCIONES) {
   276→   276→      const families = familiasBySeccion[sec.key] || []
   277→   277→      if (families.length > 0) result.push(sec.key)
   278→   278→    }
   279→   279→    return result
   280→   280→  }, [familiasBySeccion])
   281→   281→
   282→   282→  // Compute product counts per family
   283→   283→  const familiaData = useMemo(() => {
   284→   284→    const data: Record<string, { count: number; hasProducts: boolean }> = {}
   285→   285→    for (const familia of familias) {
   286→   286→      const prods = productos.filter(
   287→   287→        (p) => p.categoria.nombre === familia.nombre
   288→   288→      )
   289→   289→      data[familia.nombre] = {
   290→   290→        count: prods.length,
   291→   291→        hasProducts: prods.length > 0,
   292→   292→      }
   293→   293→    }
   294→   294→    return data
   295→   295→  }, [productos, familias])
   296→   296→
   297→   297→  // Products for the active family — paginated
   298→   298→  const productosFamilia = useMemo(() => {
   299→   299→    if (!familiaActiva) return []
   300→   300→    return productos.filter((p) => p.categoria.nombre === familiaActiva)
   301→   301→  }, [productos, familiaActiva])
   302→   302→
   303→   303→  const productosVisibles = useMemo(() => {
   304→   304→    return productosFamilia.slice(0, visibleCount)
   305→   305→  }, [productosFamilia, visibleCount])
   306→   306→
   307→   307→  const hasMore = visibleCount < productosFamilia.length
   308→   308→
   309→   309→  const handleFamiliaClick = useCallback((nombre: string) => {
   310→   310→    setFamiliaActiva((prev) => (prev === nombre ? null : nombre))
   311→   311→  }, [])
   312→   312→
   313→   313→  // Smooth scroll al contenedor de productos cuando se selecciona una familia
   314→   314→  useEffect(() => {
   315→   315→    if (!familiaActiva || !productosGridRef.current) return
   316→   316→    const yOffset = -100
   317→   317→    const y = productosGridRef.current.getBoundingClientRect().top + window.pageYOffset + yOffset
   318→   318→    if (y < window.pageYOffset || y > window.pageYOffset + window.innerHeight - 200) {
   319→   319→      window.scrollTo({ top: y, behavior: 'smooth' })
   320→   320→    }
   321→   321→  }, [familiaActiva])
   322→   322→
   323→   323→  const handleLoadMore = useCallback(() => {
   324→   324→    setVisibleCount((prev) => prev + PRODUCTS_PER_PAGE)
   325→   325→  }, [])
   326→   326→
   327→   327→  // Filter out families with no products
   328→   328→  const familiasVisibles = useMemo(
   329→   329→    () => familias.filter((f) => familiaData[f.nombre]?.hasProducts),
   330→   330→    [familias, familiaData]
   331→   331→  )
   332→   332→
   333→   333→  // Retry: re-fetch all
   334→   334→  const handleRetry = useCallback(async () => {
   335→   335→    fetchedRef.current = false
   336→   336→    setLoading(true)
   337→   337→    setError('')
   338→   338→    try {
   339→   339→      const results = await Promise.all(
   340→   340→        TODOS_FILTROS.map(async (tipo) => {
   341→   341→          const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   342→   342→          if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   343→   343→          const data = await res.json()
   344→   344→          return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   345→   345→        })
   346→   346→      )
   347→   347→      setCache((prev) => {
   348→   348→        const next = { ...prev }
   349→   349→        for (const r of results) next[r.tipo] = r.productos
   350→   350→        return next
   351→   351→      })
   352→   352→    } catch {
   353→   353→      setError('Error de conexión. Verificá tu internet e intentá de nuevo.')
   354→   354→    } finally {
   355→   355→      setLoading(false)
   356→   356→    }
   357→   357→  }, [])
   358→   358→
   359→   359→  const seccionConfig = seccionActiva ? SECCIONES.find((s) => s.key === seccionActiva)! : null
   360→   360→
   361→   361→  return (
   362→   362→    <section id="productos" className="min-h-screen flex flex-col justify-center py-12 sm:py-16 md:py-20 bg-crema">
   363→   363→      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 w-full">
   364→   364→        {/* Title */}
   365→   365→        <div className="text-center mb-8">
   366→   366→          <h2 className="text-3xl sm:text-4xl font-bold text-marron">
   367→   367→            Nuestros <span className="text-rojo">Productos</span>
   368→   368→          </h2>
   369→   369→          <div className="h-1 w-20 bg-mostaza mx-auto mt-4 rounded-full" />
   370→   370→          <p className="text-muted-foreground mt-3 max-w-lg mx-auto text-sm sm:text-base">
   371→   371→            Elaborados con ingredientes frescos y de calidad. Elegí la sección que más te guste.
   372→   372→          </p>
   373→   373→        </div>
   374→   374→
   375→   375→        {/* ═══════════════════════════════════════════════════════════
   376→   376→            STATE 1: No section selected → Show section hero cards
   377→   377→            ═══════════════════════════════════════════════════════════ */}
   378→   378→        {!seccionActiva && (
   379→   379→          <>
   380→   380→            {loading ? (
   381→   381→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-6 max-w-3xl mx-auto">
   382→   382→                {Array.from({ length: 2 }).map((_, i) => (
   383→   383→                  <div key={i} className="rounded-2xl border border-marron/10 bg-white p-8 animate-pulse flex flex-col items-center gap-4 min-h-[280px]">
   384→   384→                    <div className="h-16 w-16 bg-muted rounded-full" />
   385→   385→                    <div className="h-6 w-32 bg-muted rounded" />
   386→   386→                    <div className="h-4 w-48 bg-muted rounded" />
   387→   387→                    <div className="h-4 w-40 bg-muted rounded" />
   388→   388→                  </div>
   389→   389→                ))}
   390→   390→              </div>
   391→   391→            ) : error ? (
   392→   392→              <div className="text-center py-12">
   393→   393→                <span className="text-5xl mb-4 block">⚠️</span>
   394→   394→                <p className="text-muted-foreground mb-4">{error}</p>
   395→   395→                <Button
   396→   396→                  variant="outline"
   397→   397→                  onClick={handleRetry}
   398→   398→                  className="border-mostaza text-marron hover:bg-mostaza hover:text-marron"
   399→   399→                >
   400→   400→                  Reintentar
   401→   401→                </Button>
   402→   402→              </div>
   403→   403→            ) : (
   404→   404→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-6 max-w-3xl mx-auto">
   405→   405→                {SECCIONES.map((sec) => {
   406→   406→                  const hasProducts = seccionesConProductos.includes(sec.key)
   407→   407→                  return (
   408→   408→                    <button
