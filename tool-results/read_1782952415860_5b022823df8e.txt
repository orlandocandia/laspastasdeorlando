  1106→interface StatementResult {

  1112→/**
  1113→ * Parse DDL SQL by splitting on -- CreateTable / -- CreateIndex boundaries.
  1114→ * Each section after such a comment marker is treated as a single SQL statement.
  1115→ */
  1116→function parseDDLStatements(sql: string): string[] {
  1117→  const statements: string[] = []
  1118→
  1119→  // Split on the boundary markers (-- CreateTable or -- CreateIndex)
  1120→  const parts = sql.split(/\n-- Create(?:Table|Index)\n/)
  1121→
  1122→  for (const part of parts) {
  1123→    const trimmed = part.trim()
  1124→    if (trimmed.length === 0) continue
  1125→
  1126→    // Each part may end with a trailing semicolon and blank lines; keep it as-is
  1127→    // but ensure it's a valid SQL statement
  1128→    if (trimmed.startsWith('CREATE ') || trimmed.startsWith('create ')) {
  1129→      statements.push(trimmed)
  1130→    }
  1131→  }
  1132→
  1133→  return statements
  1134→}
