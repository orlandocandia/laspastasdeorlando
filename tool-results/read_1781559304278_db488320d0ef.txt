   125→   125→        <button
   126→   126→          onClick={() => inputRef.current?.click()}
   127→   127→          disabled={uploading}

   136→   136→        </button>
   137→   137→      )}
   138→   138→    </div>
   139→   139→  )

   142→   142→export default function CategoriasManager() {
   143→   143→  const [activeTab, setActiveTab] = useState<TipoCategoria>('materias-primas')
   144→   144→  const [categorias, setCategorias] = useState<CategoriaItem[]>([])
   145→   145→  const [loading, setLoading] = useState(true)
   146→   146→  const [nuevoNombre, setNuevoNombre] = useState('')
   147→   147→  const [nuevaDescripcion, setNuevaDescripcion] = useState('')
   148→   148→  const [nuevaImagen, setNuevaImagen] = useState<string | null>(null)
   149→   149→  const [creating, setCreating] = useState(false)
   150→   150→
   151→   151→  // Edit dialog
   152→   152→  const [editItem, setEditItem] = useState<CategoriaItem | null>(null)
   153→   153→  const [editNombre, setEditNombre] = useState('')
   154→   154→  const [editDescripcion, setEditDescripcion] = useState('')
   155→   155→  const [editImagen, setEditImagen] = useState<string | null>(null)
   156→   156→  const [editImagenIntegral, setEditImagenIntegral] = useState<string | null>(null)
   157→   157→  const [editImagenSinGluten, setEditImagenSinGluten] = useState<string | null>(null)
   158→   158→  const [editOpen, setEditOpen] = useState(false)
   159→   159→  const [saving, setSaving] = useState(false)
   160→   160→
   161→   161→  // Delete
   162→   162→  const [deleteItem, setDeleteItem] = useState<CategoriaItem | null>(null)
   163→   163→
   164→   164→  // Upload states
   165→   165→  const [uploadingNew, setUploadingNew] = useState(false)
   166→   166→  const [uploadingEditBase, setUploadingEditBase] = useState(false)
   167→   167→  const [uploadingEditIntegral, setUploadingEditIntegral] = useState(false)
   168→   168→  const [uploadingEditSinGluten, setUploadingEditSinGluten] = useState(false)
   169→   169→  const newInputRef = useRef<HTMLInputElement>(null)
   170→   170→  const editBaseInputRef = useRef<HTMLInputElement>(null)
   171→   171→  const editIntegralInputRef = useRef<HTMLInputElement>(null)
   172→   172→  const editSinGlutenInputRef = useRef<HTMLInputElement>(null)
   173→   173→
   174→   174→  const isProductosTab = activeTab === 'productos-terminados'
   175→   175→
   176→   176→  const fetchCategorias = useCallback(async () => {
   177→   177→    setLoading(true)

   179→   179→      const res = await fetch(`/api/categorias?tipo=${activeTab}`)
   180→   180→      if (!res.ok) throw new Error('Error al cargar categorías')
   181→   181→      const data = await res.json()

   184→   184→      toast.error('Error al cargar categorías')
   185→   185→    } finally {
   186→   186→      setLoading(false)
   187→   187→    }
   188→   188→  }, [activeTab])
   189→   189→
   190→   190→  useEffect(() => {
   191→   191→    fetchCategorias()
   192→   192→  }, [fetchCategorias])
   193→   193→
   194→   194→  const uploadImage = async (file: File): Promise<string | null> => {
   195→   195→    const formData = new FormData()
   196→   196→    formData.append('file', file)
   197→   197→
   198→   198→    const res = await fetch('/api/upload/categoria', {
   199→   199→      method: 'POST',
   200→   200→      body: formData,
   201→   201→    })
   202→   202→
   203→   203→    if (!res.ok) throw new Error('Error al subir imagen')
   204→   204→    const data = await res.json()
   205→   205→    return data.url
   206→   206→  }
   207→   207→
   208→   208→  const handleFileUpload = async (
   209→   209→    file: File,
   210→   210→    setUploading: (v: boolean) => void,
   211→   211→    inputRef: React.RefObject<HTMLInputElement | null>,
   212→   212→    onUrl: (url: string) => void
   213→   213→  ) => {
   214→   214→    if (!file.type.startsWith('image/')) {
   215→   215→      toast.error('Solo se permiten imágenes')
   216→   216→      return
   217→   217→    }
   218→   218→    setUploading(true)
   219→   219→    try {
   220→   220→      const url = await uploadImage(file)
   221→   221→      if (url) {
   222→   222→        onUrl(url)
   223→   223→        toast.success('Imagen subida')
   224→   224→      }
   225→   225→    } catch {
   226→   226→      toast.error('Error al subir la imagen')
   227→   227→    } finally {
   228→   228→      setUploading(false)
   229→   229→      if (inputRef.current) inputRef.current.value = ''
   230→   230→    }
   231→   231→  }
   232→   232→
   233→   233→  const handleNewImageUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   234→   234→    const file = e.target.files?.[0]
   235→   235→    if (!file) return
   236→   236→    await handleFileUpload(file, setUploadingNew, newInputRef, setNuevaImagen)
   237→   237→  }
   238→   238→
   239→   239→  const handleEditBaseUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   240→   240→    const file = e.target.files?.[0]
   241→   241→    if (!file) return
   242→   242→    await handleFileUpload(file, setUploadingEditBase, editBaseInputRef, setEditImagen)
   243→   243→  }
   244→   244→
   245→   245→  const handleEditIntegralUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   246→   246→    const file = e.target.files?.[0]
   247→   247→    if (!file) return
   248→   248→    await handleFileUpload(file, setUploadingEditIntegral, editIntegralInputRef, (url) => setEditImagenIntegral(url))
   249→   249→  }
   250→   250→
   251→   251→  const handleEditSinGlutenUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   252→   252→    const file = e.target.files?.[0]
   253→   253→    if (!file) return
   254→   254→    await handleFileUpload(file, setUploadingEditSinGluten, editSinGlutenInputRef, (url) => setEditImagenSinGluten(url))
   255→   255→  }
   256→   256→
   257→   257→  const handleCreate = async () => {
   258→   258→    if (!nuevoNombre.trim()) {
   259→   259→      toast.error('El nombre es requerido')
   260→   260→      return
   261→   261→    }
   262→   262→    setCreating(true)
   263→   263→    try {
   264→   264→      const body: Record<string, unknown> = {
   265→   265→        tipo: activeTab,
   266→   266→        nombre: nuevoNombre.trim(),
   267→   267→        descripcion: nuevaDescripcion.trim() || null,
   268→   268→      }
   269→   269→      if (isProductosTab) {
   270→   270→        body.imagen = nuevaImagen || null
   271→   271→      }
   272→   272→      const res = await fetch('/api/categorias', {
   273→   273→        method: 'POST',
   274→   274→        headers: { 'Content-Type': 'application/json' },
   275→   275→        body: JSON.stringify(body),
   276→   276→      })
   277→   277→      if (!res.ok) {
   278→   278→        const data = await res.json().catch(() => ({}))
   279→   279→        throw new Error(data.error || 'Error al crear categoría')
   280→   280→      }
   281→   281→      toast.success('Categoría creada')
   282→   282→      setNuevoNombre('')
   283→   283→      setNuevaDescripcion('')
   284→   284→      setNuevaImagen(null)
   285→   285→      fetchCategorias()
   286→   286→    } catch (error: any) {
   287→   287→      toast.error(error.message || 'Error al crear categoría')
   288→   288→    } finally {
   289→   289→      setCreating(false)
   290→   290→    }
   291→   291→  }
   292→   292→
   293→   293→  const handleEdit = (item: CategoriaItem) => {
   294→   294→    setEditItem(item)
   295→   295→    setEditNombre(item.nombre)
   296→   296→    setEditDescripcion(item.descripcion || '')
   297→   297→    setEditImagen(item.imagen || null)
   298→   298→    setEditImagenIntegral(item.imagen_integral || null)
   299→   299→    setEditImagenSinGluten(item.imagen_sin_gluten || null)
   300→   300→    setEditOpen(true)
   301→   301→  }
   302→   302→
   303→   303→  const handleSaveEdit = async () => {
   304→   304→    if (!editItem || !editNombre.trim()) return
   305→   305→    setSaving(true)
   306→   306→    try {
   307→   307→      const body: Record<string, unknown> = {
   308→   308→        id: editItem.id,
   309→   309→        tipo: activeTab,
   310→   310→        nombre: editNombre.trim(),
   311→   311→        descripcion: editDescripcion.trim() || null,
   312→   312→      }
   313→   313→      if (isProductosTab) {
   314→   314→        body.imagen = editImagen || null
   315→   315→        body.imagen_integral = editImagenIntegral || null
   316→   316→        body.imagen_sin_gluten = editImagenSinGluten || null
   317→   317→      }
   318→   318→      const res = await fetch('/api/categorias', {
   319→   319→        method: 'PUT',
   320→   320→        headers: { 'Content-Type': 'application/json' },
   321→   321→        body: JSON.stringify(body),
   322→   322→      })
   323→   323→      if (!res.ok) {
   324→   324→        const data = await res.json().catch(() => ({}))
   325→   325→        throw new Error(data.error || 'Error al actualizar categoría')
   326→   326→      }
   327→   327→      toast.success('Categoría actualizada')
   328→   328→      setEditOpen(false)
   329→   329→      setEditItem(null)
   330→   330→      fetchCategorias()
   331→   331→    } catch (error: any) {
   332→   332→      toast.error(error.message || 'Error al actualizar categoría')
   333→   333→    } finally {
   334→   334→      setSaving(false)
   335→   335→    }
   336→   336→  }
   337→   337→
   338→   338→  const handleDelete = async () => {
   339→   339→    if (!deleteItem) return
   340→   340→    try {
   341→   341→      const res = await fetch(`/api/categorias?id=${deleteItem.id}&tipo=${activeTab}`, {
   342→   342→        method: 'DELETE',
   343→   343→      })
   344→   344→      if (!res.ok) {
   345→   345→        const data = await res.json().catch(() => ({}))
   346→   346→        throw new Error(data.error || 'Error al eliminar categoría')
   347→   347→      }
   348→   348→      toast.success('Categoría eliminada')
   349→   349→      fetchCategorias()
   350→   350→    } catch (error: any) {
   351→   351→      toast.error(error.message || 'Error al eliminar categoría')
   352→   352→    } finally {
   353→   353→      setDeleteItem(null)
   354→   354→    }
   355→   355→  }
   356→   356→
   357→   357→  return (
   358→   358→    <div className="space-y-4">
   359→   359→      <Tabs value={activeTab} onValueChange={(v) => setActiveTab(v as TipoCategoria)}>
   360→   360→        <TabsList className="grid w-full grid-cols-3">
   361→   361→          {TABS.map((tab) => (
   362→   362→            <TabsTrigger key={tab.value} value={tab.value}>
   363→   363→              {tab.label}

   366→   366→        </TabsList>
   367→   367→
   368→   368→        {TABS.map((tab) => (
   369→   369→          <TabsContent key={tab.value} value={tab.value} className="space-y-4">

   372→   372→              <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-end">

   374→   374→                  <label className="text-sm font-medium text-marron mb-1 block">Nombre *</label>

   378→   378→                    onChange={(e) => setNuevoNombre(e.target.value)}
   379→   379→                    onKeyDown={(e) => {
   380→   380→                      if (e.key === 'Enter') handleCreate()

   382→   382→                  />
   383→   383→                </div>

   388→   388→                    value={tab.value === activeTab ? nuevaDescripcion : ''}
   389→   389→                    onChange={(e) => setNuevaDescripcion(e.target.value)}
   390→   390→                    onKeyDown={(e) => {
   391→   391→                      if (e.key === 'Enter') handleCreate()
   392→   392→                    }}
   393→   393→                  />
   394→   394→                </div>
   395→   395→                <Button
   396→   396→                  onClick={handleCreate}
   397→   397→                  disabled={creating || !nuevoNombre.trim()}
   398→   398→                  className="bg-mostaza hover:bg-mostaza/90 text-marron font-semibold shrink-0"
   399→   399→                >
   400→   400→                  {creating ? (
   401→   401→                    <Loader2 className="h-4 w-4 animate-spin" />
   402→   402→                  ) : (
   403→   403→                    <Plus className="mr-2 h-4 w-4" />
   404→   404→                  )}
   405→   405→                  Agregar
   406→   406→                </Button>
   407→   407→              </div>
   408→   408→
   409→   409→              {/* Image upload for productos-terminados */}
   410→   410→              {tab.value === 'productos-terminados' && (
   411→   411→                <div className="flex items-center gap-3">
   412→   412→                  <input
   413→   413→                    ref={newInputRef}
   414→   414→                    type="file"
   415→   415→                    accept="image/*"
   416→   416→                    onChange={handleNewImageUpload}
   417→   417→                    className="hidden"
   418→   418→                  />
   419→   419→                  {nuevaImagen ? (
   420→   420→                    <div className="relative group w-16 h-16 rounded-lg overflow-hidden border border-marron/10">
   421→   421→                      <Image
   422→   422→                        src={nuevaImagen}
   423→   423→                        alt="Imagen de categoría"
   424→   424→                        fill
   425→   425→                        loading="lazy"
   426→   426→                        className="object-cover"
   427→   427→                        sizes="96px"
   428→   428→                      />
   429→   429→                      <div className="absolute inset-0 bg-black/40 opacity-0 group-hover:opacity-100 transition-opacity flex items-center justify-center">
   430→   430→                        <Button
   431→   431→                          type="button"
   432→   432→                          variant="ghost"
   433→   433→                          size="icon"
   434→   434→                          className="h-6 w-6 text-white hover:bg-white/20"
   435→   435→                          onClick={() => setNuevaImagen(null)}
   436→   436→                        >
   437→   437→                          <X className="h-3 w-3" />
   438→   438→                        </Button>
   439→   439→                      </div>
   440→   440→                    </div>
   441→   441→                  ) : (
   442→   442→                    <button
   443→   443→                      onClick={() => newInputRef.current?.click()}
   444→   444→                      disabled={uploadingNew}
   445→   445→                      className="flex items-center gap-2 px-3 py-2 rounded-lg border-2 border-dashed border-marron/20 hover:border-mostaza/50 hover:bg-mostaza/5 transition-colors text-sm text-muted-foreground"
   446→   446→                    >
   447→   447→                      {uploadingNew ? (
   448→   448→                        <Loader2 className="h-4 w-4 animate-spin" />
   449→   449→                      ) : (
   450→   450→                        <ImagePlus className="h-4 w-4" />
   451→   451→                      )}
   452→   452→                      Subir imagen
   453→   453→                    </button>
   454→   454→                  )}
   455→   455→                </div>
   456→   456→              )}
   457→   457→            </div>
   458→   458→
   459→   459→            {/* Table */}
   460→   460→            {loading ? (
   461→   461→              <div className="flex items-center justify-center py-8">
   462→   462→                <Loader2 className="h-6 w-6 animate-spin text-mostaza" />
   463→   463→              </div>
   464→   464→            ) : categorias.length === 0 ? (
   465→   465→              <div className="text-center py-8 text-muted-foreground">
   466→   466→                No hay categorías de {tab.label.toLowerCase()} cargadas
   467→   467→              </div>
   468→   468→            ) : (
