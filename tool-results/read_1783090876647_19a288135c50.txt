   133→   133→interface ProductosProps {

   138→   138→export default function Productos({ filtroActivo = 'con_gluten', onFiltroChange }: ProductosProps) {
   139→   139→  // Cache: productos pre-fetcheados por tipo de harina
   140→   140→  const [cache, setCache] = useState<Record<FiltroHarina, ProductoPublico[]>>({
   141→   141→    con_gluten: [],
   142→   142→    integral: [],
   143→   143→    sin_gluten: [],
   144→   144→  })
   145→   145→  const [loading, setLoading] = useState(true)
   146→   146→  const [error, setError] = useState('')
   147→   147→  const [filtro, setFiltro] = useState<FiltroHarina>(filtroActivo)
   148→   148→  // null = no section selected (step 1), 'pastas'|'horneados' = section selected (step 2)
   149→   149→  const [seccionActiva, setSeccionActiva] = useState<Seccion | null>(null)
   150→   150→  const [familiaActiva, setFamiliaActiva] = useState<string | null>(null)
   151→   151→  const [visibleCount, setVisibleCount] = useState(PRODUCTS_PER_PAGE)
   152→   152→  const fetchedRef = useRef(false)
   153→   153→  const productosGridRef = useRef<HTMLDivElement>(null)
   154→   154→
   155→   155→  // Pre-fetch all 3 filter types in parallel on mount — no refetch on filter change
   156→   156→  useEffect(() => {
   157→   157→    if (fetchedRef.current) return
   158→   158→    fetchedRef.current = true
   159→   159→
   160→   160→    async function fetchAll() {
   161→   161→      setLoading(true)
   162→   162→      setError('')
   163→   163→      try {
   164→   164→        const results = await Promise.all(
   165→   165→          TODOS_FILTROS.map(async (tipo) => {
   166→   166→            const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   167→   167→            if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   168→   168→            const data = await res.json()
   169→   169→            return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   170→   170→          })
   171→   171→        )
   172→   172→        setCache((prev) => {
   173→   173→          const next = { ...prev }
   174→   174→          for (const r of results) next[r.tipo] = r.productos
   175→   175→          return next
   176→   176→        })
   177→   177→      } catch {
   178→   178→        setError('No se pudieron cargar los productos. Intentá más tarde.')
   179→   179→      } finally {
   180→   180→        setLoading(false)
   181→   181→      }
   182→   182→    }
   183→   183→    fetchAll()
   184→   184→  }, [])
   185→   185→
   186→   186→  // Reset visible count when family or section changes
   187→   187→  useEffect(() => {
   188→   188→    setVisibleCount(PRODUCTS_PER_PAGE)
   189→   189→  }, [familiaActiva, seccionActiva])
   190→   190→
   191→   191→  // Sync with parent filtro state — instant, no refetch
   192→   192→  useEffect(() => {
   193→   193→    if (filtroActivo !== filtro) {
   194→   194→      setFiltro(filtroActivo ?? 'con_gluten')
   195→   195→      setFamiliaActiva(null)
   196→   196→    }
   197→   197→  }, [filtroActivo, filtro])
   198→   198→
   199→   199→  const handleFiltroChange = useCallback((nuevoFiltro: FiltroHarina) => {
   200→   200→    setFiltro(nuevoFiltro)
   201→   201→    setFamiliaActiva(null)
   202→   202→    onFiltroChange?.(nuevoFiltro)
   203→   203→  }, [onFiltroChange])
   204→   204→
   205→   205→  const handleSeccionChange = useCallback((nuevaSeccion: Seccion) => {
   206→   206→    setSeccionActiva(nuevaSeccion)
   207→   207→    setFamiliaActiva(null)
   208→   208→    setFiltro('con_gluten') // Reset flour filter when changing section
   209→   209→    onFiltroChange?.('con_gluten')
   210→   210→  }, [onFiltroChange])
   211→   211→
   212→   212→  const handleVolverASecciones = useCallback(() => {
   213→   213→    setSeccionActiva(null)
   214→   214→    setFamiliaActiva(null)
   215→   215→    setFiltro('con_gluten')
   216→   216→    onFiltroChange?.('con_gluten')
   217→   217→  }, [onFiltroChange])
   218→   218→
   219→   219→  // Current products from cache — instant swap, no network
   220→   220→  const productos = cache[filtro]
   221→   221→
   222→   222→  // Build families dynamically from product categories, grouped by seccion
   223→   223→  // Effective section: producto.seccion ?? categoria.seccion
   224→   224→  // Products with null effective section are hidden (requirement 7)
   225→   225→  // A category can appear in BOTH sections if it has products of both types
   226→   226→  const familiasBySeccion = useMemo(() => {
   227→   227→    const result: Record<string, Familia[]> = { pastas: [], horneados: [], other: [] }
   228→   228→    // Dedup by (categoria nombre + effective seccion) so the same category
   229→   229→    // can appear once in pastas and once in horneados
   230→   230→    const seen = new Map<string, Familia>()
   231→   231→
   232→   232→    for (const p of productos) {
   233→   233→      const catNombre = p.categoria.nombre
   234→   234→      const effectiveSeccion = p.seccion ?? p.categoria.seccion
   235→   235→      // Skip products without an effective section (requirement 7)
   236→   236→      if (effectiveSeccion !== 'pastas' && effectiveSeccion !== 'horneados') continue
   237→   237→
   238→   238→      const dedupKey = `${catNombre}__${effectiveSeccion}`
   239→   239→      if (!seen.has(dedupKey)) {
   240→   240→        seen.set(dedupKey, {
   241→   241→          nombre: catNombre,
   242→   242→          seccion: effectiveSeccion,
   243→   243→          imagen: p.categoria.imagen || null,
   244→   244→          imagen_integral: p.categoria.imagen_integral || null,
   245→   245→          imagen_sin_gluten: p.categoria.imagen_sin_gluten || null,
   246→   246→          descripcion: p.categoria.descripcion || DESCRIPCIONES_DEFAULT[catNombre] || `Productos de ${catNombre.toLowerCase()}`,
   247→   247→          descripcionDB: p.categoria.descripcion,
   248→   248→          categoriaId: p.categoria.id,
   249→   249→        })
   250→   250→      }
   251→   251→    }
   252→   252→
   253→   253→    const knownOrder = ['Sorrentinos', 'Ñoquis', 'Tallarines', 'Cintas Anchas', 'Ravioles', 'Tapas', 'Empanadas', 'Tartas', 'Pastas frescas', 'Pastas secas', 'Salsas', 'Lasagnas y canelones', 'Postres', 'Facturas', 'Chipás', 'Pizzas']
   254→   254→
   255→   255→    const sorted = Array.from(seen.values()).sort((a, b) => {
   256→   256→      const seccionOrder = { pastas: 0, horneados: 1 }
   257→   257→      const sa = seccionOrder[a.seccion as keyof typeof seccionOrder] ?? 2
   258→   258→      const sb = seccionOrder[b.seccion as keyof typeof seccionOrder] ?? 2
   259→   259→      if (sa !== sb) return sa - sb
   260→   260→
   261→   261→      const ia = knownOrder.indexOf(a.nombre)
   262→   262→      const ib = knownOrder.indexOf(b.nombre)
   263→   263→      if (ia !== -1 && ib !== -1) return ia - ib
   264→   264→      if (ia !== -1) return -1
   265→   265→      if (ib !== -1) return 1
   266→   266→      return a.nombre.localeCompare(b.nombre)
   267→   267→    })
   268→   268→
   269→   269→    for (const familia of sorted) {
   270→   270→      const sec = familia.seccion === 'pastas' ? 'pastas' : familia.seccion === 'horneados' ? 'horneados' : 'other'
   271→   271→      result[sec].push(familia)
   272→   272→    }
   273→   273→
   274→   274→    return result
   275→   275→  }, [productos])
   276→   276→
   277→   277→  // Current section families — only families with an explicit section
   278→   278→  // (products with null effective section are hidden, requirement 7)
   279→   279→  const familias = useMemo(() => {
   280→   280→    if (!seccionActiva) return []
   281→   281→    return familiasBySeccion[seccionActiva] || []
   282→   282→  }, [familiasBySeccion, seccionActiva])
   283→   283→
   284→   284→  // Check which sections have products for current filter
   285→   285→  const seccionesConProductos = useMemo(() => {
   286→   286→    const result: Seccion[] = []
   287→   287→    for (const sec of SECCIONES) {
   288→   288→      const families = familiasBySeccion[sec.key] || []
   289→   289→      if (families.length > 0) result.push(sec.key)
   290→   290→    }
   291→   291→    return result
   292→   292→  }, [familiasBySeccion])
   293→   293→
   294→   294→  // Compute product counts per family — must respect effective section
   295→   295→  // so a category that appears in both sections only counts its
   296→   296→  // products from the corresponding section
   297→   297→  const familiaData = useMemo(() => {
   298→   298→    const data: Record<string, { count: number; hasProducts: boolean }> = {}
   299→   299→    for (const familia of familias) {
   300→   300→      const prods = productos.filter((p) => {
   301→   301→        if (p.categoria.nombre !== familia.nombre) return false
   302→   302→        const effSec = p.seccion ?? p.categoria.seccion
   303→   303→        return effSec === familia.seccion
   304→   304→      })
   305→   305→      data[familia.nombre] = {
   306→   306→        count: prods.length,
   307→   307→        hasProducts: prods.length > 0,
   308→   308→      }
   309→   309→    }
   310→   310→    return data
   311→   311→  }, [productos, familias])
   312→   312→
   313→   313→  // Products for the active family — filtered by both family name and effective section
   314→   314→  const productosFamilia = useMemo(() => {
   315→   315→    if (!familiaActiva || !seccionActiva) return []
   316→   316→    return productos.filter((p) => {
   317→   317→      if (p.categoria.nombre !== familiaActiva) return false
   318→   318→      const effSec = p.seccion ?? p.categoria.seccion
   319→   319→      return effSec === seccionActiva
   320→   320→    })
   321→   321→  }, [productos, familiaActiva, seccionActiva])
   322→   322→
   323→   323→  const productosVisibles = useMemo(() => {
   324→   324→    return productosFamilia.slice(0, visibleCount)
   325→   325→  }, [productosFamilia, visibleCount])
   326→   326→
   327→   327→  const hasMore = visibleCount < productosFamilia.length
   328→   328→
   329→   329→  const handleFamiliaClick = useCallback((nombre: string) => {
   330→   330→    setFamiliaActiva((prev) => (prev === nombre ? null : nombre))
   331→   331→  }, [])
   332→   332→
   333→   333→  // Smooth scroll al contenedor de productos cuando se selecciona una familia
   334→   334→  useEffect(() => {
   335→   335→    if (!familiaActiva || !productosGridRef.current) return
   336→   336→    const yOffset = -100
   337→   337→    const y = productosGridRef.current.getBoundingClientRect().top + window.pageYOffset + yOffset
   338→   338→    if (y < window.pageYOffset || y > window.pageYOffset + window.innerHeight - 200) {
   339→   339→      window.scrollTo({ top: y, behavior: 'smooth' })
   340→   340→    }
   341→   341→  }, [familiaActiva])
   342→   342→
   343→   343→  const handleLoadMore = useCallback(() => {
   344→   344→    setVisibleCount((prev) => prev + PRODUCTS_PER_PAGE)
   345→   345→  }, [])
   346→   346→
   347→   347→  // Filter out families with no products
   348→   348→  const familiasVisibles = useMemo(
   349→   349→    () => familias.filter((f) => familiaData[f.nombre]?.hasProducts),
   350→   350→    [familias, familiaData]
   351→   351→  )
   352→   352→
   353→   353→  // Retry: re-fetch all
   354→   354→  const handleRetry = useCallback(async () => {
   355→   355→    fetchedRef.current = false
   356→   356→    setLoading(true)
   357→   357→    setError('')
   358→   358→    try {
   359→   359→      const results = await Promise.all(
   360→   360→        TODOS_FILTROS.map(async (tipo) => {
   361→   361→          const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   362→   362→          if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   363→   363→          const data = await res.json()
   364→   364→          return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   365→   365→        })
   366→   366→      )
   367→   367→      setCache((prev) => {
   368→   368→        const next = { ...prev }
   369→   369→        for (const r of results) next[r.tipo] = r.productos
   370→   370→        return next
   371→   371→      })
   372→   372→    } catch {
   373→   373→      setError('Error de conexión. Verificá tu internet e intentá de nuevo.')
   374→   374→    } finally {
   375→   375→      setLoading(false)
   376→   376→    }
   377→   377→  }, [])
   378→   378→
   379→   379→  const seccionConfig = seccionActiva ? SECCIONES.find((s) => s.key === seccionActiva)! : null
   380→   380→
   381→   381→  return (
   382→   382→    <section id="productos" className="min-h-screen flex flex-col justify-start py-12 sm:py-16 md:py-20 bg-crema">
   383→   383→      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 w-full">
   384→   384→        {/* Title */}
   385→   385→        <div className="text-center mb-12">
   386→   386→          <h2 className="text-3xl sm:text-4xl font-bold text-marron">
   387→   387→            Nuestros <span className="text-rojo">Productos</span>
   388→   388→          </h2>
   389→   389→          <div className="h-1 w-20 bg-mostaza mx-auto mt-4 rounded-full" />
   390→   390→          <p className="text-muted-foreground mt-3 max-w-lg mx-auto text-sm sm:text-base">
   391→   391→            Elaborados con ingredientes frescos y de calidad. Elegí la sección que más te guste.
   392→   392→          </p>
   393→   393→        </div>
   394→   394→
   395→   395→        {/* ═══════════════════════════════════════════════════════════
   396→   396→            STATE 1: No section selected → Show section hero cards
   397→   397→            ═══════════════════════════════════════════════════════════ */}
   398→   398→        {!seccionActiva && (
   399→   399→          <>
   400→   400→            {loading ? (
   401→   401→              <div className="flex flex-col md:flex-row justify-center items-stretch gap-6 md:gap-10 w-full max-w-5xl mx-auto">
