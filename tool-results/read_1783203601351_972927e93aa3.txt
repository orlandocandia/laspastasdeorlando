   102→// Imagen dinámica según filtro para familias con variante integral/sin gluten
   103→function getFamiliaImagen(familia: Familia, filtro: FiltroHarina): string {
   104→  if (filtro === 'integral' && familia.imagen_integral) return familia.imagen_integral
   105→  if (filtro === 'sin_gluten' && familia.imagen_sin_gluten) return familia.imagen_sin_gluten
   106→
   107→  if (familia.nombre === 'Tallarines' && filtro === 'integral') {
   108→    return '/images/familias/tallarinesintegrales.png'
   109→  }
   110→  if (familia.nombre === 'Cintas Anchas' && filtro === 'integral') {
   111→    return '/images/familias/cintasanchasintegrales.png'
   112→  }
   113→  if (familia.nombre === 'Tapas' && filtro === 'integral') {
   114→    return '/images/familias/tapasintegrales.png'
   115→  }
   116→  if (familia.nombre === 'Tapas' && filtro === 'sin_gluten') {
   117→    return '/images/familias/tapassingluten.png'
   118→  }
   119→
   120→  if (familia.imagen) return familia.imagen
   121→  return IMAGENES_DEFAULT[familia.nombre] || '/images/placeholder-producto.jpg'
   122→}

   124→const FILTROS: { key: FiltroHarina; label: string; icon: React.ReactNode }[] = [

   130→const TODOS_FILTROS: FiltroHarina[] = ['con_gluten', 'integral', 'sin_gluten']

   132→const PRODUCTS_PER_PAGE = 50
   133→
   134→interface ProductosProps {
   135→  filtroActivo?: FiltroHarina

   139→export default function Productos({ filtroActivo = 'con_gluten', onFiltroChange }: ProductosProps) {
   140→  // Cache: productos pre-fetcheados por tipo de harina
   141→  const [cache, setCache] = useState<Record<FiltroHarina, ProductoPublico[]>>({
   142→    con_gluten: [],
   143→    integral: [],
   144→    sin_gluten: [],
   145→  })
   146→  const [loading, setLoading] = useState(true)
   147→  const [error, setError] = useState('')
   148→  const [filtro, setFiltro] = useState<FiltroHarina>(filtroActivo)
   149→  // null = no section selected (step 1), 'pastas'|'horneados' = section selected (step 2)
   150→  const [seccionActiva, setSeccionActiva] = useState<Seccion | null>(null)
   151→  const [familiaActiva, setFamiliaActiva] = useState<string | null>(null)
   152→  const [visibleCount, setVisibleCount] = useState(PRODUCTS_PER_PAGE)
   153→  const [productoPromociones, setProductoPromociones] = useState<Record<number, PromocionInfo>>({})
   154→  const fetchedRef = useRef(false)
   155→  const productosGridRef = useRef<HTMLDivElement>(null)
   156→
   157→  // Pre-fetch all 3 filter types in parallel on mount — no refetch on filter change
   158→  useEffect(() => {
   159→    if (fetchedRef.current) return
   160→    fetchedRef.current = true
   161→
   162→    async function fetchAll() {
   163→      setLoading(true)
   164→      setError('')
   165→      try {
   166→        const [results, promoRes] = await Promise.all([
   167→          Promise.all(
   168→            TODOS_FILTROS.map(async (tipo) => {
   169→              const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   170→              if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   171→              const data = await res.json()
   172→              return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   173→            })
   174→          ),
   175→          fetch('/api/promociones/public').then(async (res) => {
   176→            if (res.ok) {
   177→              const data = await res.json()
   178→              return data.productoPromociones || {}
   179→            }
   180→            return {}
   181→          }).catch(() => ({})),
   182→        ])
   183→        setCache((prev) => {
   184→          const next = { ...prev }
   185→          for (const r of results) next[r.tipo] = r.productos
   186→          return next
   187→        })
   188→        setProductoPromociones(promoRes as Record<number, PromocionInfo>)
   189→      } catch {
   190→        setError('No se pudieron cargar los productos. Intentá más tarde.')
   191→      } finally {
   192→        setLoading(false)
   193→      }
   194→    }
   195→    fetchAll()
   196→  }, [])
   197→
   198→  // Reset visible count when family or section changes
   199→  useEffect(() => {
   200→    setVisibleCount(PRODUCTS_PER_PAGE)
   201→  }, [familiaActiva, seccionActiva])
   202→
   203→  // Sync with parent filtro state — instant, no refetch
   204→  useEffect(() => {
   205→    if (filtroActivo !== filtro) {
   206→      setFiltro(filtroActivo ?? 'con_gluten')
   207→      setFamiliaActiva(null)
   208→    }
   209→  }, [filtroActivo, filtro])
   210→
   211→  const handleFiltroChange = useCallback((nuevoFiltro: FiltroHarina) => {
   212→    setFiltro(nuevoFiltro)
   213→    setFamiliaActiva(null)
   214→    onFiltroChange?.(nuevoFiltro)
   215→  }, [onFiltroChange])
   216→
   217→  const handleSeccionChange = useCallback((nuevaSeccion: Seccion) => {
   218→    setSeccionActiva(nuevaSeccion)
   219→    setFamiliaActiva(null)
   220→    setFiltro('con_gluten') // Reset flour filter when changing section
   221→    onFiltroChange?.('con_gluten')
   222→  }, [onFiltroChange])
   223→
   224→  const handleVolverASecciones = useCallback(() => {
   225→    setSeccionActiva(null)
   226→    setFamiliaActiva(null)
   227→    setFiltro('con_gluten')
   228→    onFiltroChange?.('con_gluten')
   229→  }, [onFiltroChange])
   230→
   231→  // Current products from cache — instant swap, no network
   232→  const productos = cache[filtro]
   233→
   234→  // Build families dynamically from product categories, grouped by seccion
   235→  // Effective section: producto.seccion ?? categoria.seccion
   236→  // Products with null effective section are hidden (requirement 7)
   237→  // A category can appear in BOTH sections if it has products of both types
   238→  const familiasBySeccion = useMemo(() => {
   239→    const result: Record<string, Familia[]> = { pastas: [], horneados: [], other: [] }
   240→    // Dedup by (categoria nombre + effective seccion) so the same category
   241→    // can appear once in pastas and once in horneados
   242→    const seen = new Map<string, Familia>()
   243→
   244→    for (const p of productos) {
   245→      const catNombre = p.categoria.nombre
   246→      const effectiveSeccion = p.seccion ?? p.categoria.seccion
   247→      // Skip products without an effective section (requirement 7)
   248→      if (effectiveSeccion !== 'pastas' && effectiveSeccion !== 'horneados') continue
   249→
   250→      const dedupKey = `${catNombre}__${effectiveSeccion}`
   251→      if (!seen.has(dedupKey)) {
   252→        seen.set(dedupKey, {
   253→          nombre: catNombre,
   254→          seccion: effectiveSeccion,
   255→          imagen: p.categoria.imagen || null,
   256→          imagen_integral: p.categoria.imagen_integral || null,
   257→          imagen_sin_gluten: p.categoria.imagen_sin_gluten || null,
   258→          descripcion: p.categoria.descripcion || DESCRIPCIONES_DEFAULT[catNombre] || `Productos de ${catNombre.toLowerCase()}`,
   259→          descripcionDB: p.categoria.descripcion,
   260→          categoriaId: p.categoria.id,
   261→        })
   262→      }
   263→    }
   264→
   265→    const knownOrder = ['Sorrentinos', 'Ñoquis', 'Tallarines', 'Cintas Anchas', 'Ravioles', 'Tapas', 'Empanadas', 'Tartas', 'Pastas frescas', 'Pastas secas', 'Salsas', 'Lasagnas y canelones', 'Postres', 'Facturas', 'Chipás', 'Pizzas']
   266→
   267→    const sorted = Array.from(seen.values()).sort((a, b) => {
   268→      const seccionOrder = { pastas: 0, horneados: 1 }
   269→      const sa = seccionOrder[a.seccion as keyof typeof seccionOrder] ?? 2
   270→      const sb = seccionOrder[b.seccion as keyof typeof seccionOrder] ?? 2
   271→      if (sa !== sb) return sa - sb
   272→
   273→      const ia = knownOrder.indexOf(a.nombre)
   274→      const ib = knownOrder.indexOf(b.nombre)
   275→      if (ia !== -1 && ib !== -1) return ia - ib
   276→      if (ia !== -1) return -1
   277→      if (ib !== -1) return 1
   278→      return a.nombre.localeCompare(b.nombre)
   279→    })
   280→
   281→    for (const familia of sorted) {
   282→      const sec = familia.seccion === 'pastas' ? 'pastas' : familia.seccion === 'horneados' ? 'horneados' : 'other'
   283→      result[sec].push(familia)
   284→    }
   285→
   286→    return result
   287→  }, [productos])
   288→
   289→  // Current section families — only families with an explicit section
   290→  // (products with null effective section are hidden, requirement 7)
   291→  const familias = useMemo(() => {
   292→    if (!seccionActiva) return []
   293→    return familiasBySeccion[seccionActiva] || []
   294→  }, [familiasBySeccion, seccionActiva])
   295→
   296→  // Check which sections have products for current filter
   297→  const seccionesConProductos = useMemo(() => {
   298→    const result: Seccion[] = []
   299→    for (const sec of SECCIONES) {
   300→      const families = familiasBySeccion[sec.key] || []
   301→      if (families.length > 0) result.push(sec.key)
   302→    }
   303→    return result
   304→  }, [familiasBySeccion])
   305→
   306→  // Compute product counts per family — must respect effective section
   307→  // so a category that appears in both sections only counts its
   308→  // products from the corresponding section
   309→  const familiaData = useMemo(() => {
   310→    const data: Record<string, { count: number; hasProducts: boolean }> = {}
   311→    for (const familia of familias) {
   312→      const prods = productos.filter((p) => {
   313→        if (p.categoria.nombre !== familia.nombre) return false
   314→        const effSec = p.seccion ?? p.categoria.seccion
   315→        return effSec === familia.seccion
   316→      })
   317→      data[familia.nombre] = {
   318→        count: prods.length,
   319→        hasProducts: prods.length > 0,
   320→      }
   321→    }
   322→    return data
   323→  }, [productos, familias])
   324→
   325→  // Products for the active family — filtered by both family name and effective section
   326→  const productosFamilia = useMemo(() => {
   327→    if (!familiaActiva || !seccionActiva) return []
   328→    return productos.filter((p) => {
   329→      if (p.categoria.nombre !== familiaActiva) return false
   330→      const effSec = p.seccion ?? p.categoria.seccion
   331→      return effSec === seccionActiva
   332→    })
   333→  }, [productos, familiaActiva, seccionActiva])
   334→
   335→  const productosVisibles = useMemo(() => {
   336→    return productosFamilia.slice(0, visibleCount)
   337→  }, [productosFamilia, visibleCount])
   338→
   339→  const hasMore = visibleCount < productosFamilia.length
   340→
   341→  const handleFamiliaClick = useCallback((nombre: string) => {
   342→    setFamiliaActiva((prev) => (prev === nombre ? null : nombre))
   343→  }, [])
   344→
   345→  // Smooth scroll al contenedor de productos cuando se selecciona una familia
   346→  useEffect(() => {
   347→    if (!familiaActiva || !productosGridRef.current) return
   348→    const yOffset = -100
   349→    const y = productosGridRef.current.getBoundingClientRect().top + window.pageYOffset + yOffset
   350→    if (y < window.pageYOffset || y > window.pageYOffset + window.innerHeight - 200) {
   351→      window.scrollTo({ top: y, behavior: 'smooth' })
   352→    }
   353→  }, [familiaActiva])
   354→
   355→  const handleLoadMore = useCallback(() => {
   356→    setVisibleCount((prev) => prev + PRODUCTS_PER_PAGE)
   357→  }, [])
   358→
   359→  // Filter out families with no products
   360→  const familiasVisibles = useMemo(
   361→    () => familias.filter((f) => familiaData[f.nombre]?.hasProducts),
   362→    [familias, familiaData]
   363→  )
   364→
   365→  // Retry: re-fetch all
   366→  const handleRetry = useCallback(async () => {
   367→    fetchedRef.current = false
   368→    setLoading(true)
   369→    setError('')
   370→    try {
   371→      const results = await Promise.all(
   372→        TODOS_FILTROS.map(async (tipo) => {
   373→          const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   374→          if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   375→          const data = await res.json()
   376→          return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   377→        })
   378→      )
   379→      setCache((prev) => {
   380→        const next = { ...prev }
   381→        for (const r of results) next[r.tipo] = r.productos
   382→        return next
   383→      })
   384→    } catch {
   385→      setError('Error de conexión. Verificá tu internet e intentá de nuevo.')
   386→    } finally {
   387→      setLoading(false)
   388→    }
   389→  }, [])
   390→
   391→  const seccionConfig = seccionActiva ? SECCIONES.find((s) => s.key === seccionActiva)! : null
   392→
   393→  return (
   394→    <section id="productos" className="min-h-screen flex flex-col justify-start py-12 sm:py-16 md:py-20 bg-crema">
   395→      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 w-full">
   396→        {/* Title */}
   397→        <div className="text-center mb-12">
   398→          <h2 className="text-3xl sm:text-4xl font-bold text-marron">
   399→            Nuestros <span className="text-rojo">Productos</span>
   400→          </h2>
   401→          <div className="h-1 w-20 bg-mostaza mx-auto mt-4 rounded-full" />
   402→          <p className="text-muted-foreground mt-3 max-w-lg mx-auto text-sm sm:text-base">
   403→            Elaborados con ingredientes frescos y de calidad. Elegí la sección que más te guste.
   404→          </p>
   405→        </div>
   406→
   407→        {/* ═══════════════════════════════════════════════════════════
   408→            STATE 1: No section selected → Show section hero cards
   409→            ═══════════════════════════════════════════════════════════ */}
   410→        {!seccionActiva && (
   411→          <>
   412→            {loading ? (
   413→              <div className="flex flex-col md:flex-row justify-center items-stretch gap-6 md:gap-10 w-full max-w-5xl mx-auto">
   414→                {Array.from({ length: 2 }).map((_, i) => (
   415→                  <div key={i} className="rounded-2xl border border-marron/10 bg-white animate-pulse flex-1 max-w-sm md:max-w-md lg:max-w-lg flex flex-col items-center p-6 sm:p-8 md:p-10 gap-5">
   416→                    <div className="w-40 h-40 md:w-48 md:h-48 lg:w-56 lg:h-56 bg-muted rounded-full" />
   417→                    <div className="h-7 w-32 bg-muted rounded" />
   418→                    <div className="h-10 w-36 bg-muted rounded-full" />
   419→                  </div>
   420→                ))}
   421→              </div>
   422→            ) : error ? (
   423→              <div className="text-center py-12">
   424→                <span className="text-5xl mb-4 block">⚠️</span>
   425→                <p className="text-muted-foreground mb-4">{error}</p>
   426→                <Button
   427→                  variant="outline"
   428→                  onClick={handleRetry}
   429→                  className="border-mostaza text-marron hover:bg-mostaza hover:text-marron"
   430→                >
   431→                  Reintentar
   432→                </Button>
   433→              </div>
   434→            ) : (
   435→              <div className="flex flex-col md:flex-row justify-center items-stretch gap-6 md:gap-10 w-full max-w-5xl mx-auto">
   436→                {SECCIONES.map((sec) => {
   437→                  const hasProducts = seccionesConProductos.includes(sec.key)
   438→                  return (
   439→                    <button
   440→                      key={sec.key}
   441→                      onClick={() => handleSeccionChange(sec.key)}
   442→                      className={`
   443→                        group relative rounded-2xl border-2 bg-white
   444→                        flex flex-col items-center text-center
   445→                        flex-1 max-w-sm md:max-w-md lg:max-w-lg
   446→                        p-6 sm:p-8 md:p-10 gap-5
   447→                        transition-all duration-300 cursor-pointer
   448→                        border-marron/10 hover:border-mostaza hover:shadow-xl hover:scale-[1.02]
   449→                        shadow-md
   450→                      `}
   451→                    >
   452→                      {/* Imagen circular — responsive: 160 / 192 / 224 px */}
   453→                      <div className="w-40 h-40 md:w-48 md:h-48 lg:w-56 lg:h-56 rounded-full overflow-hidden border-4 border-mostaza/20 flex-shrink-0 transition-all duration-300 group-hover:scale-105 group-hover:shadow-[0_8px_24px_rgba(0,0,0,0.18)] group-hover:border-mostaza/40">
   454→                        <Image
   455→                          src={sec.imagen}
   456→                          alt={sec.label}
   457→                          width={224}
   458→                          height={224}
   459→                          className="w-full h-full object-cover object-center"
   460→                        />
   461→                      </div>
   462→
   463→                      {/* Título */}
   464→                      <h3 className="text-xl md:text-2xl lg:text-3xl font-bold text-marron">
   465→                        {sec.label}
   466→                      </h3>
