   103→// Imagen dinámica según filtro para familias con variante integral/sin gluten
   104→function getFamiliaImagen(familia: Familia, filtro: FiltroHarina): string {
   105→  if (filtro === 'integral' && familia.imagen_integral) return familia.imagen_integral
   106→  if (filtro === 'sin_gluten' && familia.imagen_sin_gluten) return familia.imagen_sin_gluten
   107→
   108→  if (familia.nombre === 'Tallarines' && filtro === 'integral') {
   109→    return '/images/familias/tallarinesintegrales.png'
   110→  }
   111→  if (familia.nombre === 'Cintas Anchas' && filtro === 'integral') {
   112→    return '/images/familias/cintasanchasintegrales.png'
   113→  }
   114→  if (familia.nombre === 'Tapas' && filtro === 'integral') {
   115→    return '/images/familias/tapasintegrales.png'
   116→  }
   117→  if (familia.nombre === 'Tapas' && filtro === 'sin_gluten') {
   118→    return '/images/familias/tapassingluten.png'
   119→  }
   120→
   121→  if (familia.imagen) return familia.imagen
   122→  return IMAGENES_DEFAULT[familia.nombre] || '/images/placeholder-producto.jpg'
   123→}

   125→const FILTROS: { key: FiltroHarina; label: string; icon: React.ReactNode }[] = [

   129→]
   130→
   131→const TODOS_FILTROS: FiltroHarina[] = ['con_gluten', 'integral', 'sin_gluten']
   132→
   133→const PRODUCTS_PER_PAGE = 50
   134→
   135→// Promo summary for banner display
   136→interface PromoBanner {
   137→  id: number
   138→  nombre: string
   139→  descripcion: string | null
   140→  descuento_label: string
   141→  tipo: string
   142→  fecha_fin: string | null

   145→interface ProductosProps {
   146→  filtroActivo?: FiltroHarina
   147→  onFiltroChange?: (filtro: FiltroHarina) => void
   148→}
   149→
   150→export default function Productos({ filtroActivo = 'con_gluten', onFiltroChange }: ProductosProps) {
   151→  // Cache: productos pre-fetcheados por tipo de harina
   152→  const [cache, setCache] = useState<Record<FiltroHarina, ProductoPublico[]>>({
   153→    con_gluten: [],
   154→    integral: [],
   155→    sin_gluten: [],
   156→  })
   157→  const [loading, setLoading] = useState(true)
   158→  const [error, setError] = useState('')
   159→  const [filtro, setFiltro] = useState<FiltroHarina>(filtroActivo)
   160→  // null = no section selected (step 1), 'pastas'|'horneados' = section selected (step 2)
   161→  const [seccionActiva, setSeccionActiva] = useState<Seccion | null>(null)
   162→  const [familiaActiva, setFamiliaActiva] = useState<string | null>(null)
   163→  const [visibleCount, setVisibleCount] = useState(PRODUCTS_PER_PAGE)
   164→  const [productoPromociones, setProductoPromociones] = useState<Record<number, PromocionInfo>>({})
   165→  const [promociones, setPromociones] = useState<PromoBanner[]>([])
   166→  const [soloOfertas, setSoloOfertas] = useState(false)
   167→  const fetchedRef = useRef(false)
   168→  const productosGridRef = useRef<HTMLDivElement>(null)
   169→
   170→  // Pre-fetch all 3 filter types in parallel on mount — no refetch on filter change
   171→  useEffect(() => {
   172→    if (fetchedRef.current) return
   173→    fetchedRef.current = true
   174→
   175→    async function fetchAll() {
   176→      setLoading(true)
   177→      setError('')
   178→      try {
   179→        const [results, promoRes] = await Promise.all([
   180→          Promise.all(
   181→            TODOS_FILTROS.map(async (tipo) => {
   182→              const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   183→              if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   184→              const data = await res.json()
   185→              return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   186→            })
   187→          ),
   188→          fetch('/api/promociones/public').then(async (res) => {
   189→            if (res.ok) {
   190→              const data = await res.json()
   191→              return data
   192→            }
   193→            return { productoPromociones: {}, promociones: [] }
   194→          }).catch(() => ({ productoPromociones: {}, promociones: [] })),
   195→        ])
   196→        setCache((prev) => {
   197→          const next = { ...prev }
   198→          for (const r of results) next[r.tipo] = r.productos
   199→          return next
   200→        })
   201→        setProductoPromociones(promoRes.productoPromociones as Record<number, PromocionInfo> || {})
   202→        setPromociones((promoRes.promociones || []) as PromoBanner[])
   203→      } catch {
   204→        setError('No se pudieron cargar los productos. Intentá más tarde.')
   205→      } finally {
   206→        setLoading(false)
   207→      }
   208→    }
   209→    fetchAll()
   210→  }, [])
   211→
   212→  // Reset visible count when family or section changes
   213→  useEffect(() => {
   214→    setVisibleCount(PRODUCTS_PER_PAGE)
   215→  }, [familiaActiva, seccionActiva])
   216→
   217→  // Sync with parent filtro state — instant, no refetch
   218→  useEffect(() => {
   219→    if (filtroActivo !== filtro) {
   220→      setFiltro(filtroActivo ?? 'con_gluten')
   221→      setFamiliaActiva(null)
   222→    }
   223→  }, [filtroActivo, filtro])
   224→
   225→  const handleFiltroChange = useCallback((nuevoFiltro: FiltroHarina) => {
   226→    setFiltro(nuevoFiltro)
   227→    setFamiliaActiva(null)
   228→    onFiltroChange?.(nuevoFiltro)
   229→  }, [onFiltroChange])
   230→
   231→  const handleSeccionChange = useCallback((nuevaSeccion: Seccion) => {
   232→    setSeccionActiva(nuevaSeccion)
   233→    setFamiliaActiva(null)
   234→    setFiltro('con_gluten') // Reset flour filter when changing section
   235→    setSoloOfertas(false)
   236→    onFiltroChange?.('con_gluten')
   237→  }, [onFiltroChange])
   238→
   239→  const handleVolverASecciones = useCallback(() => {
   240→    setSeccionActiva(null)
   241→    setFamiliaActiva(null)
   242→    setFiltro('con_gluten')
   243→    setSoloOfertas(false)
   244→    onFiltroChange?.('con_gluten')
   245→  }, [onFiltroChange])
   246→
   247→  // Current products from cache — instant swap, no network
   248→  const productos = cache[filtro]
   249→
   250→  // Build families dynamically from product categories, grouped by seccion
   251→  const familiasBySeccion = useMemo(() => {
   252→    const result: Record<string, Familia[]> = { pastas: [], horneados: [], other: [] }
   253→    const seen = new Map<string, Familia>()
   254→
   255→    for (const p of productos) {
   256→      const catNombre = p.categoria.nombre
   257→      const effectiveSeccion = p.seccion ?? p.categoria.seccion
   258→      if (effectiveSeccion !== 'pastas' && effectiveSeccion !== 'horneados') continue
   259→
   260→      const dedupKey = `${catNombre}__${effectiveSeccion}`
   261→      if (!seen.has(dedupKey)) {
   262→        seen.set(dedupKey, {
   263→          nombre: catNombre,
   264→          seccion: effectiveSeccion,
   265→          imagen: p.categoria.imagen || null,
   266→          imagen_integral: p.categoria.imagen_integral || null,
   267→          imagen_sin_gluten: p.categoria.imagen_sin_gluten || null,
   268→          descripcion: p.categoria.descripcion || DESCRIPCIONES_DEFAULT[catNombre] || `Productos de ${catNombre.toLowerCase()}`,
   269→          descripcionDB: p.categoria.descripcion,
   270→          categoriaId: p.categoria.id,
   271→        })
   272→      }
   273→    }
   274→
   275→    const knownOrder = ['Sorrentinos', 'Ñoquis', 'Tallarines', 'Cintas Anchas', 'Ravioles', 'Tapas', 'Empanadas', 'Tartas', 'Pastas frescas', 'Pastas secas', 'Salsas', 'Lasagnas y canelones', 'Postres', 'Facturas', 'Chipás', 'Pizzas']
   276→
   277→    const sorted = Array.from(seen.values()).sort((a, b) => {
   278→      const seccionOrder = { pastas: 0, horneados: 1 }
   279→      const sa = seccionOrder[a.seccion as keyof typeof seccionOrder] ?? 2
   280→      const sb = seccionOrder[b.seccion as keyof typeof seccionOrder] ?? 2
   281→      if (sa !== sb) return sa - sb
   282→
   283→      const ia = knownOrder.indexOf(a.nombre)
   284→      const ib = knownOrder.indexOf(b.nombre)
   285→      if (ia !== -1 && ib !== -1) return ia - ib
   286→      if (ia !== -1) return -1
   287→      if (ib !== -1) return 1
   288→      return a.nombre.localeCompare(b.nombre)
   289→    })
   290→
   291→    for (const familia of sorted) {
   292→      const sec = familia.seccion === 'pastas' ? 'pastas' : familia.seccion === 'horneados' ? 'horneados' : 'other'
   293→      result[sec].push(familia)
   294→    }
   295→
   296→    return result
   297→  }, [productos])
   298→
   299→  // Current section families
   300→  const familias = useMemo(() => {
   301→    if (!seccionActiva) return []
   302→    return familiasBySeccion[seccionActiva] || []
   303→  }, [familiasBySeccion, seccionActiva])
   304→
   305→  // Check which sections have products for current filter
   306→  const seccionesConProductos = useMemo(() => {
   307→    const result: Seccion[] = []
   308→    for (const sec of SECCIONES) {
   309→      const families = familiasBySeccion[sec.key] || []
   310→      if (families.length > 0) result.push(sec.key)
   311→    }
   312→    return result
   313→  }, [familiasBySeccion])
   314→
   315→  // Compute product counts per family
   316→  const familiaData = useMemo(() => {
   317→    const data: Record<string, { count: number; hasProducts: boolean; hasPromos: boolean }> = {}
   318→    for (const familia of familias) {
   319→      const prods = productos.filter((p) => {
   320→        if (p.categoria.nombre !== familia.nombre) return false
   321→        const effSec = p.seccion ?? p.categoria.seccion
   322→        return effSec === familia.seccion
   323→      })
   324→      const hasPromos = prods.some((p) => !!productoPromociones[p.id])
   325→      data[familia.nombre] = {
   326→        count: prods.length,
   327→        hasProducts: prods.length > 0,
   328→        hasPromos,
   329→      }
   330→    }
   331→    return data
   332→  }, [productos, familias, productoPromociones])
   333→
   334→  // Products for the active family — filtered by both family name and effective section
   335→  const productosFamilia = useMemo(() => {
   336→    if (!familiaActiva || !seccionActiva) return []
   337→    const base = productos.filter((p) => {
   338→      if (p.categoria.nombre !== familiaActiva) return false
   339→      const effSec = p.seccion ?? p.categoria.seccion
   340→      return effSec === seccionActiva
   341→    })
   342→    // Apply "solo ofertas" filter
   343→    if (soloOfertas) {
   344→      return base.filter((p) => !!productoPromociones[p.id])
   345→    }
   346→    return base
   347→  }, [productos, familiaActiva, seccionActiva, soloOfertas, productoPromociones])
   348→
   349→  const productosVisibles = useMemo(() => {
   350→    return productosFamilia.slice(0, visibleCount)
   351→  }, [productosFamilia, visibleCount])
   352→
   353→  const hasMore = visibleCount < productosFamilia.length
   354→
   355→  const handleFamiliaClick = useCallback((nombre: string) => {
   356→    setFamiliaActiva((prev) => (prev === nombre ? null : nombre))
   357→  }, [])
   358→
   359→  // Smooth scroll al contenedor de productos cuando se selecciona una familia
   360→  useEffect(() => {
   361→    if (!familiaActiva || !productosGridRef.current) return
   362→    const yOffset = -100
   363→    const y = productosGridRef.current.getBoundingClientRect().top + window.pageYOffset + yOffset
   364→    if (y < window.pageYOffset || y > window.pageYOffset + window.innerHeight - 200) {
   365→      window.scrollTo({ top: y, behavior: 'smooth' })
   366→    }
   367→  }, [familiaActiva])
   368→
   369→  const handleLoadMore = useCallback(() => {
   370→    setVisibleCount((prev) => prev + PRODUCTS_PER_PAGE)
   371→  }, [])
   372→
   373→  // Filter out families with no products
   374→  const familiasVisibles = useMemo(
   375→    () => familias.filter((f) => familiaData[f.nombre]?.hasProducts),
   376→    [familias, familiaData]
   377→  )
   378→
   379→  // Get promos for the active section
   380→  const promosForSection = useMemo(() => {
   381→    if (!seccionActiva) return []
   382→    // Get product IDs in this section
   383→    const sectionProductIds = new Set(
   384→      productos
   385→        .filter((p) => {
   386→          const effSec = p.seccion ?? p.categoria.seccion
   387→          return effSec === seccionActiva
   388→        })
   389→        .map((p) => p.id)
   390→    )
   391→    // Filter promos that have products in this section
   392→    return promociones.filter((promo) => {
   393→      // Check if any product with this promo is in this section
   394→      return Object.entries(productoPromociones).some(
   395→        ([productId, pInfo]) =>
   396→          pInfo.promocionId === promo.id && sectionProductIds.has(Number(productId))
   397→      )
   398→    })
   399→  }, [seccionActiva, productos, promociones, productoPromociones])
   400→
   401→  // How many products have promos in the active family
   402→  const promoCountInFamily = useMemo(() => {
   403→    if (!familiaActiva) return 0
   404→    const base = productos.filter((p) => {
   405→      if (p.categoria.nombre !== familiaActiva) return false
   406→      const effSec = p.seccion ?? p.categoria.seccion
   407→      return effSec === seccionActiva
   408→    })
   409→    return base.filter((p) => !!productoPromociones[p.id]).length
   410→  }, [productos, familiaActiva, seccionActiva, productoPromociones])
   411→
   412→  // Retry: re-fetch all
   413→  const handleRetry = useCallback(async () => {
   414→    fetchedRef.current = false
   415→    setLoading(true)
   416→    setError('')
   417→    try {
   418→      const results = await Promise.all(
   419→        TODOS_FILTROS.map(async (tipo) => {
   420→          const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   421→          if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   422→          const data = await res.json()
   423→          return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   424→        })
   425→      )
   426→      setCache((prev) => {
   427→        const next = { ...prev }
   428→        for (const r of results) next[r.tipo] = r.productos
   429→        return next
   430→      })
   431→    } catch {
   432→      setError('Error de conexión. Verificá tu internet e intentá de nuevo.')
   433→    } finally {
   434→      setLoading(false)
   435→    }
   436→  }, [])
   437→
   438→  const seccionConfig = seccionActiva ? SECCIONES.find((s) => s.key === seccionActiva)! : null
   439→
   440→  return (
   441→    <section id="productos" className="min-h-screen flex flex-col justify-start py-12 sm:py-16 md:py-20 bg-crema">
   442→      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 w-full">
   443→        {/* Title */}
   444→        <div className="text-center mb-12">
   445→          <h2 className="text-3xl sm:text-4xl font-bold text-marron">
   446→            Nuestros <span className="text-rojo">Productos</span>
   447→          </h2>
   448→          <div className="h-1 w-20 bg-mostaza mx-auto mt-4 rounded-full" />
   449→          <p className="text-muted-foreground mt-3 max-w-lg mx-auto text-sm sm:text-base">
   450→            Elaborados con ingredientes frescos y de calidad. Elegí la sección que más te guste.
   451→          </p>
   452→        </div>
   453→
   454→        {/* ═══════════════════════════════════════════════════════════
   455→            STATE 1: No section selected → Show section hero cards
   456→            ═══════════════════════════════════════════════════════════ */}
   457→        {!seccionActiva && (
   458→          <>
   459→            {loading ? (
   460→              <div className="flex flex-col md:flex-row justify-center items-stretch gap-6 md:gap-10 w-full max-w-5xl mx-auto">
   461→                {Array.from({ length: 2 }).map((_, i) => (
   462→                  <div key={i} className="rounded-2xl border border-marron/10 bg-white animate-pulse flex-1 max-w-sm md:max-w-md lg:max-w-lg flex flex-col items-center p-6 sm:p-8 md:p-10 gap-5">
   463→                    <div className="w-40 h-40 md:w-48 md:h-48 lg:w-56 lg:h-56 bg-muted rounded-full" />
   464→                    <div className="h-7 w-32 bg-muted rounded" />
   465→                    <div className="h-10 w-36 bg-muted rounded-full" />
   466→                  </div>
   467→                ))}
   468→              </div>
   469→            ) : error ? (
   470→              <div className="text-center py-12">
   471→                <span className="text-5xl mb-4 block">⚠️</span>
   472→                <p className="text-muted-foreground mb-4">{error}</p>
   473→                <Button
   474→                  variant="outline"
   475→                  onClick={handleRetry}
