    82→function formatDate(ts: number): string {
    83→  const d = new Date(ts)
    84→  const pad = (n: number) => String(n).padStart(2, '0')
    85→  return `${pad(d.getDate())}/${pad(d.getMonth() + 1)}/${d.getFullYear()} ${pad(d.getHours())}:${pad(d.getMinutes())}`
    86→}
    87→
    88→// epoch ms -> 'YYYY-MM-DD' para input[type=date]
    89→function epochToDateInput(ts: number | null | undefined): string {
    90→  if (!ts) return ''
    91→  const d = new Date(ts)
    92→  if (isNaN(d.getTime())) return ''
    93→  const pad = (n: number) => String(n).padStart(2, '0')
    94→  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
    95→}
    96→
    97→// 'YYYY-MM-DD' -> epoch ms (a la medianoche local)
    98→function dateInputToEpoch(s: string): number | null {
    99→  if (!s) return null
   100→  const ts = new Date(s + 'T00:00:00').getTime()
   101→  return isNaN(ts) ? null : ts
   102→}

   104→function roleBadgeClass(role: CmRole): string {
   105→  if (role === 'superadmin') return 'border-[#B91C1C] text-[#B91C1C] bg-[#B91C1C]/5'
   106→  if (role === 'admin') return 'border-[#5C3A21] text-[#5C3A21] bg-[#5C3A21]/5'
   107→  return 'border-[#E1AD01] text-[#7a5c00] bg-[#E1AD01]/10'
   108→}
   109→
   110→function avatarColorClass(role: CmRole): string {
   111→  if (role === 'superadmin') return 'bg-[#B91C1C] text-[#FFF8E7]'
   112→  if (role === 'admin') return 'bg-[#5C3A21] text-[#FFF8E7]'
   113→  return 'bg-[#E1AD01] text-[#5C3A21]'
   114→}

   948→function PasswordChangeDialog({
   949→  user,
   950→  onClose,
   951→  onSaved,
   952→}: {
   953→  user: CmUserRecord | null
   954→  onClose: () => void
   955→  onSaved: () => void
   956→}) {
   957→  const [newPassword, setNewPassword] = React.useState('')
   958→  const [confirmPassword, setConfirmPassword] = React.useState('')
   959→  const [saving, setSaving] = React.useState(false)
   960→  const [error, setError] = React.useState<string | null>(null)
   961→  const [showNewPassword, setShowNewPassword] = React.useState(false)
   962→  const [showConfirmPassword, setShowConfirmPassword] = React.useState(false)
   963→
   964→  React.useEffect(() => {
   965→    if (user) {
   966→      setNewPassword('')
   967→      setConfirmPassword('')
   968→      setError(null)
   969→      setShowNewPassword(false)
   970→      setShowConfirmPassword(false)
   971→    }
   972→  }, [user])
   973→
   974→  if (!user) return null
   975→
   976→  const handleSubmit = async (e: React.FormEvent) => {
   977→    e.preventDefault()
   978→    setError(null)
   979→    if (newPassword.length < 6) return setError('Mínimo 6 caracteres')
   980→    if (newPassword !== confirmPassword) return setError('Las contraseñas no coinciden')
   981→
   982→    setSaving(true)
   983→    try {
   984→      const res = await fetch(`/api/cocina-movil/users/${user.id}/password`, {
   985→        method: 'PUT',
   986→        headers: { 'Content-Type': 'application/json' },
   987→        body: JSON.stringify({ newPassword }),
   988→      })
   989→      if (!res.ok) {
   990→        const data = await res.json().catch(() => ({}))
   991→        throw new Error(data.error || 'HTTP ' + res.status)
   992→      }
   993→      toast.success('Contraseña actualizada')
   994→      onSaved()
   995→    } catch (err) {
   996→      setError(err instanceof Error ? err.message : 'Error')
   997→    } finally {
   998→      setSaving(false)
   999→    }
  1000→  }
  1001→
  1002→  return (
  1003→    <Dialog open={!!user} onOpenChange={(o) => !o && onClose()}>
  1004→      <DialogContent className="sm:max-w-md">
  1005→        <DialogHeader>
  1006→          <DialogTitle className="text-[#5C3A21]">Cambiar contraseña</DialogTitle>
  1007→          <DialogDescription>
  1008→            Cambiando la contraseña de <strong>{getFullName(user)}</strong> ({user.email})
  1009→          </DialogDescription>
  1010→        </DialogHeader>
  1011→        <form onSubmit={handleSubmit} className="space-y-3">
  1012→          {error && (
  1013→            <div className="text-sm text-[#B91C1C] bg-[#B91C1C]/5 border border-[#B91C1C]/20 rounded-md px-3 py-2">
  1014→              {error}
  1015→            </div>
  1016→          )}
  1017→          <div className="space-y-1.5">
  1018→            <Label className="text-[#5C3A21]">Nueva contraseña</Label>
  1019→            <div className="relative">
  1020→              <Input
  1021→                type={showNewPassword ? 'text' : 'password'}
  1022→                value={newPassword}
  1023→                onChange={(e) => setNewPassword(e.target.value)}
  1024→                placeholder="••••••••"
  1025→                className="border-[#5C3A21]/15 pr-9"
  1026→                autoFocus
  1027→              />
  1028→              <button
