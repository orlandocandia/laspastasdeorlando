    88→    88→const IVA_RATE = 0.21
    89→    89→
    90→    90→export default function CompraForm({ compra, onSuccess, onCancel }: CompraFormProps) {
    91→    91→  const isEditing = !!compra
    92→    92→
    93→    93→  // Form fields
    94→    94→  const [idProveedor, setIdProveedor] = useState('')
    95→    95→  const [fechaCompra, setFechaCompra] = useState('')
    96→    96→  const [numeroFactura, setNumeroFactura] = useState('')
    97→    97→  const [idFormaPago, setIdFormaPago] = useState('')
    98→    98→  const [idEstado, setIdEstado] = useState('')
    99→    99→  const [observaciones, setObservaciones] = useState('')
   100→   100→
   101→   101→  // Detail rows
   102→   102→  const [detalles, setDetalles] = useState<DetalleRow[]>([])
   103→   103→
   104→   104→  // Data
   105→   105→  const [proveedores, setProveedores] = useState<Proveedor[]>([])
   106→   106→  const [formasPago, setFormasPago] = useState<FormaPago[]>([])
   107→   107→  const [estados, setEstados] = useState<EstadoGeneral[]>([])
   108→   108→  const [materiasPrimas, setMateriasPrimas] = useState<ProductoItem[]>([])
   109→   109→  const [insumos, setInsumos] = useState<ProductoItem[]>([])
   110→   110→  const [marcas, setMarcas] = useState<Marca[]>([])
   111→   111→  const [submitting, setSubmitting] = useState(false)
   112→   112→  const [loadingData, setLoadingData] = useState(true)
   113→   113→
   114→   114→  // Combobox open states
   115→   115→  const [proveedorOpen, setProveedorOpen] = useState(false)
   116→   116→
   117→   117→  // Load data
   118→   118→  useEffect(() => {
   119→   119→    async function loadData() {
   120→   120→      try {
   121→   121→        const [provRes, fpRes, estRes, mpRes, insRes, marRes] = await Promise.all([
   122→   122→          fetch('/api/personas?tipo=proveedor&limite=100'),
   123→   123→          fetch('/api/formas-pago'),
   124→   124→          fetch('/api/estados-generales?entidad_aplicable=compra'),
   125→   125→          fetch('/api/materias-primas?limite=200&estado=true'),
   126→   126→          fetch('/api/insumos?limite=200&estado=true'),
   127→   127→          fetch('/api/marcas'),
   128→   128→        ])
   129→   129→
   130→   130→        const provData = await provRes.json()
   131→   131→        const fpData = await fpRes.json()
   132→   132→        const estData = await estRes.json()
   133→   133→        const mpData = await mpRes.json()
   134→   134→        const insData = await insRes.json()
   135→   135→        const marData = await marRes.json()
   136→   136→
   137→   137→        setProveedores(provData.personas || [])
   138→   138→        setFormasPago(Array.isArray(fpData) ? fpData : [])
   139→   139→        setEstados(Array.isArray(estData) ? estData : [])
   140→   140→        setMateriasPrimas(
   141→   141→          (mpData.data || []).map((mp: any) => ({
   142→   142→            id: mp.id,
   143→   143→            nombre: mp.nombre,
   144→   144→            id_unidad_base: mp.id_unidad_base,
   145→   145→            unidadBase: mp.unidadBase,
   146→   146→            precio_compra_referencia: mp.precio_compra_referencia,
   147→   147→            tipo: 'mp' as const,
   148→   148→          }))
   149→   149→        )
   150→   150→        setInsumos(
   151→   151→          (insData.data || []).map((ins: any) => ({
   152→   152→            id: ins.id,
   153→   153→            nombre: ins.nombre,
   154→   154→            id_unidad_base: ins.id_unidad_base,
   155→   155→            unidadBase: ins.unidadBase,
   156→   156→            precio_compra_referencia: ins.precio_compra_referencia,
   157→   157→            tipo: 'insumo' as const,
   158→   158→          }))
   159→   159→        )
   160→   160→        setMarcas(Array.isArray(marData) ? marData : [])
   161→   161→      } catch {
   162→   162→        toast.error('Error al cargar datos del formulario')
   163→   163→      } finally {
   164→   164→        setLoadingData(false)
   165→   165→      }
   166→   166→    }
   167→   167→    loadData()
   168→   168→  }, [])
   169→   169→
   170→   170→  // Populate form when editing
   171→   171→  useEffect(() => {
   172→   172→    if (compra) {
   173→   173→      setIdProveedor(compra.id_proveedor?.toString() || '')
   174→   174→      setFechaCompra(compra.fecha_compra ? compra.fecha_compra.split('T')[0] : '')
   175→   175→      setNumeroFactura(compra.numero_factura || '')
   176→   176→      setIdFormaPago(compra.id_forma_pago?.toString() || '')
   177→   177→      setIdEstado(compra.id_estado?.toString() || '')
   178→   178→      setObservaciones(compra.observaciones || '')
   179→   179→
   180→   180→      if (compra.detalle && Array.isArray(compra.detalle)) {
   181→   181→        setDetalles(
   182→   182→          compra.detalle.map((d: any) => ({
   183→   183→            key: `row-${d.id}`,
   184→   184→            tipo: d.id_materia_prima ? 'mp' : 'insumo',
   185→   185→            id_producto: (d.id_materia_prima || d.id_insumo)?.toString() || '',
   186→   186→            id_marca: d.id_marca?.toString() || '',
   187→   187→            cantidad: d.cantidad_comprada?.toString() || '',
   188→   188→            unidad: d.unidadCompra?.id?.toString() || '',
   189→   189→            unidadCodigo: d.unidadCompra?.codigo || '',
   190→   190→            precioUnitario: d.precio_unitario?.toString() || '',
   191→   191→            precioTotal: d.precio_total?.toString() || '',
   192→   192→            fechaVencimiento: d.fecha_vencimiento ? d.fecha_vencimiento.split('T')[0] : '',
   193→   193→            lote: d.lote || '',
   194→   194→          }))
   195→   195→        )
   196→   196→      }
   197→   197→    } else {
   198→   198→      // Defaults for new
   199→   199→      setFechaCompra(new Date().toISOString().split('T')[0])
   200→   200→      addDetailRow()
   201→   201→    }
   202→   202→  }, [compra])
   203→   203→
   204→   204→  // Products filtered by selected tipo for each row
   205→   205→  const getProductsForTipo = (tipo: 'mp' | 'insumo') => {
   206→   206→    return tipo === 'mp' ? materiasPrimas : insumos
   207→   207→  }
   208→   208→
   209→   209→  // Find product by id
   210→   210→  const findProduct = (tipo: 'mp' | 'insumo', id: string) => {
   211→   211→    const products = getProductsForTipo(tipo)
   212→   212→    return products.find((p) => p.id.toString() === id)
   213→   213→  }
   214→   214→
   215→   215→  // Add detail row
   216→   216→  const addDetailRow = () => {
   217→   217→    setDetalles((prev) => [
   218→   218→      ...prev,
   219→   219→      {
   220→   220→        key: `row-${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
   221→   221→        tipo: 'mp',
   222→   222→        id_producto: '',
   223→   223→        id_marca: '',
   224→   224→        cantidad: '',
   225→   225→        unidad: '',
   226→   226→        unidadCodigo: '',
   227→   227→        precioUnitario: '',
   228→   228→        precioTotal: '',
   229→   229→        fechaVencimiento: '',
   230→   230→        lote: '',
   231→   231→      },
   232→   232→    ])
   233→   233→  }
   234→   234→
   235→   235→  // Remove detail row
   236→   236→  const removeDetailRow = (key: string) => {
   237→   237→    setDetalles((prev) => prev.filter((d) => d.key !== key))
   238→   238→  }
   239→   239→
   240→   240→  // Update detail row
   241→   241→  const updateDetalle = (key: string, field: keyof DetalleRow, value: string) => {
   242→   242→    setDetalles((prev) =>
   243→   243→      prev.map((d) => {
   244→   244→        if (d.key !== key) return d
   245→   245→
   246→   246→        const updated = { ...d, [field]: value }
   247→   247→
   248→   248→        // When tipo changes, reset producto and unidad
   249→   249→        if (field === 'tipo') {
   250→   250→          updated.id_producto = ''
   251→   251→          updated.unidad = ''
   252→   252→          updated.unidadCodigo = ''
   253→   253→          updated.precioUnitario = ''
   254→   254→          updated.precioTotal = ''
   255→   255→        }
   256→   256→
   257→   257→        // When product changes, auto-fill unidad and precio_referencia
   258→   258→        if (field === 'id_producto' && value) {
   259→   259→          const product = findProduct(updated.tipo, value)
   260→   260→          if (product) {
   261→   261→            updated.unidad = product.unidadBase?.id?.toString() || ''
   262→   262→            updated.unidadCodigo = product.unidadBase?.codigo || ''
   263→   263→            updated.precioUnitario = product.precio_compra_referencia?.toString() || ''
   264→   264→          }
   265→   265→        }
   266→   266→
   267→   267→        // Auto-calculate precio_total when cantidad or precio_unitario changes
   268→   268→        if (field === 'cantidad' || field === 'precioUnitario') {
   269→   269→          const cant = parseFloat(field === 'cantidad' ? value : updated.cantidad)
   270→   270→          const precio = parseFloat(field === 'precioUnitario' ? value : updated.precioUnitario)
   271→   271→          if (!isNaN(cant) && !isNaN(precio)) {
   272→   272→            updated.precioTotal = (cant * precio).toFixed(2)
   273→   273→          } else {
   274→   274→            updated.precioTotal = ''
   275→   275→          }
   276→   276→        }
   277→   277→
   278→   278→        return updated
   279→   279→      })
   280→   280→    )
   281→   281→  }
   282→   282→
   283→   283→  // Calculate totals
   284→   284→  const { subtotal, iva, total } = useMemo(() => {
   285→   285→    const sub = detalles.reduce((sum, d) => {
   286→   286→      const pt = parseFloat(d.precioTotal)
   287→   287→      return sum + (isNaN(pt) ? 0 : pt)
   288→   288→    }, 0)
   289→   289→    const ivaCalc = sub * IVA_RATE
   290→   290→    return {
   291→   291→      subtotal: sub,
   292→   292→      iva: ivaCalc,
   293→   293→      total: sub + ivaCalc,
   294→   294→    }
   295→   295→  }, [detalles])
   296→   296→
   297→   297→  const formatCurrency = (value: number) => {
   298→   298→    return new Intl.NumberFormat('es-AR', { style: 'currency', currency: 'ARS' }).format(value)
   299→   299→  }
   300→   300→
   301→   301→  // Submit
   302→   302→  const handleSubmit = async () => {
   303→   303→    // Validation
   304→   304→    if (!isEditing) {
   305→   305→      if (!idProveedor) {
   306→   306→        toast.error('Seleccioná un proveedor')
   307→   307→        return
   308→   308→      }
   309→   309→      if (!fechaCompra) {
   310→   310→        toast.error('Ingresá la fecha de compra')
   311→   311→        return
   312→   312→      }
   313→   313→      if (!idFormaPago) {
   314→   314→        toast.error('Seleccioná una forma de pago')
   315→   315→        return
   316→   316→      }
   317→   317→      if (detalles.length === 0) {
   318→   318→        toast.error('Agregá al menos un detalle de compra')
   319→   319→        return
   320→   320→      }
   321→   321→      for (let i = 0; i < detalles.length; i++) {
   322→   322→        const d = detalles[i]
   323→   323→        if (!d.id_producto) {
   324→   324→          toast.error(`Fila ${i + 1}: Seleccioná un producto`)
   325→   325→          return
   326→   326→        }
   327→   327→        if (!d.cantidad || parseFloat(d.cantidad) <= 0) {
   328→   328→          toast.error(`Fila ${i + 1}: Ingresá una cantidad válida`)
   329→   329→          return
   330→   330→        }
   331→   331→        if (!d.precioUnitario || parseFloat(d.precioUnitario) <= 0) {
   332→   332→          toast.error(`Fila ${i + 1}: Ingresá un precio unitario válido`)
   333→   333→          return
   334→   334→        }
   335→   335→      }
   336→   336→    }
   337→   337→
   338→   338→    setSubmitting(true)
   339→   339→    try {
   340→   340→      const payload = isEditing
   341→   341→        ? {
   342→   342→            id: compra.id,
   343→   343→            numero_factura: numeroFactura.trim() || null,
   344→   344→            observaciones: observaciones.trim() || null,
   345→   345→            id_estado: idEstado ? parseInt(idEstado) : undefined,
   346→   346→          }
   347→   347→        : {
   348→   348→            id_proveedor: parseInt(idProveedor),
   349→   349→            fecha_compra: fechaCompra,
   350→   350→            numero_factura: numeroFactura.trim() || null,
   351→   351→            id_forma_pago: parseInt(idFormaPago),
   352→   352→            observaciones: observaciones.trim() || null,
   353→   353→            subtotal,
   354→   354→            iva,
   355→   355→            total,
   356→   356→            detalles: detalles.map((d) => ({
   357→   357→              tipo: d.tipo === 'mp' ? 'materia_prima' : 'insumo',
   358→   358→              id_materia_prima: d.tipo === 'mp' ? parseInt(d.id_producto) : null,
   359→   359→              id_insumo: d.tipo === 'insumo' ? parseInt(d.id_producto) : null,
   360→   360→              id_marca: d.id_marca ? parseInt(d.id_marca) : null,
   361→   361→              cantidad_comprada: parseFloat(d.cantidad),
   362→   362→              id_unidad_compra: parseInt(d.unidad),
   363→   363→              precio_unitario: parseFloat(d.precioUnitario),
   364→   364→              fecha_vencimiento: d.fechaVencimiento || null,
   365→   365→              lote: d.lote.trim() || null,
   366→   366→            })),
   367→   367→          }
   368→   368→
   369→   369→      const url = isEditing ? `/api/compras/${compra.id}` : '/api/compras'
   370→   370→      const method = isEditing ? 'PUT' : 'POST'
   371→   371→
   372→   372→      const res = await fetch(url, {
   373→   373→        method,
   374→   374→        headers: { 'Content-Type': 'application/json' },
   375→   375→        body: JSON.stringify(payload),
   376→   376→      })
   377→   377→
   378→   378→      if (!res.ok) {
   379→   379→        const errorData = await res.json().catch(() => ({}))
   380→   380→        throw new Error(errorData.error || 'Error al guardar compra')
   381→   381→      }
   382→   382→
   383→   383→      toast.success(isEditing ? 'Compra actualizada' : 'Compra registrada', {
   384→   384→        description: isEditing
   385→   385→          ? 'Los cambios se guardaron correctamente'
   386→   386→          : 'La nueva compra se registró exitosamente',
   387→   387→      })
   388→   388→      onSuccess()
   389→   389→    } catch (error: any) {
   390→   390→      toast.error('Error al guardar', {
   391→   391→        description: error.message || 'Intentá de nuevo más tarde',
   392→   392→      })
   393→   393→    } finally {
   394→   394→      setSubmitting(false)
   395→   395→    }
   396→   396→  }
   397→   397→
   398→   398→  if (loadingData) {
   399→   399→    return (
   400→   400→      <div className="flex items-center justify-center py-8">
   401→   401→        <Loader2 className="h-6 w-6 animate-spin text-mostaza" />
   402→   402→        <span className="ml-2 text-muted-foreground">Cargando datos...</span>
   403→   403→      </div>
   404→   404→    )
   405→   405→  }
   406→   406→
   407→   407→  return (
   408→   408→    <div className="space-y-6">
   409→   409→      {/* Header info */}
   410→   410→      {isEditing && (
   411→   411→        <div className="bg-muted/30 rounded-lg p-3 border border-marron/10">
   412→   412→          <p className="text-sm text-muted-foreground">
   413→   413→            Compra #{compra.id} — Proveedor: {compra.proveedor?.razon_social || `${compra.proveedor?.nombre} ${compra.proveedor?.apellido}`}
   414→   414→          </p>
   415→   415→        </div>
   416→   416→      )}
   417→   417→
   418→   418→      {/* Main fields */}
   419→   419→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   420→   420→        {/* Proveedor - searchable combobox (only in create mode) */}
   421→   421→        <div>
   422→   422→          <Label className="text-sm font-medium text-marron mb-1 block">Proveedor *</Label>
   423→   423→          {isEditing ? (
   424→   424→            <Input
   425→   425→              value={compra.proveedor?.razon_social || `${compra.proveedor?.nombre} ${compra.proveedor?.apellido}`}
   426→   426→              disabled
   427→   427→              className="bg-muted/50"
   428→   428→            />
   429→   429→          ) : (
   430→   430→            <Popover open={proveedorOpen} onOpenChange={setProveedorOpen}>
   431→   431→              <PopoverTrigger asChild>
   432→   432→                <Button
   433→   433→                  variant="outline"
   434→   434→                  role="combobox"
   435→   435→                  aria-expanded={proveedorOpen}
   436→   436→                  className={cn(
   437→   437→                    'w-full justify-between',
   438→   438→                    !idProveedor && 'text-muted-foreground'
   439→   439→                  )}
   440→   440→                >
   441→   441→                  {idProveedor
   442→   442→                    ? (() => {
   443→   443→                        const prov = proveedores.find((p) => p.id.toString() === idProveedor)
   444→   444→                        return prov?.razon_social || `${prov?.nombre} ${prov?.apellido}` || 'Seleccionar...'
   445→   445→                      })()
   446→   446→                    : 'Buscar proveedor...'}
   447→   447→                  <ChevronsUpDown className="ml-2 h-4 w-4 shrink-0 opacity-50" />
   448→   448→                </Button>
   449→   449→              </PopoverTrigger>
   450→   450→              <PopoverContent className="w-full p-0" align="start">
   451→   451→                <Command>
   452→   452→                  <CommandInput placeholder="Buscar proveedor..." />
   453→   453→                  <CommandList>
   454→   454→                    <CommandEmpty>No se encontró proveedor</CommandEmpty>
   455→   455→                    <CommandGroup>
   456→   456→                      {proveedores.map((prov) => (
   457→   457→                        <CommandItem
   458→   458→                          key={prov.id}
   459→   459→                          value={prov.razon_social || `${prov.nombre} ${prov.apellido}`}
   460→   460→                          onSelect={() => {
   461→   461→                            setIdProveedor(prov.id.toString())
   462→   462→                            setProveedorOpen(false)
   463→   463→                          }}
