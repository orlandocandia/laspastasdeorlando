   100→// Imagen dinámica según filtro para familias con variante integral/sin gluten
   101→function getFamiliaImagen(familia: Familia, filtro: FiltroHarina): string {
   102→  if (filtro === 'integral' && familia.imagen_integral) return familia.imagen_integral
   103→  if (filtro === 'sin_gluten' && familia.imagen_sin_gluten) return familia.imagen_sin_gluten
   104→
   105→  if (familia.nombre === 'Tallarines' && filtro === 'integral') {
   106→    return '/images/familias/tallarinesintegrales.png'
   107→  }
   108→  if (familia.nombre === 'Cintas Anchas' && filtro === 'integral') {
   109→    return '/images/familias/cintasanchasintegrales.png'
   110→  }
   111→  if (familia.nombre === 'Tapas' && filtro === 'integral') {
   112→    return '/images/familias/tapasintegrales.png'
   113→  }
   114→  if (familia.nombre === 'Tapas' && filtro === 'sin_gluten') {
   115→    return '/images/familias/tapassingluten.png'
   116→  }
   117→
   118→  if (familia.imagen) return familia.imagen
   119→  return IMAGENES_DEFAULT[familia.nombre] || '/images/placeholder-producto.jpg'
   120→}

   122→const FILTROS: { key: FiltroHarina; label: string; icon: React.ReactNode }[] = [
   123→  { key: 'con_gluten', label: 'CON GLUTEN', icon: <Wheat className="h-3.5 w-3.5" /> },

   126→]

   128→const TODOS_FILTROS: FiltroHarina[] = ['con_gluten', 'integral', 'sin_gluten']
   129→
   130→const PRODUCTS_PER_PAGE = 50
   131→
   132→interface ProductosProps {
   133→  filtroActivo?: FiltroHarina
   134→  onFiltroChange?: (filtro: FiltroHarina) => void

   136→
   137→export default function Productos({ filtroActivo = 'con_gluten', onFiltroChange }: ProductosProps) {
   138→  // Cache: productos pre-fetcheados por tipo de harina
   139→  const [cache, setCache] = useState<Record<FiltroHarina, ProductoPublico[]>>({
   140→    con_gluten: [],
   141→    integral: [],
   142→    sin_gluten: [],
   143→  })
   144→  const [loading, setLoading] = useState(true)
   145→  const [error, setError] = useState('')
   146→  const [filtro, setFiltro] = useState<FiltroHarina>(filtroActivo)
   147→  // null = no section selected (step 1), 'pastas'|'horneados' = section selected (step 2)
   148→  const [seccionActiva, setSeccionActiva] = useState<Seccion | null>(null)
   149→  const [familiaActiva, setFamiliaActiva] = useState<string | null>(null)
   150→  const [visibleCount, setVisibleCount] = useState(PRODUCTS_PER_PAGE)
   151→  const fetchedRef = useRef(false)
   152→  const productosGridRef = useRef<HTMLDivElement>(null)
   153→
   154→  // Pre-fetch all 3 filter types in parallel on mount — no refetch on filter change
   155→  useEffect(() => {
   156→    if (fetchedRef.current) return
   157→    fetchedRef.current = true
   158→
   159→    async function fetchAll() {
   160→      setLoading(true)
   161→      setError('')
   162→      try {
   163→        const results = await Promise.all(
   164→          TODOS_FILTROS.map(async (tipo) => {
   165→            const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   166→            if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   167→            const data = await res.json()
   168→            return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   169→          })
   170→        )
   171→        setCache((prev) => {
   172→          const next = { ...prev }
   173→          for (const r of results) next[r.tipo] = r.productos
   174→          return next
   175→        })
   176→      } catch {
   177→        setError('No se pudieron cargar los productos. Intentá más tarde.')
   178→      } finally {
   179→        setLoading(false)
   180→      }
   181→    }
   182→    fetchAll()
   183→  }, [])
   184→
   185→  // Reset visible count when family or section changes
   186→  useEffect(() => {
   187→    setVisibleCount(PRODUCTS_PER_PAGE)
   188→  }, [familiaActiva, seccionActiva])
   189→
   190→  // Sync with parent filtro state — instant, no refetch
   191→  useEffect(() => {
   192→    if (filtroActivo !== filtro) {
   193→      setFiltro(filtroActivo ?? 'con_gluten')
   194→      setFamiliaActiva(null)
   195→    }
   196→  }, [filtroActivo, filtro])
   197→
   198→  const handleFiltroChange = useCallback((nuevoFiltro: FiltroHarina) => {
   199→    setFiltro(nuevoFiltro)
   200→    setFamiliaActiva(null)
   201→    onFiltroChange?.(nuevoFiltro)
   202→  }, [onFiltroChange])
   203→
   204→  const handleSeccionChange = useCallback((nuevaSeccion: Seccion) => {
   205→    setSeccionActiva(nuevaSeccion)
   206→    setFamiliaActiva(null)
   207→    setFiltro('con_gluten') // Reset flour filter when changing section
   208→    onFiltroChange?.('con_gluten')
   209→  }, [onFiltroChange])
   210→
   211→  const handleVolverASecciones = useCallback(() => {
   212→    setSeccionActiva(null)
   213→    setFamiliaActiva(null)
   214→    setFiltro('con_gluten')
   215→    onFiltroChange?.('con_gluten')
   216→  }, [onFiltroChange])
   217→
   218→  // Current products from cache — instant swap, no network
   219→  const productos = cache[filtro]
   220→
   221→  // Build families dynamically from product categories, grouped by seccion
   222→  const familiasBySeccion = useMemo(() => {
   223→    const result: Record<string, Familia[]> = { pastas: [], horneados: [], other: [] }
   224→    const seen = new Map<string, Familia>()
   225→
   226→    for (const p of productos) {
   227→      const catNombre = p.categoria.nombre
   228→      if (!seen.has(catNombre)) {
   229→        seen.set(catNombre, {
   230→          nombre: catNombre,
   231→          seccion: p.categoria.seccion,
   232→          imagen: p.categoria.imagen || null,
   233→          imagen_integral: p.categoria.imagen_integral || null,
   234→          imagen_sin_gluten: p.categoria.imagen_sin_gluten || null,
   235→          descripcion: p.categoria.descripcion || DESCRIPCIONES_DEFAULT[catNombre] || `Productos de ${catNombre.toLowerCase()}`,
   236→          descripcionDB: p.categoria.descripcion,
   237→          categoriaId: p.categoria.id,
   238→        })
   239→      }
   240→    }
   241→
   242→    const knownOrder = ['Sorrentinos', 'Ñoquis', 'Tallarines', 'Cintas Anchas', 'Ravioles', 'Tapas', 'Empanadas', 'Tartas', 'Pastas frescas', 'Pastas secas', 'Salsas', 'Lasagnas y canelones', 'Postres', 'Facturas', 'Chipás', 'Pizzas']
   243→
   244→    const sorted = Array.from(seen.values()).sort((a, b) => {
   245→      const seccionOrder = { pastas: 0, horneados: 1 }
   246→      const sa = seccionOrder[a.seccion as keyof typeof seccionOrder] ?? 2
   247→      const sb = seccionOrder[b.seccion as keyof typeof seccionOrder] ?? 2
   248→      if (sa !== sb) return sa - sb
   249→
   250→      const ia = knownOrder.indexOf(a.nombre)
   251→      const ib = knownOrder.indexOf(b.nombre)
   252→      if (ia !== -1 && ib !== -1) return ia - ib
   253→      if (ia !== -1) return -1
   254→      if (ib !== -1) return 1
   255→      return a.nombre.localeCompare(b.nombre)
   256→    })
   257→
   258→    for (const familia of sorted) {
   259→      const sec = familia.seccion === 'pastas' ? 'pastas' : familia.seccion === 'horneados' ? 'horneados' : 'other'
   260→      result[sec].push(familia)
   261→    }
   262→
   263→    return result
   264→  }, [productos])
   265→
   266→  // Current section families
   267→  const familias = useMemo(() => {
   268→    if (!seccionActiva) return []
   269→    const secFamilies = familiasBySeccion[seccionActiva] || []
   270→    const otherFamilies = familiasBySeccion.other || []
   271→    return [...secFamilies, ...otherFamilies]
   272→  }, [familiasBySeccion, seccionActiva])
   273→
   274→  // Check which sections have products for current filter
   275→  const seccionesConProductos = useMemo(() => {
   276→    const result: Seccion[] = []
   277→    for (const sec of SECCIONES) {
   278→      const families = familiasBySeccion[sec.key] || []
   279→      if (families.length > 0) result.push(sec.key)
   280→    }
   281→    return result
   282→  }, [familiasBySeccion])
   283→
   284→  // Compute product counts per family
   285→  const familiaData = useMemo(() => {
   286→    const data: Record<string, { count: number; hasProducts: boolean }> = {}
   287→    for (const familia of familias) {
   288→      const prods = productos.filter(
   289→        (p) => p.categoria.nombre === familia.nombre
   290→      )
   291→      data[familia.nombre] = {
   292→        count: prods.length,
   293→        hasProducts: prods.length > 0,
   294→      }
   295→    }
   296→    return data
   297→  }, [productos, familias])
   298→
   299→  // Products for the active family — paginated
   300→  const productosFamilia = useMemo(() => {
   301→    if (!familiaActiva) return []
   302→    return productos.filter((p) => p.categoria.nombre === familiaActiva)
   303→  }, [productos, familiaActiva])
   304→
   305→  const productosVisibles = useMemo(() => {
   306→    return productosFamilia.slice(0, visibleCount)
   307→  }, [productosFamilia, visibleCount])
   308→
   309→  const hasMore = visibleCount < productosFamilia.length
   310→
   311→  const handleFamiliaClick = useCallback((nombre: string) => {
   312→    setFamiliaActiva((prev) => (prev === nombre ? null : nombre))
   313→  }, [])
   314→
   315→  // Smooth scroll al contenedor de productos cuando se selecciona una familia
   316→  useEffect(() => {
   317→    if (!familiaActiva || !productosGridRef.current) return
   318→    const yOffset = -100
   319→    const y = productosGridRef.current.getBoundingClientRect().top + window.pageYOffset + yOffset
   320→    if (y < window.pageYOffset || y > window.pageYOffset + window.innerHeight - 200) {
   321→      window.scrollTo({ top: y, behavior: 'smooth' })
   322→    }
   323→  }, [familiaActiva])
   324→
   325→  const handleLoadMore = useCallback(() => {
   326→    setVisibleCount((prev) => prev + PRODUCTS_PER_PAGE)
   327→  }, [])
   328→
   329→  // Filter out families with no products
   330→  const familiasVisibles = useMemo(
   331→    () => familias.filter((f) => familiaData[f.nombre]?.hasProducts),
   332→    [familias, familiaData]
   333→  )
   334→
   335→  // Retry: re-fetch all
   336→  const handleRetry = useCallback(async () => {
   337→    fetchedRef.current = false
   338→    setLoading(true)
   339→    setError('')
   340→    try {
   341→      const results = await Promise.all(
   342→        TODOS_FILTROS.map(async (tipo) => {
   343→          const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   344→          if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   345→          const data = await res.json()
   346→          return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   347→        })
   348→      )
   349→      setCache((prev) => {
   350→        const next = { ...prev }
   351→        for (const r of results) next[r.tipo] = r.productos
   352→        return next
   353→      })
   354→    } catch {
   355→      setError('Error de conexión. Verificá tu internet e intentá de nuevo.')
   356→    } finally {
   357→      setLoading(false)
   358→    }
   359→  }, [])
   360→
   361→  const seccionConfig = seccionActiva ? SECCIONES.find((s) => s.key === seccionActiva)! : null
   362→
   363→  return (
   364→    <section id="productos" className="min-h-screen flex flex-col justify-center py-12 sm:py-16 md:py-20 bg-crema">
   365→      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 w-full">
   366→        {/* Title */}
   367→        <div className="text-center mb-8">
   368→          <h2 className="text-3xl sm:text-4xl font-bold text-marron">
   369→            Nuestros <span className="text-rojo">Productos</span>
   370→          </h2>
   371→          <div className="h-1 w-20 bg-mostaza mx-auto mt-4 rounded-full" />
   372→          <p className="text-muted-foreground mt-3 max-w-lg mx-auto text-sm sm:text-base">
   373→            Elaborados con ingredientes frescos y de calidad. Elegí la sección que más te guste.
   374→          </p>
   375→        </div>
   376→
   377→        {/* ═══════════════════════════════════════════════════════════
   378→            STATE 1: No section selected → Show section hero cards
   379→            ═══════════════════════════════════════════════════════════ */}
   380→        {!seccionActiva && (
   381→          <>
   382→            {loading ? (
   383→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-6 sm:gap-8 max-w-4xl mx-auto">
   384→                {Array.from({ length: 2 }).map((_, i) => (
   385→                  <div key={i} className="rounded-2xl border border-marron/10 bg-white overflow-hidden animate-pulse flex flex-col min-h-[420px]">
   386→                    <div className="w-full aspect-square bg-muted" />
   387→                    <div className="p-6 flex flex-col items-center gap-3">
   388→                      <div className="h-7 w-32 bg-muted rounded" />
   389→                      <div className="h-4 w-48 bg-muted rounded" />
   390→                      <div className="h-3 w-40 bg-muted rounded" />
   391→                    </div>
   392→                  </div>
   393→                ))}
   394→              </div>
   395→            ) : error ? (
   396→              <div className="text-center py-12">
   397→                <span className="text-5xl mb-4 block">⚠️</span>
   398→                <p className="text-muted-foreground mb-4">{error}</p>
   399→                <Button
   400→                  variant="outline"
   401→                  onClick={handleRetry}
   402→                  className="border-mostaza text-marron hover:bg-mostaza hover:text-marron"
   403→                >
   404→                  Reintentar
   405→                </Button>
   406→              </div>
   407→            ) : (
   408→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-6 sm:gap-8 max-w-4xl mx-auto">
   409→                {SECCIONES.map((sec) => {
   410→                  const hasProducts = seccionesConProductos.includes(sec.key)
   411→                  return (
   412→                    <button
   413→                      key={sec.key}
   414→                      onClick={() => handleSeccionChange(sec.key)}
   415→                      className={`
   416→                        group relative rounded-2xl border-2 bg-white overflow-hidden
   417→                        flex flex-col text-center
   418→                        transition-all duration-300 cursor-pointer
   419→                        border-marron/10 hover:border-mostaza hover:shadow-2xl hover:scale-[1.02]
   420→                        shadow-md
   421→                      `}
   422→                    >
   423→                      {/* Product image — full width, rounded top */}
   424→                      <div className="relative w-full aspect-square overflow-hidden">
   425→                        <Image
   426→                          src={sec.imagen}
   427→                          alt={sec.label}
   428→                          width={400}
   429→                          height={400}
   430→                          className="object-cover w-full h-full transition-transform duration-500 group-hover:scale-105"
   431→                        />
   432→                        {/* Subtle gradient overlay at bottom for text readability */}
   433→                        <div className="absolute inset-x-0 bottom-0 h-24 bg-gradient-to-t from-black/30 to-transparent" />
   434→                      </div>
   435→
   436→                      {/* Card content */}
   437→                      <div className="p-6 sm:p-8 flex flex-col items-center">
   438→                        {/* Label */}
   439→                        <h3 className="text-2xl sm:text-3xl font-bold text-marron mb-2">
   440→                          {sec.label}
   441→                        </h3>
   442→
   443→                        {/* Description */}
   444→                        <p className="text-sm sm:text-base text-muted-foreground mb-2 max-w-xs">
   445→                          {sec.description}
   446→                        </p>
   447→
   448→                        {/* Subtext */}
   449→                        <p className="text-xs text-muted-foreground/70 mb-4">
   450→                          {sec.subtext}
   451→                        </p>
   452→
   453→                        {/* CTA arrow */}
   454→                        <div className="mt-auto flex items-center gap-1.5 text-mostaza font-semibold text-sm group-hover:gap-2.5 transition-all">
   455→                          Ver productos
   456→                          <ChevronRight className="h-4 w-4" />
   457→                        </div>
   458→                      </div>
   459→
   460→                      {/* "Próximamente" badge for sections without products */}
   461→                      {!hasProducts && (
   462→                        <div className="absolute top-4 right-4 bg-mostaza/90 text-marron text-xs rounded-full px-3 py-1 font-bold shadow-sm">
   463→                          Próximamente
   464→                        </div>
   465→                      )}
   466→                    </button>
   467→                  )
   468→                })}
   469→              </div>
   470→            )}
   471→          </>
   472→        )}
   473→
   474→        {/* ═══════════════════════════════════════════════════════════
   475→            STATE 2: Section selected → Show filters + products
   476→            ═══════════════════════════════════════════════════════════ */}
   477→        {seccionActiva && seccionConfig && (
   478→          <>
   479→            {/* Breadcrumb: Volver a secciones + Section title */}
   480→            <div className="flex items-center gap-3 mb-6">
   481→              <Button
   482→                variant="ghost"
   483→                size="sm"
   484→                onClick={handleVolverASecciones}
   485→                className="text-marron hover:bg-mostaza/20 gap-1.5 -ml-2"
   486→              >
   487→                <ArrowLeft className="h-4 w-4" />
   488→                Volver
