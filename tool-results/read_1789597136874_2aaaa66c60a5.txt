    79→// Convierte un valor string de input number a number | null
    80→// (vacío => null, NaN => null)
    81→function toNumberOrNull(value: string): number | null {
    82→  if (!value.trim()) return null
    83→  const n = Number(value)
    84→  return isNaN(n) ? null : n
    85→}

   172→function CmLugaresPageContent() {

   178→  const [search, setSearch] = React.useState('')
   179→  const [statusFilter, setStatusFilter] = React.useState<'all' | 'active' | 'inactive'>('all')
   180→
   181→  // Modales

   183→  const [formMode, setFormMode] = React.useState<FormMode>('create')

   186→
   187→  // Abrir modal de creación si viene ?action=new
   188→  const openCreateForm = React.useCallback(() => {
   189→    setFormMode('create')
   190→    setEditingPlace(null)

   193→
   194→  React.useEffect(() => {

   202→    try {
   203→      const params = new URLSearchParams()
   204→      if (search) params.set('search', search)
   205→      if (statusFilter === 'active') params.set('isActive', 'true')
   206→      if (statusFilter === 'inactive') params.set('isActive', 'false')

   211→      setPlaces(data.places || [])
   212→      setTotal(data.total || 0)

   221→  React.useEffect(() => {

   225→
   226→  // ----- Handlers -----

   233→  const handleToggleActive = async (p: CmPlaceRecord) => {

   236→        method: 'PUT',
   237→        headers: { 'Content-Type': 'application/json' },
   238→        body: JSON.stringify({ isActive: !p.isActive }),

   241→      toast.success(`Lugar ${p.isActive ? 'desactivado' : 'activado'}`)
   242→      loadPlaces()
   243→    } catch (err) {
   244→      toast.error('Error al cambiar estado')
   245→      console.error(err)
   246→    }
   247→  }

   250→    if (!deletePlace) return
   251→    try {
   252→      const res = await fetch(`/api/cocina-movil/places/${deletePlace.id}`, { method: 'DELETE' })

   254→        const data = await res.json().catch(() => ({}))
   255→        throw new Error(data.error || 'HTTP ' + res.status)
   256→      }
   257→      toast.success('Lugar eliminado')
   258→      setDeletePlace(null)
   259→      loadPlaces()
   260→    } catch (err) {
   261→      toast.error(err instanceof Error ? err.message : 'Error al eliminar')
   262→    }
   263→  }
   264→

   268→    if (statusFilter === 'active') params.set('isActive', 'true')
   269→    if (statusFilter === 'inactive') params.set('isActive', 'false')
   270→    window.open(`/api/cocina-movil/places/export?${params.toString()}`, '_blank')
   271→  }
   272→
   273→  const handlePrint = () => {
   274→    window.print()
   275→  }
   276→
   277→  // ----- Render -----

   280→      {/* Header */}

   284→            <MapPin className="h-6 w-6" />
   285→            Lugares
   286→          </h1>
   287→          <p className="text-sm text-[#8A7E70]">
   288→            {total} lugar{total !== 1 ? 'es' : ''} en total
   289→          </p>
   290→        </div>
   291→        <Button onClick={openCreateForm} className="bg-[#E1AD01] hover:bg-[#E1AD01]/90 text-[#1F1611]">
   292→          <Plus className="h-4 w-4" />
   293→          Nuevo Lugar
   294→        </Button>
   295→      </div>
   296→
   297→      {/* Filtros + búsqueda + export */}
   298→      <Card className="border-[#5C3A21]/10 shadow-sm">
   299→        <CardContent className="p-4 space-y-3">
   300→          <div className="flex flex-col lg:flex-row gap-3">
   301→            {/* Search */}
   302→            <div className="relative flex-1">
   303→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-[#8A7E70]" />
   304→              <Input
   305→                placeholder="Buscar por nombre, dirección o responsable…"
   306→                value={search}
   307→                onChange={(e) => setSearch(e.target.value)}
   308→                className="pl-9 border-[#5C3A21]/15 bg-white"
   309→              />
   310→            </div>
   311→            {/* Estado filter */}
   312→            <Select
   313→              value={statusFilter}
   314→              onValueChange={(v) => setStatusFilter(v as 'all' | 'active' | 'inactive')}
   315→            >
   316→              <SelectTrigger className="w-full lg:w-44 border-[#5C3A21]/15">
   317→                <SelectValue placeholder="Estado" />
   318→              </SelectTrigger>
   319→              <SelectContent>
   320→                <SelectItem value="all">Todos</SelectItem>
   321→                <SelectItem value="active">Activos</SelectItem>
   322→                <SelectItem value="inactive">Inactivos</SelectItem>
   323→              </SelectContent>
   324→            </Select>
   325→          </div>

   338→            </Button>
   339→            <Button
   340→              size="sm"
   341→              variant="outline"
   342→              onClick={() => handleExport('pdf')}
   343→              className="border-[#5C3A21]/20 text-[#5C3A21]"
   344→            >
   345→              <FileText className="h-3.5 w-3.5" />
   346→              PDF
   347→            </Button>
   348→            <Button
   349→              size="sm"
   350→              variant="outline"

   355→              Word
   356→            </Button>
   357→            <Button
   358→              size="sm"
   359→              variant="outline"
   360→              onClick={() => handleExport('excel')}
   361→              className="border-[#5C3A21]/20 text-[#5C3A21]"

   375→              <Loader2 className="h-6 w-6 animate-spin text-[#E1AD01]" />
   376→              <span className="ml-2 text-sm text-[#8A7E70]">Cargando lugares…</span>
   377→            </div>
   378→          ) : places.length === 0 ? (
   379→            <div className="py-16 text-center">
   380→              <MapPin className="h-10 w-10 mx-auto mb-2 text-[#8A7E70]/40" />

   382→              <p className="text-xs text-[#8A7E70]/70 mt-1">
   383→                Probá cambiar los filtros o crear uno nuevo.
   384→              </p>

   387→            <div className="overflow-x-auto">
   388→              <Table>

   390→                  <TableRow className="bg-[#FBF1DC] border-[#5C3A21]/15">

   392→                    <TableHead className="w-[100px]">Imagen</TableHead>

   476→                                onClick={() => setDeletePlace(p)}
   477→                                className="text-[#B91C1C] focus:text-[#B91C1C]"
   478→                              >
   479→                                <Trash2 className="h-4 w-4 mr-2" />
   480→                                Eliminar
   481→                              </DropdownMenuItem>
   482→                            </DropdownMenuContent>
   483→                          </DropdownMenu>

   495→      {/* ====== Modal: Crear/Editar Lugar ====== */}

   501→        onSaved={() => {

   503→          loadPlaces()
   504→        }}
   505→      />
   506→
   507→      {/* ====== Modal: Confirmar Eliminación ====== */}
   508→      <Dialog open={!!deletePlace} onOpenChange={(o) => !o && setDeletePlace(null)}>
   509→        <DialogContent className="sm:max-w-md">

   522→              onClick={handleDelete}

   525→              <Trash2 className="h-4 w-4" />
   526→              Eliminar
   527→            </Button>
   528→          </DialogFooter>
   529→        </DialogContent>
   530→      </Dialog>
   531→    </div>
   532→  )
   533→}

   551→// Sub-componente: Form Dialog (Crear/Editar) — 4 secciones
   552→// ============================================================
   553→function PlaceFormDialog({
   554→  open,
   555→  mode,
   556→  place,
   557→  onClose,
   558→  onSaved,
   559→}: {
   560→  open: boolean
   561→  mode: FormMode
   562→  place: CmPlaceRecord | null
   563→  onClose: () => void
   564→  onSaved: () => void
   565→}) {
   566→  const [form, setForm] = React.useState<PlaceFormState>(emptyForm)
   567→  const [saving, setSaving] = React.useState(false)
   568→  const [error, setError] = React.useState<string | null>(null)
   569→
   570→  // Sincronizar form cuando se abre el modal
   571→  React.useEffect(() => {
   572→    if (open) {
   573→      if (mode === 'edit' && place) {
   574→        setForm(formFromPlace(place))
   575→      } else {
   576→        setForm(emptyForm())
   577→      }
   578→      setError(null)
   579→    }
   580→  }, [open, mode, place])
   581→
   582→  // Helper para actualizar un campo
   583→  const setField = <K extends keyof PlaceFormState>(key: K, value: PlaceFormState[K]) => {
   584→    setForm((prev) => ({ ...prev, [key]: value }))
   585→  }
   586→
   587→  const handleSubmit = async (e: React.FormEvent) => {
   588→    e.preventDefault()
   589→    setError(null)
   590→
   591→    // Validaciones
   592→    if (!form.name.trim()) return setError('El nombre es obligatorio')
   593→    if (form.contactEmail.trim() && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.contactEmail.trim())) {
   594→      return setError('El email de contacto es inválido')
   595→    }
   596→
   597→    setSaving(true)
   598→    try {
   599→      const url = mode === 'create'
   600→        ? '/api/cocina-movil/places'
   601→        : `/api/cocina-movil/places/${place!.id}`
   602→      const body: Record<string, unknown> = {
   603→        name: form.name.trim(),
   604→        description: form.description.trim() || null,
   605→        contactName: form.contactName.trim() || null,
   606→        contactPhone: form.contactPhone.trim() || null,
   607→        contactEmail: form.contactEmail.trim() || null,
   608→        // Domicilio
   609→        address: form.address.trim() || null,
   610→        country: form.country.trim() || null,
   611→        province: form.province.trim() || null,
   612→        department: form.department.trim() || null,
   613→        municipality: form.municipality.trim() || null,
   614→        location: form.location,
   615→        // Imagen
   616→        image: form.image || null,
   617→        // Costos
   618→        isOwned: form.isOwned,
   619→        rentCost: form.isOwned ? null : toNumberOrNull(form.rentCost),
   620→        utilityCost: toNumberOrNull(form.utilityCost),
   621→        otherFixedCosts: toNumberOrNull(form.otherFixedCosts),
   622→      }
   623→      if (mode === 'create') {
   624→        body.isActive = true
   625→      } else {
   626→        body.isActive = form.isActive
   627→      }
   628→
   629→      const res = await fetch(url, {
   630→        method: mode === 'create' ? 'POST' : 'PUT',
   631→        headers: { 'Content-Type': 'application/json' },
   632→        body: JSON.stringify(body),
   633→      })
   634→      const data = await res.json().catch(() => ({}))
   635→      if (!res.ok) throw new Error(data.error || 'HTTP ' + res.status)
   636→
   637→      toast.success(mode === 'create' ? 'Lugar creado' : 'Lugar actualizado')
   638→      onSaved()
   639→    } catch (err) {
   640→      setError(err instanceof Error ? err.message : 'Error al guardar')
   641→    } finally {
   642→      setSaving(false)
   643→    }
   644→  }
   645→
   646→  return (
   647→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   648→      <DialogContent className="sm:max-w-2xl max-h-[90vh] flex flex-col overflow-hidden p-0">
   649→        <DialogHeader className="px-6 pt-6 pb-4 shrink-0">
   650→          <DialogTitle className="text-[#5C3A21]">
   651→            {mode === 'create' ? 'Nuevo Lugar' : 'Editar Lugar'}
   652→          </DialogTitle>
   653→          <DialogDescription>
   654→            {mode === 'create'
   655→              ? 'Creá un nuevo lugar para la Cocina Móvil'
   656→              : place
   657→              ? `Editando «${place.name}»`
   658→              : 'Editar lugar'}
   659→          </DialogDescription>
   660→        </DialogHeader>
   661→
   662→        <form onSubmit={handleSubmit} className="flex-1 flex flex-col overflow-hidden">
   663→          <div className="flex-1 overflow-y-auto px-6 space-y-5 pb-6">
   664→          {error && (
   665→            <div className="text-sm text-[#B91C1C] bg-[#B91C1C]/5 border border-[#B91C1C]/20 rounded-md px-3 py-2">
   666→              {error}
   667→            </div>
   668→          )}
   669→
   670→          {/* ====== Sección 1: Datos del Lugar ====== */}
   671→          <div className="space-y-3">
   672→            <SectionHeader num={1} title="Datos del Lugar" />
   673→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 pl-8 sm:pl-0">
   674→              <div className="space-y-1.5 sm:col-span-2">
   675→                <Label className="text-[#5C3A21]">Nombre *</Label>
   676→                <Input
   677→                  value={form.name}
   678→                  onChange={(e) => setField('name', e.target.value)}
   679→                  placeholder="Ej: Cocina Central, Carrito Móvil Centro…"
   680→                  className="border-[#5C3A21]/15"
   681→                  autoFocus
   682→                />
   683→              </div>
   684→              <div className="space-y-1.5 sm:col-span-2">
   685→                <Label className="text-[#5C3A21]">Descripción</Label>
   686→                <Textarea
   687→                  value={form.description}
   688→                  onChange={(e) => setField('description', e.target.value)}
   689→                  placeholder="Breve descripción del lugar, su función, etc."
   690→                  className="border-[#5C3A21]/15 min-h-[72px]"
   691→                  rows={2}
   692→                />
   693→              </div>
   694→              <div className="space-y-1.5">
   695→                <Label className="text-[#5C3A21]">Responsable</Label>
   696→                <Input
   697→                  value={form.contactName}
   698→                  onChange={(e) => setField('contactName', e.target.value)}
   699→                  placeholder="Nombre y apellido"
   700→                  className="border-[#5C3A21]/15"
   701→                />
   702→              </div>
   703→              <div className="space-y-1.5">
   704→                <Label className="text-[#5C3A21]">Teléfono</Label>
   705→                <Input
   706→                  value={form.contactPhone}
   707→                  onChange={(e) => setField('contactPhone', e.target.value)}
   708→                  placeholder="3754-419324"
   709→                  className="border-[#5C3A21]/15"
   710→                />
   711→              </div>
   712→              <div className="space-y-1.5 sm:col-span-2">
   713→                <Label className="text-[#5C3A21]">Email de contacto</Label>
   714→                <Input
   715→                  type="email"
   716→                  value={form.contactEmail}
   717→                  onChange={(e) => setField('contactEmail', e.target.value)}
   718→                  placeholder="responsable@ejemplo.com"
   719→                  className="border-[#5C3A21]/15"
   720→                />
   721→              </div>
   722→            </div>
   723→          </div>
   724→
   725→          <Separator className="bg-[#5C3A21]/15" />
   726→
   727→          {/* ====== Sección 2: Domicilio ====== */}
   728→          <div className="space-y-3">
   729→            <SectionHeader num={2} title="Domicilio" />
   730→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 pl-8 sm:pl-0">
   731→              <div className="space-y-1.5 sm:col-span-2">
   732→                <Label className="text-[#5C3A21]">Dirección</Label>
   733→                <Input
   734→                  value={form.address}
   735→                  onChange={(e) => setField('address', e.target.value)}
   736→                  placeholder="Calle, número, piso, depto…"
   737→                  className="border-[#5C3A21]/15"
   738→                />
   739→              </div>
   740→              <div className="space-y-1.5">
   741→                <Label className="text-[#5C3A21]">País</Label>
   742→                <Input
   743→                  value={form.country}
   744→                  onChange={(e) => setField('country', e.target.value)}
   745→                  className="border-[#5C3A21]/15"
   746→                />
   747→              </div>
   748→              <div className="space-y-1.5">
   749→                <Label className="text-[#5C3A21]">Provincia</Label>
   750→                <Input
   751→                  value={form.province}
   752→                  onChange={(e) => setField('province', e.target.value)}
   753→                  className="border-[#5C3A21]/15"
   754→                />
   755→              </div>
   756→              <div className="space-y-1.5">
   757→                <Label className="text-[#5C3A21]">Departamento</Label>
   758→                <Input
   759→                  value={form.department}
   760→                  onChange={(e) => setField('department', e.target.value)}
   761→                  placeholder="Capital, Iguazú…"
   762→                  className="border-[#5C3A21]/15"
   763→                />
   764→              </div>

   769→                  onChange={(e) => setField('municipality', e.target.value)}
   770→                  placeholder="Posadas, Oberá…"
   771→                  className="border-[#5C3A21]/15"
   772→                />
   773→              </div>

   781→              />
   782→            </div>

   789→            <SectionHeader num={3} title="Imagen" />

   793→                onChange={(url) => setField('image', url)}
   794→                uploadUrl="/api/cocina-movil/places/upload-image"
   795→                aspectRatio="4/3"
   796→                label="Imagen del lugar"
   797→                disabled={saving}
   798→              />
   799→            </div>

   806→            <SectionHeader num={4} title="Costos Fijos" />

   808→              <div className="space-y-1.5 sm:col-span-2">
   809→                <Label className="text-[#5C3A21]">Tipo de tenencia</Label>
   810→                <div className="flex items-center gap-3 h-9 px-3 border border-[#5C3A21]/15 rounded-md bg-[#FFF8E7]/30">

   815→                  />

   865→                      onCheckedChange={(c) => setField('isActive', c)}

   876→          </div>
