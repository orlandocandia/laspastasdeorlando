   317→function IngredientFormDialog({ open, mode, item, onClose, onSaved }: { open: boolean; mode: FormMode; item: CmIngredient | null; onClose: () => void; onSaved: () => void }) {
   318→  const [form, setForm] = React.useState({
   319→    name: '', description: '', category: '' as string,
   320→    purchaseUnit: 'kg' as CmUnit, purchasePrice: '', gramsPerUnit: '',
   321→    purchaseUnitType: '' as string, unitsPurchased: '', weightPerUnit: '',
   322→    weightUnit: 'kg' as string, totalPrice: '',
   323→    image: '' as string, isActive: true,
   324→  })
   325→  const [saving, setSaving] = React.useState(false)
   326→  const [error, setError] = React.useState<string | null>(null)
   327→
   328→  React.useEffect(() => {
   329→    if (open) {
   330→      if (mode === 'edit' && item) {
   331→        setForm({
   332→          name: item.name, description: item.description || '', category: item.category || '',
   333→          purchaseUnit: item.purchaseUnit, purchasePrice: String(item.purchasePrice),
   334→          gramsPerUnit: item.gramsPerUnit ? String(item.gramsPerUnit) : '',
   335→          purchaseUnitType: item.purchaseUnitType || '', unitsPurchased: item.unitsPurchased ? String(item.unitsPurchased) : '',
   336→          weightPerUnit: item.weightPerUnit ? String(item.weightPerUnit) : '',
   337→          weightUnit: item.weightUnit || 'kg', totalPrice: item.totalPrice ? String(item.totalPrice) : '',
   338→          image: item.image || '', isActive: item.isActive,
   339→        })
   340→      } else {
   341→        setForm({ name: '', description: '', category: '', purchaseUnit: 'kg', purchasePrice: '', gramsPerUnit: '', image: '', isActive: true })
   342→      }
   343→      setError(null)
   344→    }
   345→  }, [open, mode, item])
   346→
   347→  const setField = <K extends keyof typeof form>(key: K, value: (typeof form)[K]) => setForm((f) => ({ ...f, [key]: value }))
   348→
   349→  const handleSubmit = async (e: React.FormEvent) => {
   350→    e.preventDefault()
   351→    setError(null)
   352→    if (!form.name.trim()) return setError('El nombre es obligatorio')
   353→    // Validate totalPrice (the new field the user fills in)
   354→    // purchasePrice is auto-calculated in the backend from totalPrice
   355→    const totalPrice = parseFloat(form.totalPrice)
   356→    if (form.totalPrice && (isNaN(totalPrice) || totalPrice < 0)) {
   357→      return setError('El precio total debe ser un número válido')
   358→    }
   359→
   360→    setSaving(true)
   361→    try {
   362→      const body: Record<string, unknown> = {
   363→        name: form.name, description: form.description || null,
   364→        category: form.category || null, purchaseUnit: form.purchaseUnit,
   365→        gramsPerUnit: form.gramsPerUnit ? parseFloat(form.gramsPerUnit) : null,
   366→        purchaseUnitType: form.purchaseUnitType || null,
   367→        unitsPurchased: form.unitsPurchased ? parseFloat(form.unitsPurchased) : null,
   368→        weightPerUnit: form.weightPerUnit ? parseFloat(form.weightPerUnit) : null,
   369→        weightUnit: form.weightUnit || null,
   370→        totalPrice: form.totalPrice ? parseFloat(form.totalPrice) : null,
   371→        image: form.image || null, isActive: form.isActive,
   372→      }
   373→      const url = mode === 'create' ? '/api/cocina-movil/ingredients' : `/api/cocina-movil/ingredients/${item!.id}`
   374→      const res = await fetch(url, { method: mode === 'create' ? 'POST' : 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
   375→      const data = await res.json().catch(() => ({}))
   376→      if (!res.ok) throw new Error(data.error || 'HTTP ' + res.status)
   377→      toast.success(mode === 'create' ? 'Materia prima creada' : 'Materia prima actualizada')
   378→      onSaved()
   379→    } catch (err) { setError(err instanceof Error ? err.message : 'Error al guardar') }
   380→    finally { setSaving(false) }
   381→  }
   382→
   383→  return (
   384→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   385→      <DialogContent className="sm:max-w-2xl max-h-[90vh] flex flex-col overflow-hidden p-0">
   386→        <DialogHeader className="px-6 pt-6 pb-4 shrink-0">
   387→          <DialogTitle className="text-[#5C3A21]">{mode === 'create' ? 'Nueva Materia Prima' : 'Editar Materia Prima'}</DialogTitle>
   388→          <DialogDescription>{mode === 'create' ? 'Agregar ingrediente para recetas y producción' : `Editando: ${item?.name}`}</DialogDescription>
   389→        </DialogHeader>
   390→        <form onSubmit={handleSubmit} className="flex-1 flex flex-col overflow-hidden">
   391→          <div className="flex-1 overflow-y-auto px-6 space-y-4 pb-6">
   392→          {error && <div className="text-sm text-[#B91C1C] bg-[#B91C1C]/5 border border-[#B91C1C]/20 rounded-md px-3 py-2">{error}</div>}
   393→
   394→          <div className="flex items-center gap-3"><div className="h-7 w-7 rounded-full bg-[#5C3A21] text-[#FFF8E7] flex items-center justify-center text-xs font-bold">1</div><h3 className="text-sm font-semibold text-[#5C3A21]">Datos de la Materia Prima</h3></div>
   395→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 pl-10">
   396→            <div className="space-y-1.5"><Label className="text-[#5C3A21]">Nombre *</Label><Input value={form.name} onChange={(e) => setField('name', e.target.value)} className="border-[#5C3A21]/15" autoFocus /></div>
   397→            <div className="space-y-1.5"><Label className="text-[#5C3A21]">Categoría</Label><Select value={form.category || 'none'} onValueChange={(v) => setField('category', v === 'none' ? '' : v)}><SelectTrigger className="border-[#5C3A21]/15"><SelectValue placeholder="Sin categoría" /></SelectTrigger><SelectContent><SelectItem value="none">— Sin categoría —</SelectItem>{CATEGORIES.map((c) => <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>)}</SelectContent></Select></div>
   398→            <div className="space-y-1.5 sm:col-span-2"><Label className="text-[#5C3A21]">Descripción</Label><Textarea value={form.description} onChange={(e) => setField('description', e.target.value)} className="border-[#5C3A21]/15" rows={2} /></div>
   399→            <div className="space-y-1.5"><Label className="text-[#5C3A21]">Tipo de Unidad de Compra</Label><Select value={form.purchaseUnitType || 'none'} onValueChange={(v) => {
   400→              const newType = v === 'none' ? '' : v
   401→              setField('purchaseUnitType', newType)
   402→              // Auto-set weight unit based on type
   403→              if (newType) setField('weightUnit', getDefaultWeightUnit(newType))
   404→              // For kg_suelto/litro_suelto: weightPerUnit = 1 (quantity IS the weight)
   405→              if (isWeightFieldHidden(newType)) setField('weightPerUnit', '1')
   406→            }}><SelectTrigger className="border-[#5C3A21]/15"><SelectValue placeholder="Sin tipo" /></SelectTrigger><SelectContent><SelectItem value="none">— Sin tipo —</SelectItem>{PURCHASE_TYPES.map((t) => <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>)}</SelectContent></Select></div>
   407→            <div className="space-y-1.5"><Label className="text-[#5C3A21]">{getQuantityLabel(form.purchaseUnitType)}</Label><Input type="number" step="0.01" min="0" value={form.unitsPurchased} onChange={(e) => setField('unitsPurchased', e.target.value)} placeholder={form.purchaseUnitType === 'kg_suelto' ? 'ej: 5 (5 kg sueltos)' : form.purchaseUnitType === 'litro_suelto' ? 'ej: 3 (3 litros sueltos)' : 'ej: 1 (bulto), 6 (botellas)'} className="border-[#5C3A21]/15" /></div>
   408→            {!isWeightFieldHidden(form.purchaseUnitType) && (
   409→              <div className="space-y-1.5 grid grid-cols-2 gap-2">
   410→                <div className="space-y-1.5"><Label className="text-[#5C3A21]">{getWeightLabel(form.purchaseUnitType)}</Label><Input type="number" step="0.01" min="0" value={form.weightPerUnit} onChange={(e) => setField('weightPerUnit', e.target.value)} placeholder="ej: 25" className="border-[#5C3A21]/15" /></div>
   411→                <div className="space-y-1.5"><Label className="text-[#5C3A21]">Unidad</Label><Select value={form.weightUnit} onValueChange={(v) => setField('weightUnit', v)}><SelectTrigger className="border-[#5C3A21]/15"><SelectValue /></SelectTrigger><SelectContent>{WEIGHT_UNITS.map((u) => <SelectItem key={u.value} value={u.value}>{u.label}</SelectItem>)}</SelectContent></Select></div>
   412→              </div>
   413→            )}
   414→            <div className="space-y-1.5"><Label className="text-[#5C3A21]">Precio Total Pagado ($)</Label><Input type="number" step="0.01" min="0" value={form.totalPrice} onChange={(e) => setField('totalPrice', e.target.value)} placeholder="ej: 450" className="border-[#5C3A21]/15" /></div>
   415→            {form.unitsPurchased && form.totalPrice && parseFloat(form.unitsPurchased) > 0 && parseFloat(form.totalPrice) > 0 && (
   416→              <div className="bg-[#E1AD01]/10 border border-[#E1AD01]/30 rounded-md p-3 space-y-1">
   417→                <p className="text-xs font-semibold text-[#7a5c00]">📊 Cálculo automático:</p>
   418→                {(() => {
   419→                  const qty = parseFloat(form.unitsPurchased)
   420→                  const wpu = isWeightFieldHidden(form.purchaseUnitType) ? 1 : (form.weightPerUnit ? parseFloat(form.weightPerUnit) : 0)
   421→                  const wu = form.weightUnit || 'kg'
   422→                  const total = qty * wpu
   423→                  const pricePerUnit = total > 0 ? parseFloat(form.totalPrice) / total : 0
   424→                  const totalGrams = wu === 'kg' ? total * 1000 : wu === 'l' ? total * 1000 : wu === 'g' ? total : total
   425→                  return (<>
   426→                    <p className="text-sm text-[#5C3A21]">Precio por {wu}: <strong>{fmtCurrency(pricePerUnit)}</strong></p>
   427→                    <p className="text-sm text-[#5C3A21]">Total: <strong>{total.toLocaleString('es-AR')} {wu}</strong> ({totalGrams.toLocaleString('es-AR')} g)</p>
   428→                  </>)
   429→                })()}
   430→              </div>
   431→            )}
   432→            {mode === 'edit' && (
   433→              <div className="space-y-1.5 flex items-center gap-3 pt-5"><Label className="text-[#5C3A21]">Activo</Label><Switch checked={form.isActive} onCheckedChange={(v) => setField('isActive', v)} /></div>
   434→            )}
   435→          </div>
   436→
   437→          <Separator />
   438→          <div className="flex items-center gap-3"><div className="h-7 w-7 rounded-full bg-[#5C3A21] text-[#FFF8E7] flex items-center justify-center text-xs font-bold">2</div><h3 className="text-sm font-semibold text-[#5C3A21]">Imagen</h3></div>
   439→          <div className="pl-10">
   440→            <ImageUploader value={form.image || null} onChange={(url) => setField('image', url || '')} uploadUrl="/api/cocina-movil/ingredients/upload-image" label="Imagen de la materia prima" aspectRatio="4/3" disabled={saving} />
   441→          </div>
   442→
   443→          </div>
   444→          <div className="shrink-0 bg-white border-t border-[#5C3A21]/10 px-6 py-3 flex justify-end gap-2">
   445→            <Button type="button" variant="outline" onClick={onClose}>Cancelar</Button>
   446→            <Button type="submit" disabled={saving} className="bg-[#E1AD01] hover:bg-[#E1AD01]/90 text-[#1F1611]">{saving ? <Loader2 className="h-4 w-4 animate-spin" /> : null}{mode === 'create' ? 'Crear' : 'Guardar'}</Button>
   447→          </div>
   448→        </form>
   449→      </DialogContent>
   450→    </Dialog>
   451→  )
   452→}
