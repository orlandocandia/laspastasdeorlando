     5→// Helper: detect if running on Turso (libsql:// URL or TURSO_DATABASE_URL env var)
     6→function isTurso(): boolean {
     7→  const url = process.env.DATABASE_URL || ''
     8→  return url.startsWith('libsql://') || url.startsWith('http') || !!process.env.TURSO_DATABASE_URL
     9→}

    11→// Helper: create a libsql client for Turso
    12→function getTursoClient(): Client {
    13→  const url = process.env.TURSO_DATABASE_URL || process.env.DATABASE_URL || ''
    14→  const authToken = process.env.TURSO_AUTH_TOKEN || process.env.DATABASE_AUTH_TOKEN || ''
    15→  return createClient({ url, authToken: authToken || undefined })
    16→}
