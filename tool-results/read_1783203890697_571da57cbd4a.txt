   104→   104→function getFamiliaImagen(familia: Familia, filtro: FiltroHarina): string {

   106→   106→  if (filtro === 'sin_gluten' && familia.imagen_sin_gluten) return familia.imagen_sin_gluten
   107→   107→
   108→   108→  if (familia.nombre === 'Tallarines' && filtro === 'integral') {
   109→   109→    return '/images/familias/tallarinesintegrales.png'
   110→   110→  }
   111→   111→  if (familia.nombre === 'Cintas Anchas' && filtro === 'integral') {
   112→   112→    return '/images/familias/cintasanchasintegrales.png'
   113→   113→  }
   114→   114→  if (familia.nombre === 'Tapas' && filtro === 'integral') {
   115→   115→    return '/images/familias/tapasintegrales.png'
   116→   116→  }
   117→   117→  if (familia.nombre === 'Tapas' && filtro === 'sin_gluten') {
   118→   118→    return '/images/familias/tapassingluten.png'
   119→   119→  }
   120→   120→

   150→   150→export default function Productos({ filtroActivo = 'con_gluten', onFiltroChange }: ProductosProps) {
   151→   151→  // Cache: productos pre-fetcheados por tipo de harina
   152→   152→  const [cache, setCache] = useState<Record<FiltroHarina, ProductoPublico[]>>({
   153→   153→    con_gluten: [],
   154→   154→    integral: [],
   155→   155→    sin_gluten: [],
   156→   156→  })
   157→   157→  const [loading, setLoading] = useState(true)
   158→   158→  const [error, setError] = useState('')
   159→   159→  const [filtro, setFiltro] = useState<FiltroHarina>(filtroActivo)
   160→   160→  // null = no section selected (step 1), 'pastas'|'horneados' = section selected (step 2)
   161→   161→  const [seccionActiva, setSeccionActiva] = useState<Seccion | null>(null)
   162→   162→  const [familiaActiva, setFamiliaActiva] = useState<string | null>(null)
   163→   163→  const [visibleCount, setVisibleCount] = useState(PRODUCTS_PER_PAGE)
   164→   164→  const [productoPromociones, setProductoPromociones] = useState<Record<number, PromocionInfo>>({})
   165→   165→  const [promociones, setPromociones] = useState<PromoBanner[]>([])
   166→   166→  const [soloOfertas, setSoloOfertas] = useState(false)
   167→   167→  const fetchedRef = useRef(false)
   168→   168→  const productosGridRef = useRef<HTMLDivElement>(null)
   169→   169→
   170→   170→  // Pre-fetch all 3 filter types in parallel on mount — no refetch on filter change
   171→   171→  useEffect(() => {
   172→   172→    if (fetchedRef.current) return
   173→   173→    fetchedRef.current = true
   174→   174→
   175→   175→    async function fetchAll() {
   176→   176→      setLoading(true)
   177→   177→      setError('')
   178→   178→      try {
   179→   179→        const [results, promoRes] = await Promise.all([
   180→   180→          Promise.all(
   181→   181→            TODOS_FILTROS.map(async (tipo) => {
   182→   182→              const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   183→   183→              if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   184→   184→              const data = await res.json()
   185→   185→              return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   186→   186→            })
   187→   187→          ),
   188→   188→          fetch('/api/promociones/public').then(async (res) => {
   189→   189→            if (res.ok) {
   190→   190→              const data = await res.json()
   191→   191→              return data
   192→   192→            }
   193→   193→            return { productoPromociones: {}, promociones: [] }
   194→   194→          }).catch(() => ({ productoPromociones: {}, promociones: [] })),
   195→   195→        ])
   196→   196→        setCache((prev) => {
   197→   197→          const next = { ...prev }
   198→   198→          for (const r of results) next[r.tipo] = r.productos
   199→   199→          return next
   200→   200→        })
   201→   201→        setProductoPromociones(promoRes.productoPromociones as Record<number, PromocionInfo> || {})
   202→   202→        setPromociones((promoRes.promociones || []) as PromoBanner[])
   203→   203→      } catch {
   204→   204→        setError('No se pudieron cargar los productos. Intentá más tarde.')
   205→   205→      } finally {
   206→   206→        setLoading(false)
   207→   207→      }
   208→   208→    }
   209→   209→    fetchAll()
   210→   210→  }, [])
   211→   211→
   212→   212→  // Reset visible count when family or section changes
   213→   213→  useEffect(() => {
   214→   214→    setVisibleCount(PRODUCTS_PER_PAGE)
   215→   215→  }, [familiaActiva, seccionActiva])
   216→   216→
   217→   217→  // Sync with parent filtro state — instant, no refetch
   218→   218→  useEffect(() => {
   219→   219→    if (filtroActivo !== filtro) {
   220→   220→      setFiltro(filtroActivo ?? 'con_gluten')
   221→   221→      setFamiliaActiva(null)
   222→   222→    }
   223→   223→  }, [filtroActivo, filtro])
   224→   224→
   225→   225→  const handleFiltroChange = useCallback((nuevoFiltro: FiltroHarina) => {
   226→   226→    setFiltro(nuevoFiltro)
   227→   227→    setFamiliaActiva(null)
   228→   228→    onFiltroChange?.(nuevoFiltro)
   229→   229→  }, [onFiltroChange])
   230→   230→
   231→   231→  const handleSeccionChange = useCallback((nuevaSeccion: Seccion) => {
   232→   232→    setSeccionActiva(nuevaSeccion)
   233→   233→    setFamiliaActiva(null)
   234→   234→    setFiltro('con_gluten') // Reset flour filter when changing section
   235→   235→    setSoloOfertas(false)
   236→   236→    onFiltroChange?.('con_gluten')
   237→   237→  }, [onFiltroChange])
   238→   238→
   239→   239→  const handleVolverASecciones = useCallback(() => {
   240→   240→    setSeccionActiva(null)
   241→   241→    setFamiliaActiva(null)
   242→   242→    setFiltro('con_gluten')
   243→   243→    setSoloOfertas(false)
   244→   244→    onFiltroChange?.('con_gluten')
   245→   245→  }, [onFiltroChange])
   246→   246→
   247→   247→  // Current products from cache — instant swap, no network
   248→   248→  const productos = cache[filtro]
   249→   249→
   250→   250→  // Build families dynamically from product categories, grouped by seccion
   251→   251→  const familiasBySeccion = useMemo(() => {
   252→   252→    const result: Record<string, Familia[]> = { pastas: [], horneados: [], other: [] }
   253→   253→    const seen = new Map<string, Familia>()
   254→   254→
   255→   255→    for (const p of productos) {
   256→   256→      const catNombre = p.categoria.nombre
   257→   257→      const effectiveSeccion = p.seccion ?? p.categoria.seccion
   258→   258→      if (effectiveSeccion !== 'pastas' && effectiveSeccion !== 'horneados') continue
   259→   259→
   260→   260→      const dedupKey = `${catNombre}__${effectiveSeccion}`
   261→   261→      if (!seen.has(dedupKey)) {
   262→   262→        seen.set(dedupKey, {
   263→   263→          nombre: catNombre,
   264→   264→          seccion: effectiveSeccion,
   265→   265→          imagen: p.categoria.imagen || null,
   266→   266→          imagen_integral: p.categoria.imagen_integral || null,
   267→   267→          imagen_sin_gluten: p.categoria.imagen_sin_gluten || null,
   268→   268→          descripcion: p.categoria.descripcion || DESCRIPCIONES_DEFAULT[catNombre] || `Productos de ${catNombre.toLowerCase()}`,
   269→   269→          descripcionDB: p.categoria.descripcion,
   270→   270→          categoriaId: p.categoria.id,
   271→   271→        })
   272→   272→      }
   273→   273→    }
   274→   274→
   275→   275→    const knownOrder = ['Sorrentinos', 'Ñoquis', 'Tallarines', 'Cintas Anchas', 'Ravioles', 'Tapas', 'Empanadas', 'Tartas', 'Pastas frescas', 'Pastas secas', 'Salsas', 'Lasagnas y canelones', 'Postres', 'Facturas', 'Chipás', 'Pizzas']
   276→   276→
   277→   277→    const sorted = Array.from(seen.values()).sort((a, b) => {
   278→   278→      const seccionOrder = { pastas: 0, horneados: 1 }
   279→   279→      const sa = seccionOrder[a.seccion as keyof typeof seccionOrder] ?? 2
   280→   280→      const sb = seccionOrder[b.seccion as keyof typeof seccionOrder] ?? 2
   281→   281→      if (sa !== sb) return sa - sb
   282→   282→
   283→   283→      const ia = knownOrder.indexOf(a.nombre)
   284→   284→      const ib = knownOrder.indexOf(b.nombre)
   285→   285→      if (ia !== -1 && ib !== -1) return ia - ib
   286→   286→      if (ia !== -1) return -1
   287→   287→      if (ib !== -1) return 1
   288→   288→      return a.nombre.localeCompare(b.nombre)
   289→   289→    })
   290→   290→
   291→   291→    for (const familia of sorted) {
   292→   292→      const sec = familia.seccion === 'pastas' ? 'pastas' : familia.seccion === 'horneados' ? 'horneados' : 'other'
   293→   293→      result[sec].push(familia)
   294→   294→    }
   295→   295→
   296→   296→    return result
   297→   297→  }, [productos])
   298→   298→
   299→   299→  // Current section families
   300→   300→  const familias = useMemo(() => {
   301→   301→    if (!seccionActiva) return []
   302→   302→    return familiasBySeccion[seccionActiva] || []
   303→   303→  }, [familiasBySeccion, seccionActiva])
   304→   304→
   305→   305→  // Check which sections have products for current filter
   306→   306→  const seccionesConProductos = useMemo(() => {
   307→   307→    const result: Seccion[] = []
   308→   308→    for (const sec of SECCIONES) {
   309→   309→      const families = familiasBySeccion[sec.key] || []
   310→   310→      if (families.length > 0) result.push(sec.key)
   311→   311→    }
   312→   312→    return result
   313→   313→  }, [familiasBySeccion])
   314→   314→
   315→   315→  // Compute product counts per family
   316→   316→  const familiaData = useMemo(() => {
   317→   317→    const data: Record<string, { count: number; hasProducts: boolean; hasPromos: boolean }> = {}
   318→   318→    for (const familia of familias) {
   319→   319→      const prods = productos.filter((p) => {
   320→   320→        if (p.categoria.nombre !== familia.nombre) return false
   321→   321→        const effSec = p.seccion ?? p.categoria.seccion
   322→   322→        return effSec === familia.seccion
   323→   323→      })
   324→   324→      const hasPromos = prods.some((p) => !!productoPromociones[p.id])
   325→   325→      data[familia.nombre] = {
   326→   326→        count: prods.length,
   327→   327→        hasProducts: prods.length > 0,
   328→   328→        hasPromos,
   329→   329→      }
   330→   330→    }
   331→   331→    return data
   332→   332→  }, [productos, familias, productoPromociones])
   333→   333→
   334→   334→  // Products for the active family — filtered by both family name and effective section
   335→   335→  const productosFamilia = useMemo(() => {
   336→   336→    if (!familiaActiva || !seccionActiva) return []
   337→   337→    const base = productos.filter((p) => {
   338→   338→      if (p.categoria.nombre !== familiaActiva) return false
   339→   339→      const effSec = p.seccion ?? p.categoria.seccion
   340→   340→      return effSec === seccionActiva
   341→   341→    })
   342→   342→    // Apply "solo ofertas" filter
   343→   343→    if (soloOfertas) {
   344→   344→      return base.filter((p) => !!productoPromociones[p.id])
   345→   345→    }
   346→   346→    return base
   347→   347→  }, [productos, familiaActiva, seccionActiva, soloOfertas, productoPromociones])
   348→   348→
   349→   349→  const productosVisibles = useMemo(() => {
   350→   350→    return productosFamilia.slice(0, visibleCount)
   351→   351→  }, [productosFamilia, visibleCount])
   352→   352→
   353→   353→  const hasMore = visibleCount < productosFamilia.length
   354→   354→
   355→   355→  const handleFamiliaClick = useCallback((nombre: string) => {
   356→   356→    setFamiliaActiva((prev) => (prev === nombre ? null : nombre))
   357→   357→  }, [])
   358→   358→
   359→   359→  // Smooth scroll al contenedor de productos cuando se selecciona una familia
   360→   360→  useEffect(() => {
   361→   361→    if (!familiaActiva || !productosGridRef.current) return
   362→   362→    const yOffset = -100
   363→   363→    const y = productosGridRef.current.getBoundingClientRect().top + window.pageYOffset + yOffset
   364→   364→    if (y < window.pageYOffset || y > window.pageYOffset + window.innerHeight - 200) {
   365→   365→      window.scrollTo({ top: y, behavior: 'smooth' })
   366→   366→    }
   367→   367→  }, [familiaActiva])
   368→   368→
   369→   369→  const handleLoadMore = useCallback(() => {
   370→   370→    setVisibleCount((prev) => prev + PRODUCTS_PER_PAGE)
   371→   371→  }, [])
   372→   372→
   373→   373→  // Filter out families with no products
   374→   374→  const familiasVisibles = useMemo(
   375→   375→    () => familias.filter((f) => familiaData[f.nombre]?.hasProducts),
   376→   376→    [familias, familiaData]
   377→   377→  )
   378→   378→
   379→   379→  // Get promos for the active section
   380→   380→  const promosForSection = useMemo(() => {
   381→   381→    if (!seccionActiva) return []
   382→   382→    // Get product IDs in this section
   383→   383→    const sectionProductIds = new Set(
   384→   384→      productos
   385→   385→        .filter((p) => {
   386→   386→          const effSec = p.seccion ?? p.categoria.seccion
   387→   387→          return effSec === seccionActiva
   388→   388→        })
   389→   389→        .map((p) => p.id)
   390→   390→    )
   391→   391→    // Filter promos that have products in this section
   392→   392→    return promociones.filter((promo) => {
   393→   393→      // Check if any product with this promo is in this section
   394→   394→      return Object.entries(productoPromociones).some(
   395→   395→        ([productId, pInfo]) =>
   396→   396→          pInfo.promocionId === promo.id && sectionProductIds.has(Number(productId))
   397→   397→      )
   398→   398→    })
   399→   399→  }, [seccionActiva, productos, promociones, productoPromociones])
   400→   400→
   401→   401→  // How many products have promos in the active family
   402→   402→  const promoCountInFamily = useMemo(() => {
   403→   403→    if (!familiaActiva) return 0
   404→   404→    const base = productos.filter((p) => {
   405→   405→      if (p.categoria.nombre !== familiaActiva) return false
   406→   406→      const effSec = p.seccion ?? p.categoria.seccion
   407→   407→      return effSec === seccionActiva
   408→   408→    })
   409→   409→    return base.filter((p) => !!productoPromociones[p.id]).length
