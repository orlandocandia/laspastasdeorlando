   101→// Imagen dinámica según filtro para familias con variante integral/sin gluten
   102→function getFamiliaImagen(familia: Familia, filtro: FiltroHarina): string {
   103→  if (filtro === 'integral' && familia.imagen_integral) return familia.imagen_integral
   104→  if (filtro === 'sin_gluten' && familia.imagen_sin_gluten) return familia.imagen_sin_gluten
   105→
   106→  if (familia.nombre === 'Tallarines' && filtro === 'integral') {
   107→    return '/images/familias/tallarinesintegrales.png'
   108→  }
   109→  if (familia.nombre === 'Cintas Anchas' && filtro === 'integral') {
   110→    return '/images/familias/cintasanchasintegrales.png'
   111→  }
   112→  if (familia.nombre === 'Tapas' && filtro === 'integral') {
   113→    return '/images/familias/tapasintegrales.png'
   114→  }
   115→  if (familia.nombre === 'Tapas' && filtro === 'sin_gluten') {
   116→    return '/images/familias/tapassingluten.png'
   117→  }
   118→
   119→  if (familia.imagen) return familia.imagen
   120→  return IMAGENES_DEFAULT[familia.nombre] || '/images/placeholder-producto.jpg'
   121→}

   123→const FILTROS: { key: FiltroHarina; label: string; icon: React.ReactNode }[] = [

   129→const TODOS_FILTROS: FiltroHarina[] = ['con_gluten', 'integral', 'sin_gluten']

   131→const PRODUCTS_PER_PAGE = 50

   133→interface ProductosProps {

   138→export default function Productos({ filtroActivo = 'con_gluten', onFiltroChange }: ProductosProps) {
   139→  // Cache: productos pre-fetcheados por tipo de harina
   140→  const [cache, setCache] = useState<Record<FiltroHarina, ProductoPublico[]>>({
   141→    con_gluten: [],
   142→    integral: [],
   143→    sin_gluten: [],
   144→  })
   145→  const [loading, setLoading] = useState(true)
   146→  const [error, setError] = useState('')
   147→  const [filtro, setFiltro] = useState<FiltroHarina>(filtroActivo)
   148→  // null = no section selected (step 1), 'pastas'|'horneados' = section selected (step 2)
   149→  const [seccionActiva, setSeccionActiva] = useState<Seccion | null>(null)
   150→  const [familiaActiva, setFamiliaActiva] = useState<string | null>(null)
   151→  const [visibleCount, setVisibleCount] = useState(PRODUCTS_PER_PAGE)
   152→  const fetchedRef = useRef(false)
   153→  const productosGridRef = useRef<HTMLDivElement>(null)
   154→
   155→  // Pre-fetch all 3 filter types in parallel on mount — no refetch on filter change
   156→  useEffect(() => {
   157→    if (fetchedRef.current) return
   158→    fetchedRef.current = true
   159→
   160→    async function fetchAll() {
   161→      setLoading(true)
   162→      setError('')
   163→      try {
   164→        const results = await Promise.all(
   165→          TODOS_FILTROS.map(async (tipo) => {
   166→            const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   167→            if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   168→            const data = await res.json()
   169→            return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   170→          })
   171→        )
   172→        setCache((prev) => {
   173→          const next = { ...prev }
   174→          for (const r of results) next[r.tipo] = r.productos
   175→          return next
   176→        })
   177→      } catch {
   178→        setError('No se pudieron cargar los productos. Intentá más tarde.')
   179→      } finally {
   180→        setLoading(false)
   181→      }
   182→    }
   183→    fetchAll()
   184→  }, [])
   185→
   186→  // Reset visible count when family or section changes
   187→  useEffect(() => {
   188→    setVisibleCount(PRODUCTS_PER_PAGE)
   189→  }, [familiaActiva, seccionActiva])
   190→
   191→  // Sync with parent filtro state — instant, no refetch
   192→  useEffect(() => {
   193→    if (filtroActivo !== filtro) {
   194→      setFiltro(filtroActivo ?? 'con_gluten')
   195→      setFamiliaActiva(null)
   196→    }
   197→  }, [filtroActivo, filtro])
   198→
   199→  const handleFiltroChange = useCallback((nuevoFiltro: FiltroHarina) => {
   200→    setFiltro(nuevoFiltro)
   201→    setFamiliaActiva(null)
   202→    onFiltroChange?.(nuevoFiltro)
   203→  }, [onFiltroChange])
   204→
   205→  const handleSeccionChange = useCallback((nuevaSeccion: Seccion) => {
   206→    setSeccionActiva(nuevaSeccion)
   207→    setFamiliaActiva(null)
   208→    setFiltro('con_gluten') // Reset flour filter when changing section
   209→    onFiltroChange?.('con_gluten')
   210→  }, [onFiltroChange])
   211→
   212→  const handleVolverASecciones = useCallback(() => {
   213→    setSeccionActiva(null)
   214→    setFamiliaActiva(null)
   215→    setFiltro('con_gluten')
   216→    onFiltroChange?.('con_gluten')
   217→  }, [onFiltroChange])
   218→
   219→  // Current products from cache — instant swap, no network
   220→  const productos = cache[filtro]
   221→
   222→  // Build families dynamically from product categories, grouped by seccion
   223→  // Effective section: producto.seccion ?? categoria.seccion
   224→  // Products with null effective section are hidden (requirement 7)
   225→  // A category can appear in BOTH sections if it has products of both types
   226→  const familiasBySeccion = useMemo(() => {
   227→    const result: Record<string, Familia[]> = { pastas: [], horneados: [], other: [] }
   228→    // Dedup by (categoria nombre + effective seccion) so the same category
   229→    // can appear once in pastas and once in horneados
   230→    const seen = new Map<string, Familia>()
   231→
   232→    for (const p of productos) {
   233→      const catNombre = p.categoria.nombre
   234→      const effectiveSeccion = p.seccion ?? p.categoria.seccion
   235→      // Skip products without an effective section (requirement 7)
   236→      if (effectiveSeccion !== 'pastas' && effectiveSeccion !== 'horneados') continue
   237→
   238→      const dedupKey = `${catNombre}__${effectiveSeccion}`
   239→      if (!seen.has(dedupKey)) {
   240→        seen.set(dedupKey, {
   241→          nombre: catNombre,
   242→          seccion: effectiveSeccion,
   243→          imagen: p.categoria.imagen || null,
   244→          imagen_integral: p.categoria.imagen_integral || null,
   245→          imagen_sin_gluten: p.categoria.imagen_sin_gluten || null,
   246→          descripcion: p.categoria.descripcion || DESCRIPCIONES_DEFAULT[catNombre] || `Productos de ${catNombre.toLowerCase()}`,
   247→          descripcionDB: p.categoria.descripcion,
   248→          categoriaId: p.categoria.id,
   249→        })
   250→      }
   251→    }
   252→
   253→    const knownOrder = ['Sorrentinos', 'Ñoquis', 'Tallarines', 'Cintas Anchas', 'Ravioles', 'Tapas', 'Empanadas', 'Tartas', 'Pastas frescas', 'Pastas secas', 'Salsas', 'Lasagnas y canelones', 'Postres', 'Facturas', 'Chipás', 'Pizzas']
   254→
   255→    const sorted = Array.from(seen.values()).sort((a, b) => {
   256→      const seccionOrder = { pastas: 0, horneados: 1 }
   257→      const sa = seccionOrder[a.seccion as keyof typeof seccionOrder] ?? 2
   258→      const sb = seccionOrder[b.seccion as keyof typeof seccionOrder] ?? 2
   259→      if (sa !== sb) return sa - sb
   260→
   261→      const ia = knownOrder.indexOf(a.nombre)
   262→      const ib = knownOrder.indexOf(b.nombre)
   263→      if (ia !== -1 && ib !== -1) return ia - ib
   264→      if (ia !== -1) return -1
   265→      if (ib !== -1) return 1
   266→      return a.nombre.localeCompare(b.nombre)
   267→    })
   268→
   269→    for (const familia of sorted) {
   270→      const sec = familia.seccion === 'pastas' ? 'pastas' : familia.seccion === 'horneados' ? 'horneados' : 'other'
   271→      result[sec].push(familia)
   272→    }
   273→
   274→    return result
   275→  }, [productos])
   276→
   277→  // Current section families — only families with an explicit section
   278→  // (products with null effective section are hidden, requirement 7)
   279→  const familias = useMemo(() => {
   280→    if (!seccionActiva) return []
   281→    return familiasBySeccion[seccionActiva] || []
   282→  }, [familiasBySeccion, seccionActiva])
   283→
   284→  // Check which sections have products for current filter
   285→  const seccionesConProductos = useMemo(() => {
   286→    const result: Seccion[] = []
   287→    for (const sec of SECCIONES) {
   288→      const families = familiasBySeccion[sec.key] || []
   289→      if (families.length > 0) result.push(sec.key)
   290→    }
   291→    return result
   292→  }, [familiasBySeccion])
   293→
   294→  // Compute product counts per family — must respect effective section
   295→  // so a category that appears in both sections only counts its
   296→  // products from the corresponding section
   297→  const familiaData = useMemo(() => {
   298→    const data: Record<string, { count: number; hasProducts: boolean }> = {}
   299→    for (const familia of familias) {
   300→      const prods = productos.filter((p) => {
   301→        if (p.categoria.nombre !== familia.nombre) return false
   302→        const effSec = p.seccion ?? p.categoria.seccion
   303→        return effSec === familia.seccion
   304→      })
   305→      data[familia.nombre] = {
   306→        count: prods.length,
   307→        hasProducts: prods.length > 0,
   308→      }
   309→    }
   310→    return data
   311→  }, [productos, familias])
   312→
   313→  // Products for the active family — filtered by both family name and effective section
   314→  const productosFamilia = useMemo(() => {
   315→    if (!familiaActiva || !seccionActiva) return []
   316→    return productos.filter((p) => {
   317→      if (p.categoria.nombre !== familiaActiva) return false
   318→      const effSec = p.seccion ?? p.categoria.seccion
   319→      return effSec === seccionActiva
   320→    })
   321→  }, [productos, familiaActiva, seccionActiva])
   322→
   323→  const productosVisibles = useMemo(() => {
   324→    return productosFamilia.slice(0, visibleCount)
   325→  }, [productosFamilia, visibleCount])
   326→
   327→  const hasMore = visibleCount < productosFamilia.length
   328→
   329→  const handleFamiliaClick = useCallback((nombre: string) => {
   330→    setFamiliaActiva((prev) => (prev === nombre ? null : nombre))
   331→  }, [])
   332→
   333→  // Smooth scroll al contenedor de productos cuando se selecciona una familia
   334→  useEffect(() => {
   335→    if (!familiaActiva || !productosGridRef.current) return
   336→    const yOffset = -100
   337→    const y = productosGridRef.current.getBoundingClientRect().top + window.pageYOffset + yOffset
   338→    if (y < window.pageYOffset || y > window.pageYOffset + window.innerHeight - 200) {
   339→      window.scrollTo({ top: y, behavior: 'smooth' })
   340→    }
   341→  }, [familiaActiva])
   342→
   343→  const handleLoadMore = useCallback(() => {
   344→    setVisibleCount((prev) => prev + PRODUCTS_PER_PAGE)
   345→  }, [])
   346→
   347→  // Filter out families with no products
   348→  const familiasVisibles = useMemo(
   349→    () => familias.filter((f) => familiaData[f.nombre]?.hasProducts),
   350→    [familias, familiaData]
   351→  )
   352→
   353→  // Retry: re-fetch all
   354→  const handleRetry = useCallback(async () => {
   355→    fetchedRef.current = false
   356→    setLoading(true)
   357→    setError('')
   358→    try {
   359→      const results = await Promise.all(
   360→        TODOS_FILTROS.map(async (tipo) => {
   361→          const res = await fetch(`/api/productos-terminados/public?tipo=${tipo}`)
   362→          if (!res.ok) throw new Error(`Error cargando ${tipo}`)
   363→          const data = await res.json()
   364→          return { tipo, productos: (data.productos || []) as ProductoPublico[] }
   365→        })
   366→      )
   367→      setCache((prev) => {
   368→        const next = { ...prev }
   369→        for (const r of results) next[r.tipo] = r.productos
   370→        return next
   371→      })
   372→    } catch {
   373→      setError('Error de conexión. Verificá tu internet e intentá de nuevo.')
   374→    } finally {
   375→      setLoading(false)
   376→    }
   377→  }, [])
   378→
   379→  const seccionConfig = seccionActiva ? SECCIONES.find((s) => s.key === seccionActiva)! : null
   380→
   381→  return (
   382→    <section id="productos" className="min-h-screen flex flex-col justify-start py-12 sm:py-16 md:py-20 bg-crema">
   383→      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 w-full">
   384→        {/* Title */}
   385→        <div className="text-center mb-12">
   386→          <h2 className="text-3xl sm:text-4xl font-bold text-marron">
   387→            Nuestros <span className="text-rojo">Productos</span>
   388→          </h2>
   389→          <div className="h-1 w-20 bg-mostaza mx-auto mt-4 rounded-full" />
   390→          <p className="text-muted-foreground mt-3 max-w-lg mx-auto text-sm sm:text-base">
   391→            Elaborados con ingredientes frescos y de calidad. Elegí la sección que más te guste.
   392→          </p>
   393→        </div>
   394→
   395→        {/* ═══════════════════════════════════════════════════════════
   396→            STATE 1: No section selected → Show section hero cards
   397→            ═══════════════════════════════════════════════════════════ */}
   398→        {!seccionActiva && (
   399→          <>
   400→            {loading ? (
   401→              <div className="flex flex-col md:flex-row justify-center items-stretch gap-6 md:gap-10 w-full max-w-5xl mx-auto">
   402→                {Array.from({ length: 2 }).map((_, i) => (
   403→                  <div key={i} className="rounded-2xl border border-marron/10 bg-white animate-pulse flex-1 max-w-sm md:max-w-md lg:max-w-lg flex flex-col items-center p-6 sm:p-8 md:p-10 gap-5">
   404→                    <div className="w-40 h-40 md:w-48 md:h-48 lg:w-56 lg:h-56 bg-muted rounded-full" />
   405→                    <div className="h-7 w-32 bg-muted rounded" />
   406→                    <div className="h-10 w-36 bg-muted rounded-full" />
   407→                  </div>
   408→                ))}
   409→              </div>
   410→            ) : error ? (
   411→              <div className="text-center py-12">
   412→                <span className="text-5xl mb-4 block">⚠️</span>
   413→                <p className="text-muted-foreground mb-4">{error}</p>
   414→                <Button
   415→                  variant="outline"
   416→                  onClick={handleRetry}
   417→                  className="border-mostaza text-marron hover:bg-mostaza hover:text-marron"
   418→                >
   419→                  Reintentar
   420→                </Button>
   421→              </div>
   422→            ) : (
   423→              <div className="flex flex-col md:flex-row justify-center items-stretch gap-6 md:gap-10 w-full max-w-5xl mx-auto">
   424→                {SECCIONES.map((sec) => {
   425→                  const hasProducts = seccionesConProductos.includes(sec.key)
   426→                  return (
   427→                    <button
   428→                      key={sec.key}
   429→                      onClick={() => handleSeccionChange(sec.key)}
   430→                      className={`
   431→                        group relative rounded-2xl border-2 bg-white
   432→                        flex flex-col items-center text-center
   433→                        flex-1 max-w-sm md:max-w-md lg:max-w-lg
   434→                        p-6 sm:p-8 md:p-10 gap-5
   435→                        transition-all duration-300 cursor-pointer
   436→                        border-marron/10 hover:border-mostaza hover:shadow-xl hover:scale-[1.02]
   437→                        shadow-md
   438→                      `}
   439→                    >
   440→                      {/* Imagen circular — responsive: 160 / 192 / 224 px */}
   441→                      <div className="w-40 h-40 md:w-48 md:h-48 lg:w-56 lg:h-56 rounded-full overflow-hidden border-4 border-mostaza/20 flex-shrink-0 transition-all duration-300 group-hover:scale-105 group-hover:shadow-[0_8px_24px_rgba(0,0,0,0.18)] group-hover:border-mostaza/40">
   442→                        <Image
   443→                          src={sec.imagen}
   444→                          alt={sec.label}
   445→                          width={224}
   446→                          height={224}
   447→                          className="w-full h-full object-cover object-center"
   448→                        />
   449→                      </div>
   450→
   451→                      {/* Título */}
   452→                      <h3 className="text-xl md:text-2xl lg:text-3xl font-bold text-marron">
   453→                        {sec.label}
   454→                      </h3>
   455→
   456→                      {/* Botón CTA con efecto profesional al hover */}
   457→                      <span className="inline-flex items-center gap-1.5 bg-mostaza text-marron text-sm md:text-base font-semibold rounded-full px-6 py-2.5 shadow-sm transition-all duration-300 ease-out group-hover:-translate-y-0.5 group-hover:shadow-lg group-hover:brightness-105 group-hover:gap-2.5">
   458→                        Ver productos
   459→                        <ChevronRight className="h-4 w-4 transition-transform duration-300 group-hover:translate-x-0.5" />
   460→                      </span>
   461→
   462→                      {/* "Próximamente" badge */}
   463→                      {!hasProducts && (
   464→                        <div className="absolute top-3 right-3 bg-mostaza/90 text-marron text-xs rounded-full px-3 py-1 font-bold shadow-sm">
   465→                          Próximamente
   466→                        </div>
   467→                      )}
